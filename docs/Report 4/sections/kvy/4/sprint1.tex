\subsection{ForumController Class}

\paragraph{ClassMethod}
\begin{ClassMethodTable}
01 & \code{ForumController(forumService: ForumService)} &
A method to initialize a new instance of the controller with the forum service.
\newline -- \code{forumService}: \code{ForumService} -- the service that owns category and post use cases.
\newline -- \textit{Returns:} \code{void} -- constructor, no return value. \\ \hline

02 & \code{getForumCategories()} &
A method to retrieve all active forum categories.
\newline -- \textit{Returns:} \code{Promise<ForumCategoryListResponseType>} -- the category list produced by the service.
\newline -- Route: \code{GET /forums/categories}; public endpoint. \\ \hline

03 & \code{getTopForumCategories(}\newline\code{query: { limit: number })} &
A method to retrieve the categories with the largest number of public posts.
\newline -- \code{query.limit}: \code{number} -- maximum number of categories, validated by \code{TopLimitSchema} with default 3 and range 1--10.
\newline -- \textit{Returns:} \code{Promise<ForumCategoryTopListResponseType>} -- ranked category list.
\newline -- Route: \code{GET /forums/categories/top}; public endpoint. \\ \hline

04 & \code{getTopActiveUsers(}\newline\code{query: { limit: number })} &
A method to retrieve the users with the most forum activity.
\newline -- \code{query.limit}: \code{number} -- maximum number of users, validated by \code{TopLimitSchema} with default 3 and range 1--10.
\newline -- \textit{Returns:} \code{Promise<ForumTopActiveUserListResponseType>} -- users with post and comment counts.
\newline -- Route: \code{GET /forums/users/top}; public endpoint. \\ \hline

05 & \code{getForumCategoryById(id: number)} &
A method to retrieve one forum category by its identifier.
\newline -- \code{id}: \code{number} -- category identifier parsed by \code{ParseIntPipe}.
\newline -- \textit{Returns:} \code{Promise<ForumCategoryDetailResponseType>} -- category detail with post count.
\newline -- Route: \code{GET /forums/categories/:id}; public endpoint. \\ \hline

06 & \code{getForumCategoryBySlug(slug: string)} &
A method to retrieve one forum category by its slug.
\newline -- \code{slug}: \code{string} -- unique category slug.
\newline -- \textit{Returns:} \code{Promise<ForumCategoryDetailResponseType>} -- category detail with post count.
\newline -- Route: \code{GET /forums/categories/by-slug/:slug}; public endpoint. \\ \hline

07 & \code{getForumPostLists(}\newline\code{filter: ForumPostFilterType)} &
A method to retrieve a paginated, filterable list of public posts.
\newline -- \code{filter}: \code{ForumPostFilterType} -- page, limit, optional categoryId, optional userId, and optional search text, validated by \code{ForumPostFilterDto}.
\newline -- \textit{Returns:} \code{Promise<ForumPostListResponseType>} -- posts plus pagination metadata.
\newline -- Route: \code{GET /forums/posts}; public endpoint. \\ \hline

08 & \code{getTopInteractedPosts(}\newline\code{query: { limit: number; categoryId?: number })} &
A method to retrieve the most interacted posts from the previous seven days.
\newline -- \code{query.limit}: \code{number} -- maximum number of posts, default 3.
\newline -- \code{query.categoryId}: \code{number | undefined} -- optional category filter.
\newline -- \textit{Returns:} \code{Promise<ForumTopPostListResponseType>} -- ranked posts with like and comment counts.
\newline -- Route: \code{GET /forums/posts/top}; public endpoint. \\ \hline

09 & \code{viewForumPostDetail(id: number)} &
A method to retrieve the public detail of one post.
\newline -- \code{id}: \code{number} -- post identifier.
\newline -- \textit{Returns:} \code{Promise<ViewForumPostDetailResponseType>} -- post, author summary, and category summary.
\newline -- Route: \code{GET /forums/posts/:id}; public endpoint. \\ \hline

10 & \code{getForumPostBySlug(slug: string)} &
A method to retrieve the public detail of one post by slug.
\newline -- \code{slug}: \code{string} -- unique post slug.
\newline -- \textit{Returns:} \code{Promise<ViewForumPostDetailResponseType>} -- post detail.
\newline -- Route: \code{GET /forums/posts/by-slug/:slug}; public endpoint. \\ \hline

11 & \code{createForumPost(}\newline\code{userId: number, body: CreateForumPostType)} &
A method to create a forum post for the authenticated user.
\newline -- \code{userId}: \code{number} -- author identifier injected by \code{@UserActive('userId')}.
\newline -- \code{body}: \code{CreateForumPostType} -- categoryId, title, content, and optional moderation result, validated by \code{CreateForumPostDto}.
\newline -- \textit{Returns:} \code{Promise<ForumPostType>} -- the created post.
\newline -- Route: \code{POST /forums/posts}; authenticated endpoint. \\ \hline

12 & \code{deleteForumPost(}\newline\code{userId: number, roleName: string,}\newline\code{id: number)} &
A method to soft-delete a post when the caller is its owner or an administrator.
\newline -- \code{userId}: \code{number} -- current user identifier.
\newline -- \code{roleName}: \code{string} -- current role name, used for the administrator override.
\newline -- \code{id}: \code{number} -- post identifier.
\newline -- \textit{Returns:} \code{Promise<ForumPostType>} -- the soft-deleted post.
\newline -- Route: \code{DELETE /forums/posts/:id}; authenticated endpoint. \\ \hline
\end{ClassMethodTable}

\paragraph{ClassAttribute}
\begin{ClassAttributeTable}
01 & \code{forumService} & \code{ForumService} -- private readonly service dependency. \\ \hline
\end{ClassAttributeTable}

\subsection{ForumService Class}

\paragraph{ClassMethod}
\begin{ClassMethodTable}
01 & \code{ForumService(forumRepository: ForumRepository)} &
A method to initialize the service with the forum repository.
\newline -- \code{forumRepository}: \code{ForumRepository} -- persistence dependency.
\newline -- \textit{Returns:} \code{void} -- constructor, no return value. \\ \hline

02 & \code{getForumCategories()} &
A method to load all active categories.
\newline -- \textit{Returns:} \code{Promise<ForumCategoryListResponseType>} -- category list with public post counts.
\newline -- Failure is mapped to \code{FailedToLoadForumCategoriesException}. \\ \hline

03 & \code{getTopForumCategories(limit: number = 3)} &
A method to load the categories with the most public posts.
\newline -- \code{limit}: \code{number} -- maximum number of categories; default 3.
\newline -- \textit{Returns:} \code{Promise<ForumCategoryTopListResponseType>} -- ranked categories.
\newline -- Failure is mapped to \code{FailedToLoadForumCategoriesException}. \\ \hline

04 & \code{getTopActiveUsers(limit: number = 5)} &
A method to load the most active users by post and comment counts.
\newline -- \code{limit}: \code{number} -- maximum number of users; default 5.
\newline -- \textit{Returns:} \code{Promise<ForumTopActiveUserListResponseType>} -- user activity summaries.
\newline -- The current implementation reuses \code{FailedToLoadForumCategoriesException} for this failure path. \\ \hline

05 & \code{getForumCategoryById(id: number)} &
A method to load one category by identifier.
\newline -- \code{id}: \code{number} -- category identifier.
\newline -- \textit{Returns:} \code{Promise<ForumCategoryDetailResponseType>} -- category detail.
\newline -- Prisma known errors are mapped to \code{FailedToLoadForumCategoriesException}. \\ \hline

06 & \code{getForumCategoryBySlug(slug: string)} &
A method to resolve a category slug and return its detail.
\newline -- \code{slug}: \code{string} -- category slug.
\newline -- \textit{Returns:} \code{Promise<ForumCategoryDetailResponseType>} -- category detail.
\newline -- Throws \code{ForumCategoryNotFoundException} when the slug does not exist. \\ \hline

07 & \code{getForumPostBySlug(slug: string)} &
A method to resolve a post slug and return its public detail.
\newline -- \code{slug}: \code{string} -- post slug.
\newline -- \textit{Returns:} \code{Promise<ViewForumPostDetailResponseType>} -- post detail.
\newline -- Throws \code{ForumPostNotFoundException} when the slug does not exist. \\ \hline

08 & \code{getForumPostLists(filter: ForumPostFilterType)} &
A method to build a paginated post list from the repository result.
\newline -- \code{filter}: \code{ForumPostFilterType} -- page, limit, category, author, and search filters.
\newline -- \textit{Returns:} \code{Promise<ForumPostListResponseType>} -- posts and pagination metadata, including \code{totalPages}.
\newline -- Prisma known errors are mapped to \code{FailedToLoadForumPostsException}. \\ \hline

09 & \code{createForumPost(}\newline\code{userId: number, body: CreateForumPostType)} &
A method to validate the optional category and create a post.
\newline -- \code{userId}: \code{number} -- author identifier.
\newline -- \code{body}: \code{CreateForumPostType} -- post data and optional AI moderation result.
\newline -- \textit{Returns:} \code{Promise<ForumPostType>} -- the created post.
\newline -- Throws \code{ForumCategoryNotFoundException} for an unknown category and \code{FailedToCreateForumPostException} for a persistence failure. \\ \hline

10 & \code{viewForumPostDetail(id: number)} &
A method to load the public detail of a post.
\newline -- \code{id}: \code{number} -- post identifier.
\newline -- \textit{Returns:} \code{Promise<ViewForumPostDetailResponseType>} -- post, author, and category summary.
\newline -- Prisma known errors are mapped to \code{FailedToViewForumPostException}. \\ \hline

11 & \code{updateForumPost(}\newline\code{id: number, userId: number,}\newline\code{body: UpdateForumPostType)} &
A method to update a post after checking existence and ownership.
\newline -- \code{id}: \code{number} -- post identifier.
\newline -- \code{userId}: \code{number} -- current user identifier.
\newline -- \code{body}: \code{UpdateForumPostType} -- updated category, title, and content.
\newline -- \textit{Returns:} \code{Promise<ForumPostType>} -- the updated post.
\newline -- Throws \code{ForumPostNotFoundException} or \code{ForumPostNotOwnedException}. The controller route is currently commented out. \\ \hline

12 & \code{deleteForumPost(}\newline\code{id: number, userId: number,}\newline\code{roleName: string)} &
A method to authorize and soft-delete a post.
\newline -- \code{id}: \code{number} -- post identifier.
\newline -- \code{userId}: \code{number} -- current user identifier.
\newline -- \code{roleName}: \code{string} -- current role name; \code{ADMIN} bypasses the owner check.
\newline -- \textit{Returns:} \code{Promise<ForumPostType>} -- the soft-deleted post.
\newline -- Throws \code{ForumPostNotFoundException}, \code{ForumPostNotOwnedException}, or \code{FailedToDeleteForumPostException}. \\ \hline

13 & \code{getTopInteractedPosts(}\newline\code{limit: number = 3,}\newline\code{categoryId?: number)} &
A method to return the most interacted posts from the last seven days.
\newline -- \code{limit}: \code{number} -- maximum number of posts; default 3.
\newline -- \code{categoryId}: \code{number | undefined} -- optional category restriction.
\newline -- \textit{Returns:} \code{Promise<ForumTopPostListResponseType>} -- ranked posts.
\newline -- Failure is mapped to \code{FailedToLoadForumPostsException}. \\ \hline
\end{ClassMethodTable}

\paragraph{ClassAttribute}
\begin{ClassAttributeTable}
01 & \code{forumRepository} & \code{ForumRepository} -- private readonly persistence dependency. \\ \hline
\end{ClassAttributeTable}

\subsection{ForumRepository Class}

\paragraph{ClassMethod}
\begin{ClassMethodTable}
01 & \code{ForumRepository(prisma: PrismaService)} &
A method to initialize the repository with the shared Prisma service.
\newline -- \code{prisma}: \code{PrismaService} -- pooled database client.
\newline -- \textit{Returns:} \code{void} -- constructor, no return value. \\ \hline

02 & \code{getForumCategories()} &
A method to query active categories ordered by name.
\newline -- \textit{Returns:} \code{Promise<Array<Pick<ForumCategoryType, 'id' | 'name' | 'slug' | 'description' | 'createdAt' | 'updatedAt' | 'postCount'>>>} -- category rows with public post counts.
\newline -- Only non-deleted posts with moderation status \code{APPROVED} are counted. \\ \hline

03 & \code{getTopForumCategories(limit: number = 3)} &
A method to query active categories ordered by public post count.
\newline -- \code{limit}: \code{number} -- maximum number of rows; default 3.
\newline -- \textit{Returns:} the same category projection as \code{getForumCategories()}, ranked by \code{posts._count} descending. \\ \hline

04 & \code{getTopActiveUsers(limit: number = 5)} &
A method to query non-deleted, non-banned users ordered by post count.
\newline -- \code{limit}: \code{number} -- maximum number of rows; default 5.
\newline -- \textit{Returns:} \code{Promise<ForumTopActiveUserType[]>} -- user id, display name, avatar, post count, and comment count.
\newline -- Comment counts include only non-deleted comments. \\ \hline

05 & \code{findForumCategoryById(id: number)} &
A method to check whether an active category exists.
\newline -- \code{id}: \code{number} -- category identifier.
\newline -- \textit{Returns:} \code{Promise<Pick<ForumCategoryType, 'id'> | null>} -- the id when found, otherwise \code{null}. \\ \hline

06 & \code{getForumCategoryById(id: number)} &
A method to load one active category with its post count.
\newline -- \code{id}: \code{number} -- category identifier.
\newline -- \textit{Returns:} category projection with \code{postCount}.
\newline -- Throws \code{ForumCategoryNotFoundException} when no active row exists. \\ \hline

07 & \code{findForumCategoryBySlug(slug: string)} &
A method to resolve an active category by slug.
\newline -- \code{slug}: \code{string} -- category slug.
\newline -- \textit{Returns:} \code{Promise<Pick<ForumCategoryType, 'id'> | null>} -- the id when found, otherwise \code{null}. \\ \hline

08 & \code{getForumPostLists(filter: ForumPostFilterType)} &
A method to query a page of public posts and the matching total in one transaction.
\newline -- \code{filter}: \code{ForumPostFilterType} -- pagination, category, author, and case-insensitive title search.
\newline -- \textit{Returns:} \code{Promise<{ posts: ForumPostWithUserType[]; total: number }>} -- mapped post rows and total count.
\newline -- Visibility requires \code{deletedAt = null} and moderation status \code{APPROVED}. \\ \hline

09 & \code{createForumPost(}\newline\code{categoryId: number | null, userId: number,}\newline\code{title: string, content: string,}\newline\code{moderation?: ForumModerationType)} &
A method to insert a post and generate its slug.
\newline -- \code{categoryId}: \code{number | null} -- optional category.
\newline -- \code{userId}: \code{number} -- author identifier.
\newline -- \code{title}, \code{content}: \code{string} -- post text.
\newline -- \code{moderation}: \code{ForumModerationType | undefined} -- optional AI result; status defaults to \code{PENDING}.
\newline -- \textit{Returns:} \code{Promise<ForumPostType>} -- the inserted post. \\ \hline

10 & \code{viewForumPostDetail(id: number)} &
A method to query one public post with author and category summaries.
\newline -- \code{id}: \code{number} -- post identifier.
\newline -- \textit{Returns:} \code{Promise<ViewForumPostDetailResponseType>} -- post detail.
\newline -- Throws \code{ForumPostNotFoundException} when the post is missing, deleted, or not approved. \\ \hline

11 & \code{findForumPostById(id: number)} &
A method to load a non-deleted post for ownership checks.
\newline -- \code{id}: \code{number} -- post identifier.
\newline -- \textit{Returns:} \code{Promise<ForumPostType | null>} -- the post or \code{null}. \\ \hline

12 & \code{findForumPostBySlug(slug: string)} &
A method to resolve a public post slug.
\newline -- \code{slug}: \code{string} -- post slug.
\newline -- \textit{Returns:} \code{Promise<Pick<ForumPostType, 'id'> | null>} -- the id when found, otherwise \code{null}. \\ \hline

13 & \code{updateForumPost(}\newline\code{id: number, data: UpdateForumPostType)} &
A method to update a non-deleted post and regenerate its slug from the title.
\newline -- \code{id}: \code{number} -- post identifier.
\newline -- \code{data}: \code{UpdateForumPostType} -- category, title, and content.
\newline -- \textit{Returns:} \code{Promise<ForumPostType>} -- the updated post. \\ \hline

14 & \code{softDeleteForumPost(id: number)} &
A method to set \code{deletedAt} on a non-deleted post.
\newline -- \code{id}: \code{number} -- post identifier.
\newline -- \textit{Returns:} \code{Promise<ForumPostType>} -- the soft-deleted post. \\ \hline

15 & \code{getTopInteractedPosts(}\newline\code{limit: number = 3,}\newline\code{categoryId?: number)} &
A method to query posts created during the previous seven days and rank them by interactions.
\newline -- \code{limit}: \code{number} -- maximum number of rows; default 3.
\newline -- \code{categoryId}: \code{number | undefined} -- optional category restriction.
\newline -- \textit{Returns:} \code{Promise<ForumTopPostType[]>} -- posts with like count, comment count, and \code{interactionScore}.
\newline -- Ordering is by likes, then comments, then creation time, all descending. \\ \hline
\end{ClassMethodTable}

\paragraph{ClassAttribute}
\begin{ClassAttributeTable}
01 & \code{prisma} & \code{PrismaService} -- private readonly database client. \\ \hline
\end{ClassAttributeTable}
