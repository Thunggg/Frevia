\subsection{BannersAdminController Class}

\paragraph{ClassMethod}
\begin{ClassMethodTable}
01 & \code{BannersAdminController(service: BannersAdminService)} &
A method to initialize the controller with the banner administration service.
\newline -- \code{service}: \code{BannersAdminService} -- service that owns banner use cases.
\newline -- \textit{Returns:} \code{void} -- constructor, no return value. \\ \hline

02 & \code{listBanners(query: BannerAdminQueryDto)} &
A method to return a filtered, paginated banner list.
\newline -- \code{query}: \code{BannerAdminQueryDto} -- page, limit, search, position, active, and sort options.
\newline -- \textit{Returns:} \code{Promise<BannerAdminListResponseType>} -- banners and pagination.
\newline -- Route: \code{GET /admin/banners}. \\ \hline

03 & \code{createBanner(body: CreateBannerBodyDto)} &
A method to create an advertisement banner.
\newline -- \code{body}: \code{CreateBannerBodyDto} -- title, image URL, link, position, active state, and dates.
\newline -- \textit{Returns:} \code{Promise<BannerAdminDetailResponseType>} -- the created banner.
\newline -- Route: \code{POST /admin/banners}. \\ \hline

04 & \code{updateBanner(id: number, body: UpdateBannerBodyDto)} &
A method to update a banner.
\newline -- \code{id}: \code{number} -- banner identifier.
\newline -- \code{body}: \code{UpdateBannerBodyDto} -- optional banner fields.
\newline -- \textit{Returns:} \code{Promise<BannerAdminDetailResponseType>} -- the updated banner.
\newline -- Route: \code{PATCH /admin/banners/:id}. \\ \hline

05 & \code{deleteBanner(id: number)} &
A method to permanently delete a banner.
\newline -- \code{id}: \code{number} -- banner identifier.
\newline -- \textit{Returns:} \code{Promise<BannerAdminDeleteResponseType>} -- a success message.
\newline -- Route: \code{DELETE /admin/banners/:id}. \\ \hline

06 & \code{uploadImage(file?: Express.Multer.File)} &
A method to upload a banner image and return its URL.
\newline -- \code{file}: \code{Express.Multer.File | undefined} -- multipart image field named \code{file}.
\newline -- \textit{Returns:} \code{Promise<{ imageUrl: string; publicId: string }>} -- the stored image reference.
\newline -- Route: \code{POST /admin/banners/upload-image}; multipart field \code{file}; maximum 10 MB. \\ \hline

07 & \code{getBannerDetail(id: number)} &
A method to return one banner.
\newline -- \code{id}: \code{number} -- banner identifier.
\newline -- \textit{Returns:} \code{Promise<BannerAdminDetailResponseType>} -- the banner detail.
\newline -- Route: \code{GET /admin/banners/:id}. \\ \hline
\end{ClassMethodTable}

\paragraph{ClassAttribute}
\begin{ClassAttributeTable}
01 & \code{service} & \code{BannersAdminService} -- private readonly banner administration service. \\ \hline
\end{ClassAttributeTable}

\subsection{BannersAdminService Class}

\paragraph{ClassMethod}
\begin{ClassMethodTable}
01 & \code{BannersAdminService(}\newline\code{repository: BannersAdminRepository,}\newline\code{cloudinaryService: CloudinaryService)} &
A method to initialize the service with persistence and image storage.
\newline -- \code{repository}: \code{BannersAdminRepository} -- banner persistence dependency.
\newline -- \code{cloudinaryService}: \code{CloudinaryService} -- remote image storage.
\newline -- \textit{Returns:} \code{void} -- constructor, no return value. \\ \hline

02 & \code{uploadImage(file?: Express.Multer.File)} &
A method to validate an image and store it on Cloudinary or the local upload directory.
\newline -- \code{file}: \code{Express.Multer.File | undefined} -- multipart image.
\newline -- \textit{Returns:} \code{Promise<{ imageUrl: string; publicId: string }>} -- the stored image reference.
\newline -- Throws \code{BannerImageRequiredException}, \code{BannerImageInvalidTypeException}, \code{BannerImageTooLargeException}, or \code{FailedToUploadBannerImageException}. \\ \hline

03 & \code{listBanners(query: BannerAdminQueryType)} &
A method to load a banner page.
\newline -- \code{query}: \code{BannerAdminQueryType} -- list filters.
\newline -- \textit{Returns:} \code{Promise<BannerAdminListResponseType>} -- banners and pagination.
\newline -- Failure is mapped to \code{FailedToLoadBannerListException}. \\ \hline

04 & \code{getBannerDetail(id: number)} &
A method to load one banner.
\newline -- \code{id}: \code{number} -- banner identifier.
\newline -- \textit{Returns:} \code{Promise<BannerAdminDetailResponseType>} -- the banner detail.
\newline -- Throws \code{BannerAdminNotFoundException} or \code{FailedToLoadBannerDetailException}. \\ \hline

05 & \code{createBanner(body: BannerCreateBodyType)} &
A method to create a banner after validating its position.
\newline -- \code{body}: \code{BannerCreateBodyType} -- banner fields.
\newline -- \textit{Returns:} \code{Promise<BannerAdminDetailResponseType>} -- the created banner.
\newline -- Throws \code{BannerPositionTakenException} or \code{FailedToCreateBannerException}. \\ \hline

06 & \code{updateBanner(id: number, body: BannerUpdateBodyType)} &
A method to update a banner after validating its position.
\newline -- \code{id}: \code{number} -- banner identifier.
\newline -- \code{body}: \code{BannerUpdateBodyType} -- optional banner fields.
\newline -- \textit{Returns:} \code{Promise<BannerAdminDetailResponseType>} -- the updated banner.
\newline -- Throws \code{BannerAdminNotFoundException}, \code{BannerPositionTakenException}, or \code{FailedToUpdateBannerException}. \\ \hline

07 & \code{deleteBanner(id: number)} &
A method to permanently delete a banner.
\newline -- \code{id}: \code{number} -- banner identifier.
\newline -- \textit{Returns:} \code{Promise<BannerAdminDeleteResponseType>} -- a success message.
\newline -- Throws \code{BannerAdminNotFoundException} or \code{FailedToDeleteBannerException}. \\ \hline
\end{ClassMethodTable}

\paragraph{ClassAttribute}
\begin{ClassAttributeTable}
01 & \code{repository} & \code{BannersAdminRepository} -- private readonly persistence dependency. \\ \hline
02 & \code{cloudinaryService} & \code{CloudinaryService} -- private readonly image storage dependency. \\ \hline
\end{ClassAttributeTable}

\newline -- \code{query}: \code{SkillAdminQueryType} -- list filters.

A method to load one skill.
\newline -- \code{id}: \code{number} -- skill identifier.

A method to generate a unique slug and insert a skill.

\newline -- Throws \code{SkillAdminNotFoundException} or \code{SkillNameAlreadyExistsException}. \\ \hline

\newline -- \textit{Returns:} \code{Promise<string>} -- the unique slug. \\ \hline
