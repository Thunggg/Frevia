\subsection{AdminExpertProfileController Class}

\paragraph{ClassMethod}
\begin{ClassMethodTable}
01 & \code{AdminExpertProfileController(service: ExpertProfileService)} &
A method to initialize the controller with the expert profile service.
\newline -- \code{service}: \code{ExpertProfileService} -- service that owns expert profile use cases.
\newline -- \textit{Returns:} \code{void} -- constructor, no return value. \\ \hline

02 & \code{update(}\newline\code{userId: number, adminId: number,}\newline\code{body: AdminUpdateExpertProfileDto)} &
A method for an administrator to update an expert profile directly.
\newline -- \code{userId}: \code{number} -- expert account identifier.
\newline -- \code{adminId}: \code{number} -- current administrator, recorded on superseded revisions.
\newline -- \code{body}: \code{AdminUpdateExpertProfileDto} -- display name, bio, title, expertise, experience, education, certifications, website, and active state.
\newline -- \textit{Returns:} \code{Promise<MessageResType>} -- a success message.
\newline -- Route: \code{PATCH /users/:id/expert-profile}. \\ \hline
\end{ClassMethodTable}

\paragraph{ClassAttribute}
\begin{ClassAttributeTable}
01 & \code{service} & \code{ExpertProfileService} -- private readonly expert profile service. \\ \hline
\end{ClassAttributeTable}

\subsection{ExpertProfileService Class}

\paragraph{ClassMethod}
\begin{ClassMethodTable}
01 & \code{ExpertProfileService(}\newline\code{repository: ExpertProfileRepository,}\newline\code{revisions: ProfileRevisionService)} &
A method to initialize the service with persistence and revision handling.
\newline -- \code{repository}: \code{ExpertProfileRepository} -- expert profile persistence.
\newline -- \code{revisions}: \code{ProfileRevisionService} -- self-service revision workflow.
\newline -- \textit{Returns:} \code{void} -- constructor, no return value. \\ \hline

02 & \code{cleanList(values: string[])} &
A private method to trim, de-duplicate, and preserve the first casing of a string list.
\newline -- \code{values}: \code{string[]} -- raw list values.
\newline -- \textit{Returns:} \code{string[]} -- the cleaned list. \\ \hline

03 & \code{sanitize(input)} &
A private method to clean list fields and normalize the website value.
\newline -- \code{input}: \code{UpdateExpertProfileType} -- profile input.
\newline -- \textit{Returns:} \code{UpdateExpertProfileType} -- the sanitized input. \\ \hline

04 & \code{context(userId: number)} &
A private method to load an expert account and verify its Expert role.
\newline -- \code{userId}: \code{number} -- target account.
\newline -- \textit{Returns:} \code{Promise<user context>} -- account, profile, and roles.
\newline -- Throws \code{ExpertProfileNotFoundException} or \code{NotAnExpertAccountException}. \\ \hline

05 & \code{getMine(userId: number)} &
A method to return the authenticated expert's profile.
\newline -- \code{userId}: \code{number} -- current expert account.
\newline -- \textit{Returns:} \code{Promise<ExpertProfileType>} -- profile and expert fields.
\newline -- Used by the self-service profile page, not by the admin endpoint. \\ \hline

06 & \code{submitUpdate(userId, input)} &
A method to submit a self-service expert profile revision.
\newline -- \code{userId}: \code{number} -- current expert account.
\newline -- \code{input}: \code{UpdateExpertProfileType} -- requested profile changes.
\newline -- \textit{Returns:} \code{Promise<revision result>} -- the pending revision.
\newline -- Throws \code{ExpertProfilePendingRevisionException} when a revision is already pending. \\ \hline

07 & \code{updateByAdmin(userId, adminId, input)} &
A method to update an expert profile directly and supersede pending revisions.
\newline -- \code{userId}: \code{number} -- target expert account.
\newline -- \code{adminId}: \code{number} -- current administrator.
\newline -- \code{input}: \code{AdminUpdateExpertProfileType} -- administrative profile changes.
\newline -- \textit{Returns:} \code{Promise<MessageResType>} -- a success message.
\newline -- Throws \code{ExpertProfileNotFoundException} or \code{NotAnExpertAccountException}. \\ \hline
\end{ClassMethodTable}

\paragraph{ClassAttribute}
\begin{ClassAttributeTable}
01 & \code{repository} & \code{ExpertProfileRepository} -- private readonly persistence dependency. \\ \hline
02 & \code{revisions} & \code{ProfileRevisionService} -- private readonly revision workflow dependency. \\ \hline
\end{ClassAttributeTable}

\subsection{ExpertProfileRepository Class}

\paragraph{ClassMethod}
\begin{ClassMethodTable}
01 & \code{ExpertProfileRepository(prisma: PrismaService)} &
A method to initialize the repository with the shared Prisma client.
\newline -- \code{prisma}: \code{PrismaService} -- database client.
\newline -- \textit{Returns:} \code{void} -- constructor, no return value. \\ \hline

02 & \code{findByUserId(userId: number)} &
A method to load an expert account with its profile and roles.
\newline -- \code{userId}: \code{number} -- target account.
\newline -- \textit{Returns:} \code{Promise<user context | null>} -- the account graph, or \code{null}. \\ \hline

03 & \code{updateDirect(userId, adminId, input, profileCompletionPercent)} &
A method to supersede pending revisions, update the profile, and upsert the expert row in one transaction.
\newline -- \code{userId}: \code{number} -- target expert account.
\newline -- \code{adminId}: \code{number} -- reviewing administrator.
\newline -- \code{input}: \code{AdminUpdateExpertProfileType} -- sanitized profile changes.
\newline -- \code{profileCompletionPercent}: \code{number} -- recalculated completeness.
\newline -- \textit{Returns:} \code{Promise<void>} -- no value.
\newline -- Creates a system notification when a pending revision was superseded. \\ \hline

04 & \code{hasPendingRevision(userId: number)} &
A method to check for a pending expert revision.
\newline -- \code{userId}: \code{number} -- target account.
\newline -- \textit{Returns:} \code{Promise<{ id: number } | null>} -- the pending revision, or \code{null}. \\ \hline

05 & \code{findPublicMany(query: PublicExpertQueryType)} &
A method to query active public expert profiles.
\newline -- \code{query}: \code{PublicExpertQueryType} -- expertise, search, and pagination filters.
\newline -- \textit{Returns:} \code{Promise<public expert page>} -- public expert cards. \\ \hline

06 & \code{findPublicById(id: number)} &
A method to load one active public expert profile.
\newline -- \code{id}: \code{number} -- expert identifier.
\newline -- \textit{Returns:} \code{Promise<public expert | null>} -- the public profile, or \code{null}. \\ \hline
\end{ClassMethodTable}

\paragraph{ClassAttribute}
\begin{ClassAttributeTable}
01 & \code{prisma} & \code{PrismaService} -- private readonly database client. \\ \hline
\end{ClassAttributeTable}

\subsection{JobCategoriesAdminRepository Class}

\paragraph{ClassMethod}
\begin{ClassMethodTable}
01 & \code{JobCategoriesAdminRepository(prisma: PrismaService)} &
A method to initialize the repository with the shared Prisma client.
\newline -- \code{prisma}: \code{PrismaService} -- database client.
\newline -- \textit{Returns:} \code{void} -- constructor, no return value. \\ \hline

02 & \code{listJobCategories(query: JobCategoryAdminQueryType)} &
A method to query categories with search, status, deleted-state, sorting, and pagination.
\newline -- \code{query}: \code{JobCategoryAdminQueryType} -- list filters.
\newline -- \textit{Returns:} \code{Promise<JobCategoryAdminListResponseType>} -- categories and pagination.
\newline -- Search matches name, slug, or description. \\ \hline

03 & \code{getJobCategoryDetail(id: number)} &
A method to load one category with its job count.
\newline -- \code{id}: \code{number} -- category identifier.
\newline -- \textit{Returns:} \code{Promise<JobCategoryAdminDetailResponseType | null>} -- the category, or \code{null}. \\ \hline

04 & \code{createJobCategory(data)} &
A method to generate a unique slug and insert a category.
\newline -- \code{data}: \code{AdminCreateJobCategoryBodyType} -- category fields.
\newline -- \textit{Returns:} \code{Promise<JobCategoryAdminDetailResponseType>} -- the created category.
\newline -- Throws \code{JobCategoryNameAlreadyExistsException}. \\ \hline

05 & \code{updateJobCategory(id, data)} &
A method to update a category and regenerate its slug when the name changes.
\newline -- \code{id}: \code{number} -- category identifier.
\newline -- \code{data}: \code{AdminUpdateJobCategoryBodyType} -- optional category fields.
\newline -- \textit{Returns:} \code{Promise<JobCategoryAdminDetailResponseType>} -- the updated category.
\newline -- Throws \code{JobCategoryAdminNotFoundException} or \code{JobCategoryNameAlreadyExistsException}. \\ \hline

06 & \code{deleteJobCategory(id: number)} &
A method to check active job references and set the delete timestamp.
\newline -- \code{id}: \code{number} -- category identifier.
\newline -- \textit{Returns:} \code{Promise<JobCategoryAdminDeleteResponseType>} -- a success message.
\newline -- Throws \code{JobCategoryAdminNotFoundException} or \code{JobCategoryHasJobsException}. \\ \hline

07 & \code{restoreJobCategory(id: number)} &
A method to clear the delete timestamp after checking name conflicts.
\newline -- \code{id}: \code{number} -- category identifier.
\newline -- \textit{Returns:} \code{Promise<JobCategoryAdminDetailResponseType>} -- the restored category.
\newline -- Throws \code{JobCategoryAdminNotFoundException} or \code{JobCategoryNameAlreadyExistsException}. \\ \hline

08 & \code{generateUniqueSlug(name: string)} &
A private method to derive a slug and append a numeric suffix on collision.
\newline -- \code{name}: \code{string} -- category name.
\newline -- \textit{Returns:} \code{Promise<string>} -- the unique slug. \\ \hline
\end{ClassMethodTable}
