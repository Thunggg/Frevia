\subsection{ContractController Class}

\paragraph{ClassMethod}
\begin{ClassMethodTable}
01 & \code{ContractController(}\newline\code{contractService: ContractService)} &
A method to initialize the controller with the contract service dependency.
\newline -- \code{contractService}: \code{ContractService} -- service that encapsulates contract lifecycle use cases.
\newline -- \textit{Returns:} \code{void} -- constructor, no return value. \\ \hline

02 & \code{getContractList(}\newline\code{userId: number, roleName: string,}\newline\code{query: GetContractListQueryDTO)} &
A method to retrieve a paginated and filterable list of contracts for the current user.
\newline -- \code{userId}: \code{number} -- caller user identifier injected by \code{@UserActive('userId')}.
\newline -- \code{roleName}: \code{string} -- active role of the caller for role-scoped visibility.
\newline -- \code{query}: \code{GetContractListQueryDTO} -- search keyword, status, date bounds, page, and limit.
\newline -- \textit{Returns:} \code{Promise<GetContractListResponseDTO>} -- list of contracts and pagination metadata.
\newline -- Route: \code{GET /contracts}; authenticated endpoint. \\ \hline

03 & \code{getContractDetail(}\newline\code{userId: number, roleName: string,}\newline\code{contractId: number)} &
A method to fetch comprehensive contract details including participants, job info, and financial metrics.
\newline -- \code{userId}: \code{number} -- caller user identifier.
\newline -- \code{roleName}: \code{string} -- caller role name.
\newline -- \code{contractId}: \code{number} -- contract identifier parsed by \code{ParseIntPipe}.
\newline -- \textit{Returns:} \code{Promise<GetContractDetailResponseDTO>} -- complete contract details.
\newline -- Route: \code{GET /contracts/:id}; authenticated endpoint. \\ \hline

04 & \code{createContract(}\newline\code{userId: number,}\newline\code{body: CreateContractBodyDTO)} &
A method to create a new contract from an accepted (hired) proposal.
\newline -- \code{userId}: \code{number} -- client identifier injected by \code{@UserActive('userId')}.
\newline -- \code{body}: \code{CreateContractBodyDTO} -- proposalId, contract title, description, and terms.
\newline -- \textit{Returns:} \code{Promise<CreateContractResponseDTO>} -- the newly created contract record.
\newline -- Route: \code{POST /contracts}; authenticated endpoint. \\ \hline

05 & \code{updateContract(}\newline\code{userId: number, contractId: number,}\newline\code{body: UpdateContractBodyDTO)} &
A method to update terms of an unfinalized contract prior to mutual signing.
\newline -- \code{userId}: \code{number} -- client identifier.
\newline -- \code{contractId}: \code{number} -- contract identifier.
\newline -- \code{body}: \code{UpdateContractBodyDTO} -- revised title, description, budget, or deadline.
\newline -- \textit{Returns:} \code{Promise<UpdateContractResponseDTO>} -- the updated contract record.
\newline -- Route: \code{PATCH /contracts/:id}; authenticated endpoint. \\ \hline

06 & \code{signContract(}\newline\code{userId: number, contractId: number)} &
A method to record a formal digital signature from the client or freelancer on the contract.
\newline -- \code{userId}: \code{number} -- user identifier signing the contract.
\newline -- \code{contractId}: \code{number} -- target contract identifier.
\newline -- \textit{Returns:} \code{Promise<SignContractResponseDTO>} -- updated signing flags and active status.
\newline -- Route: \code{PATCH /contracts/:id/sign}; authenticated endpoint. \\ \hline

07 & \code{completeContract(}\newline\code{userId: number, contractId: number)} &
A method for the client to mark an active contract as successfully completed once all deliverables are accepted.
\newline -- \code{userId}: \code{number} -- client identifier.
\newline -- \code{contractId}: \code{number} -- contract identifier.
\newline -- \textit{Returns:} \code{Promise<CompleteContractResponseDTO>} -- contract marked as completed.
\newline -- Route: \code{PATCH /contracts/:id/complete}; authenticated endpoint. \\ \hline

08 & \code{cancelContract(}\newline\code{userId: number, contractId: number)} &
A method to terminate a contract when mutual agreement or project circumstances require cancellation.
\newline -- \code{userId}: \code{number} -- caller user identifier.
\newline -- \code{contractId}: \code{number} -- contract identifier.
\newline -- \textit{Returns:} \code{Promise<CancelContractResponseDTO>} -- cancelled contract result.
\newline -- Route: \code{PATCH /contracts/:id/cancel}; authenticated endpoint. \\ \hline
\end{ClassMethodTable}

\paragraph{ClassAttribute}
\begin{ClassAttributeTable}
01 & \code{contractService} & \code{ContractService} -- private readonly service dependency. \\ \hline
\end{ClassAttributeTable}

\paragraph{ClassMethod}
\begin{ClassMethodTable}

\newline -- \code{body}: \code{UpdateContractBodyType} -- fields to update.

\newline -- \code{clientId}: \code{number} -- client identifier.
\newline -- \code{contractId}: \code{number} -- contract identifier.
\newline -- \textit{Returns:} \code{Promise<ContractType>} -- completed contract. \\ \hline

06 & \code{cancelContract(}\newline\code{userId: number, contractId: number)} &
A method to verify participant authorization, check cancellation eligibility, and set status to CANCELLED.
\newline -- \code{userId}: \code{number} -- caller user identifier.
\newline -- \code{contractId}: \code{number} -- contract identifier.
\newline -- \textit{Returns:} \code{Promise<ContractType>} -- cancelled contract. \\ \hline

07 & \code{getContractList(}\newline\code{userId: number, roleName: string,}\newline\code{query: GetContractListQueryType)} &

\newline -- \code{userId}: \code{number} -- user identifier.
\newline -- \code{roleName}: \code{string} -- role name.
\newline -- \code{query}: \code{GetContractListQueryType} -- pagination and filtering criteria.
\newline -- \textit{Returns:} \code{Promise<{ data: ContractListItemType[]; total: number }>} -- list and total count. \\ \hline

08 & \code{getContractDetail(}\newline\code{userId: number, roleName: string,}\newline\code{contractId: number)} &
A method to verify access permissions and load detailed contract specifications with participants.
\newline -- \code{userId}: \code{number} -- user identifier.
\newline -- \code{roleName}: \code{string} -- role name.
\newline -- \code{contractId}: \code{number} -- contract identifier.
\newline -- \textit{Returns:} \code{Promise<ContractDetailType>} -- full contract dossier. \\ \hline
