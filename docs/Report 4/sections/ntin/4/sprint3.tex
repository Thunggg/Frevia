\subsection{DisputeController Class}

\paragraph{ClassMethod}

\newline -- \code{disputeId}: \code{number} -- dispute identifier.
\newline -- \code{body}: \code{SubmitDisputeResponseBodyDTO} -- defense description and counter-evidence file IDs.
\newline -- \textit{Returns:} \code{Promise<DisputeDetailResponseDTO>} -- updated dispute with submitted response.
\newline -- Route: \code{POST /disputes/:id/response}; authenticated endpoint. \\ \hline

\newline -- \textit{Returns:} \code{Promise<DisputeDetailResponseDTO>} -- dispute dossier entity.
\newline -- Route: \code{GET /disputes/milestone/:milestoneId}; authenticated endpoint. \\ \hline

\newline -- \code{userId}: \code{number} -- caller user identifier.
\newline -- \code{roleName}: \code{string} -- caller role name.
\newline -- \code{disputeId}: \code{number} -- dispute case identifier.
\newline -- \textit{Returns:} \code{Promise<DisputeDetailResponseDTO>} -- comprehensive dispute details.
\newline -- Route: \code{GET /disputes/:id}; authenticated endpoint. \\ \hline

08 & \code{submitDecisionReview(}\newline\code{userId: number, disputeId: number,}\newline\code{body: ReviewDisputeDecisionBodyDTO)} &

\newline -- \code{disputeId}: \code{number} -- dispute identifier.

\newline -- \textit{Returns:} \code{Promise<DisputeDetailResponseDTO>} -- updated dispute or finalized settlement.
\newline -- Route: \code{POST /disputes/:id/review}; authenticated endpoint. \\ \hline
