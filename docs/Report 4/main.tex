%----- diagram placeholder macro ---------------------
% \kfig{width}{path WITHOUT extension}{caption}{label}
% Prefers the vector <path>.pdf exported from the draw.io SVG
% (stays sharp at any zoom level), falls back to the raster
% <path>.png, and finally to a placeholder box so the document
% always compiles.
\newcommand{\kfig}[4]{%
\begin{figure}[H]
    \centering
    \IfFileExists{#2.pdf}{%
        \includegraphics[width=#1,height=0.9\textheight,keepaspectratio]{"#2.pdf"}%
    }{%
        \IfFileExists{#2.png}{%
            \includegraphics[width=#1,height=0.9\textheight,keepaspectratio]{"#2.png"}%
        }{%
            \setlength{\fboxsep}{10pt}%
            \fbox{\parbox[c][5.5cm][c]{0.82\textwidth}{\centering
                \textit{Diagram pending export}\\[6pt]
                \texttt{#2.pdf}}}%
        }%
    }
    \caption{#3}
    \label{#4}
\end{figure}}

% Inline code that tolerates underscores and other TeX special characters.
% Breaks are allowed before camel-case capitals and after punctuation so long
% class names, paths, and error factories do not overflow narrow table cells.
\ExplSyntaxOn
\cs_new_protected:Npn \__code_char:n #1
  {
    \regex_match:nnTF { [A-Z] } {#1}
      { \allowbreak #1 }
      {
        #1
        \regex_match:nnT { [^A-Za-z0-9] } {#1} { \allowbreak }
      }
  }
\NewDocumentCommand{\code}{m}
  {
    \texttt
      {
        \tl_set:Nx \l_tmpa_tl { \tl_to_str:n {#1} }
        \tl_replace_all:Nnn \l_tmpa_tl { ~ } { \c_space_token }
        \tl_map_function:NN \l_tmpa_tl \__code_char:n
      }
  }
\ExplSyntaxOff

% Generic three-column supporting table (DTOs, errors, entities).
% The third column absorbs the remaining text width so the table never
% protrudes into the right margin. The third argument is kept for call-site
% readability; its requested width is superseded by the computed width.
\newenvironment{SupportingTable}[3]{%
    \begingroup
    \small
    \setlength{\tabcolsep}{4pt}%
    \renewcommand{\arraystretch}{1.15}%
    \begin{longtable}{|L{#1}|L{#2}|L{\dimexpr\textwidth-6\tabcolsep-1.6pt-#1-#2\relax}|}
    \hline
    \rowcolor{tableHeaderOrange}\textbf{Class / Element} & \textbf{Attributes or Signature} & \textbf{Responsibility} \\ \hline
    \endfirsthead
    \hline
    \rowcolor{tableHeaderOrange}\textbf{Class / Element} & \textbf{Attributes or Signature} & \textbf{Responsibility} \\ \hline
    \endhead
    \hline
    \endfoot
}{%
    \end{longtable}%
    \endgroup
}
