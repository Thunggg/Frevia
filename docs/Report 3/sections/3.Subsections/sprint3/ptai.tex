\subsubsection{Function: View My Proposal List}

\fnhead{Function Trigger}
A freelancer clicks ``Proposals'' in the workspace navigation menu.\newline
\textbf{Navigation path:} Freelancer Dashboard $\rightarrow$ Proposals (\texttt{/freelancer/proposals}).

\fnhead{Function Description}

\sublabel{Actors / Roles}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item \textbf{Primary Actor:} Freelancer.
    \item \textbf{Supporting Systems:} Frevia Web application, Frevia REST API, and PostgreSQL database.
\end{itemize}

\sublabel{Purpose}
\noindent Allows freelancers to review and monitor all proposals they have submitted across different lifecycle stages (\texttt{SUBMITTED}, \texttt{INTERVIEWING}, \texttt{HIRED}, \texttt{REJECTED}, \texttt{WITHDRAWN}, \texttt{EXPIRED}).

\sublabel{Interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Proposals dashboard at \texttt{/freelancer/proposals}.
    \item Tab filters: All, Active, Offers/Hired, Archived.
    \item Proposal cards displaying target job title, client name, proposed bid amount, delivery timeline, and status badge.
\end{itemize}

\sublabel{Data Processing Overview}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Client dispatches \texttt{GET /api/freelancer/proposals}.
    \item Backend extracts freelancer user ID from JWT session.
    \item Queries \texttt{Proposal} table joined with \texttt{Job} and client profiles.
    \item Returns sorted results by submission date with pagination.
\end{itemize}

\fnhead{Screen Layout (Mock-up Description)}
\screenfigure[0.95]{image/screenshot/sprint3/ptai/ViewMyProposalList.png}{Submitted proposals list interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Filter tabs separating active submissions from historical decisions.
    \item Proposal cards displaying job title, proposed bid, net earnings after fee, and status tag.
    \item Action menu for inspecting details, editing pending bids, or withdrawing applications.
\end{itemize}

\fnhead{Function Details}

\sublabel{Input Data}
\begin{specdatatable}{Data Name}{Description}
JWT Token & Authenticated freelancer token.\\ \hline
\texttt{status} & Optional proposal status filter.\\ \hline
\texttt{page} / \texttt{limit} & Pagination integers.\\ \hline
\end{specdatatable}

\sublabel{Validation Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Caller must have \texttt{FREELANCER} role.
\end{itemize}

\sublabel{Business Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]\brrulelist
    \item BR-PR-01: Freelancers may only view proposals submitted from their own account.
    \item BR-PR-02: Statuses reflect exact state machine: \texttt{SUBMITTED}, \texttt{INTERVIEWING}, \texttt{HIRED}, \texttt{REJECTED}, \texttt{WITHDRAWN}, \texttt{EXPIRED}.
\end{itemize}

\sublabel{Normal Flow}
\begin{enumerate}[leftmargin=1.8em,topsep=3pt,itemsep=2pt,parsep=0pt]
    \item Freelancer visits \texttt{/freelancer/proposals}.
    \item Frontend requests proposals from backend.
    \item Backend retrieves user's proposals and associated job summaries.
    \item UI renders cards with active status tags.
\end{enumerate}

\sublabel{Abnormal Flow / Exception Handling}
\begin{flowtable}{Case}{Description}
A1 & No proposals submitted yet. UI displays empty state with a ``Find Work'' shortcut button.\\ \hline
\end{flowtable}

\sublabel{Output Data}
\begin{specdatatable}{Data Name}{Description}
\texttt{proposals} & Array of proposal objects (\texttt{id}, \texttt{jobTitle}, \texttt{bidAmount}, \texttt{deliveryDays}, \texttt{status}, \texttt{submittedAt}).\\ \hline
\end{specdatatable}

\sublabel{Post-Conditions}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Freelancer views list and status of submitted bids.
\end{itemize}

\subsubsection{Function: Submit Proposal}

\fnhead{Function Trigger}
A freelancer clicks ``Submit a Proposal'' on a job detail page.\newline
\textbf{Navigation path:} Job Detail (\texttt{/jobs/:slug}) $\rightarrow$ Click ``Submit a Proposal'' (\texttt{/jobs/:slug/apply}).

\fnhead{Function Description}

\sublabel{Actors / Roles}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item \textbf{Primary Actor:} Freelancer.
    \item \textbf{Supporting Systems:} Frevia Web application, Frevia REST API, and PostgreSQL database.
\end{itemize}

\sublabel{Purpose}
\noindent Allows a freelancer to apply for an open job by providing a bid price, delivery timeframe, comprehensive cover letter, and optional supporting work samples.

\sublabel{Interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Proposal submission page at \texttt{/jobs/:slug/apply}.
    \item Bid terms calculator: inputs for gross bid with automatic calculation of the 10\% platform fee and net earnings.
    \item Rich text cover letter area and file uploader (up to 3 files, 10MB each).
\end{itemize}

\sublabel{Data Processing Overview}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Client submits payload to \texttt{POST /api/freelancer/proposals}.
    \item Backend verifies that: (1) user has \texttt{FREELANCER} role, (2) target job is \texttt{OPEN}, (3) user is not the job's client, (4) user has not already submitted an active proposal for this job.
    \item Creates \texttt{Proposal} record in \texttt{SUBMITTED} status and increments job's proposal count.
\end{itemize}

\fnhead{Screen Layout (Mock-up Description)}
\screenfigure[0.95]{image/screenshot/sprint3/ptai/SubmitProposal.png}{Submit proposal form interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Job summary header with client budget reference.
    \item Interactive bid calculator displaying gross amount, platform service fee, and net earnings.
    \item Textarea for cover letter and attachment uploader.
    \item Primary ``Submit Proposal Now'' button.
\end{itemize}

\fnhead{Function Details}

\sublabel{Input Data}
\begin{specdatatable}{Data Name}{Description}
\texttt{jobId} & UUID of the target job posting.\\ \hline
\texttt{bidAmount} & Number, gross proposed compensation in USD.\\ \hline
\texttt{deliveryDays} & Integer $\ge 1$, estimated completion days.\\ \hline
\texttt{coverLetter} & String (50--3000 chars), proposal text.\\ \hline
\texttt{attachments} & Optional array of uploaded file URLs.\\ \hline
\end{specdatatable}

\sublabel{Validation Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item \texttt{bidAmount} must be $> 0$.
    \item \texttt{deliveryDays} must be an integer between 1 and 365.
    \item \texttt{coverLetter} must contain at least 50 characters.
\end{itemize}

\sublabel{Business Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]\brrulelist
    \item BR-PR-05: Proposals may only be submitted to jobs with status \texttt{OPEN}.
    \item BR-PR-06: Clients cannot submit proposals to their own posted jobs.
    \item BR-PR-07: Freelancers may submit only 1 active proposal per job.
    \item BR-PR-08: Automatic fee calculation deducts standard 10\% marketplace fee.
\end{itemize}

\sublabel{Normal Flow}
\begin{enumerate}[leftmargin=1.8em,topsep=3pt,itemsep=2pt,parsep=0pt]
    \item Freelancer completes application terms and cover letter.
    \item Clicks ``Submit Proposal Now''.
    \item Backend validates eligibility and persists proposal as \texttt{SUBMITTED}.
    \item UI redirects to proposal detail view with a success confirmation toast.
\end{enumerate}

\sublabel{Abnormal Flow / Exception Handling}
\begin{flowtable}{Case}{Description}
A1 & Freelancer already submitted a proposal for this job. Backend returns HTTP 400 Bad Request with message ``You have already applied for this job.''\\ \hline
A2 & Target job has closed while application was being drafted. Returns HTTP 400 with ``Job is no longer open for proposals.''\\ \hline
\end{flowtable}

\sublabel{Output Data}
\begin{specdatatable}{Data Name}{Description}
Created Proposal & New entity with \texttt{id}, \texttt{status = SUBMITTED}, and submission timestamp.\\ \hline
\end{specdatatable}

\sublabel{Post-Conditions}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Proposal is submitted and visible in client's job proposals list.
\end{itemize}

\subsubsection{Function: Accept Proposal}

\fnhead{Function Trigger}
A client clicks ``Accept Proposal'' or ``Hire'' on an applicant's bid.\newline
\textbf{Navigation path:} Client Dashboard $\rightarrow$ Job Proposals $\rightarrow$ Proposal Detail $\rightarrow$ Click ``Accept Proposal''.

\fnhead{Function Description}

\sublabel{Actors / Roles}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item \textbf{Primary Actor:} Client.
    \item \textbf{Secondary Actor:} Payment/Escrow Gateway.
    \item \textbf{Supporting Systems:} Frevia Web application, Frevia REST API, and PostgreSQL database.
\end{itemize}

\sublabel{Purpose}
\noindent Allows a client to accept a freelancer's proposal, automatically initialize and activate a binding contract (\texttt{Contract}), increment the filled positions quota, and transition the job status if fully hired.

\sublabel{Interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Hiring confirmation and contract initialization screen.
    \item Terms summary card detailing payment value, escrow deposit requirement, and project milestones.
\end{itemize}

\sublabel{Data Processing Overview}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Client submits \texttt{POST /api/client/proposals/:id/accept}.
    \item Backend verifies in a database transaction: (1) client owns the job, (2) proposal is \texttt{SUBMITTED} or \texttt{INTERVIEWING}, (3) job still has open hiring slots (\texttt{positionsFilled < positionsRequired}).
    \item Updates proposal to \texttt{status = 'HIRED'}, creates a new \texttt{Contract} entity with terms matching the bid, and increments \texttt{positionsFilled}.
    \item If \texttt{positionsFilled === positionsRequired}, transitions job from \texttt{OPEN} to \texttt{IN\_PROGRESS}.
\end{itemize}

\fnhead{Screen Layout (Mock-up Description)}
\screenfigure[0.95]{image/screenshot/sprint3/ptai/AcceptProposal.png}{Accept proposal and hire confirmation}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Overview card detailing freelancer credentials, bid price, and delivery duration.
    \item Contract terms preview and escrow funding notice.
    \item Primary ``Confirm Hire \& Fund Escrow'' action button.
\end{itemize}

\fnhead{Function Details}

\sublabel{Input Data}
\begin{specdatatable}{Data Name}{Description}
\texttt{id} & UUID of the proposal being accepted.\\ \hline
\end{specdatatable}

\sublabel{Validation Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Calling client must be the creator of the job posting.
    \item Job must have at least one open hiring position available.
\end{itemize}

\sublabel{Business Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]\brrulelist
    \item BR-PR-13: Accepting a proposal transitions its status to \texttt{HIRED} (Accepted).
    \item BR-PR-14: Automatically generates a formal binding contract with terms matching the proposal.
    \item BR-PR-15: Increments \texttt{positionsFilled}; when quota is full, transitions job to \texttt{IN\_PROGRESS}.
\end{itemize}

\sublabel{Normal Flow}
\begin{enumerate}[leftmargin=1.8em,topsep=3pt,itemsep=2pt,parsep=0pt]
    \item Client clicks ``Accept Proposal''.
    \item Reviews terms and escrow details.
    \item Confirms hiring decision.
    \item Backend executes atomic transaction: updates proposal, initializes contract, and adjusts job status.
    \item UI redirects to active contract view.
\end{enumerate}

\sublabel{Abnormal Flow / Exception Handling}
\begin{flowtable}{Case}{Description}
A1 & Job hiring quota already filled. Backend aborts transaction and returns HTTP 409 Conflict.\\ \hline
\end{flowtable}

\sublabel{Output Data}
\begin{specdatatable}{Data Name}{Description}
Created Contract & Contract entity linked to client, freelancer, and originating job.\\ \hline
\end{specdatatable}

\sublabel{Post-Conditions}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Proposal status is HIRED and active contract is initialized.
\end{itemize}
