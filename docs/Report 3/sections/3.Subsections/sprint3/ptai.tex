% UC-32.01: View My Proposal List
% -----------------------------------------------------------------
\Needspace{12\baselineskip}
\subsubsection{View My Proposal List}

\fnhead{Function Trigger}
A freelancer clicks ``My Proposals'' in the left sidebar navigation.\newline
\textbf{Navigation path:} Freelancer Workspace $\rightarrow$ My Proposals (\texttt{/freelancer/proposals}).

\fnhead{Function Description}

\sublabel{Actors / Roles}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item \textbf{Primary Actor:} Freelancer.
    \item \textbf{Supporting Systems:} Frevia Web application (Next.js), Frevia REST API (\texttt{ProposalController}), Prisma ORM, and PostgreSQL database.
\end{itemize}

\sublabel{Purpose}
\noindent Displays a paginated list of all proposals created or submitted by the authenticated freelancer across jobs, supporting status filtering (\texttt{DRAFT}, \texttt{SUBMITTED}, \texttt{HIRED}, \texttt{REJECTED}, \texttt{WITHDRAWN}, \texttt{EXPIRED}) and synchronized expiration state.

\sublabel{Interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Freelancer proposals list page at \texttt{/freelancer/proposals}.
    \item Status filter dropdown (``All statuses'', Draft, Submitted, Hired, Rejected, Withdrawn, Expired) and proposal counter.
    \item Proposal summary rows showing status badge, submission date, linked job title, client name, proposed bid amount, delivery days, and ``View $\rightarrow$'' action button.
\end{itemize}

\sublabel{Data Processing Overview}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Frontend requests \texttt{GET /api/proposals/my} with optional \texttt{page}, \texttt{limit}, \texttt{jobId}, and \texttt{status} parameters.
    \item Backend verifies active \texttt{FREELANCER} role (\texttt{ProposalFreelancerOnlyException} if mismatched).
    \item Service first executes \texttt{expireOverdueProposals()} to mark any stale \texttt{SUBMITTED} proposals past 30 days (\texttt{PROPOSAL\_EXPIRY\_DAYS}) as \texttt{EXPIRED}.
    \item Queries strictly non-deleted proposals where \texttt{freelancerId === userId} alongside linked job/client summaries and pagination metadata.
\end{itemize}

\fnhead{Screen Layout (Mock-up Description)}
\screenfigure[0.95]{image/screenshot/sprint3/ptai/ViewMyProposalList.png}{Freelancer My Proposals list interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Header bar displaying ``My Proposals'', total proposals counter, and status dropdown filter.
    \item Clean row list presenting color-coded status pills (\texttt{Draft}, \texttt{Submitted}, \texttt{Hired}, \texttt{Rejected}, \texttt{Expired}), target job title, client name, bid amount, delivery days, and ``View $\rightarrow$'' link.
\end{itemize}

\fnhead{Function Details}

\sublabel{Input Data}
\begin{specdatatable}{Data Name}{Description}
\texttt{page} & Optional positive integer (default 1).\\ \hline
\texttt{limit} & Optional positive integer between 1 and 50 (default 10).\\ \hline
\texttt{jobId} & Optional positive integer filtering proposals for a single job.\\ \hline
\texttt{status} & Optional enum: \texttt{DRAFT}, \texttt{SUBMITTED}, \texttt{HIRED}, \texttt{REJECTED}, \texttt{WITHDRAWN}, or \texttt{EXPIRED}.\\ \hline
\end{specdatatable}

\sublabel{Validation Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Caller must have active \texttt{FREELANCER} role.
    \item \texttt{page} and \texttt{limit} must be positive integers with \texttt{limit <= 50}; \texttt{status} must belong to \texttt{ProposalStatusSchema}.
\end{itemize}

\sublabel{Business Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]\brrulelist
    \item BR-PROP-01: Freelancer Role and Owned Proposal Scope: Only authenticated \texttt{FREELANCER} accounts may list their own submitted and draft proposals, returning strictly non-deleted records owned by the caller with pagination up to 50 items per page.
    \item BR-PROP-02: Lazy Proposal Expiration Synchronization: Prior to listing or reviewing proposals, overdue \texttt{SUBMITTED} proposals past \texttt{PROPOSAL\_EXPIRY\_DAYS} (30 days) automatically transition to \texttt{EXPIRED}.
\end{itemize}

\sublabel{Normal Flow}
\begin{enumerate}[leftmargin=1.8em,topsep=3pt,itemsep=2pt,parsep=0pt]
    \item Freelancer navigates to \texttt{/freelancer/proposals}.
    \item Frontend requests \texttt{GET /api/proposals/my}.
    \item Backend validates authentication and \texttt{FREELANCER} role, expires overdue proposals, and queries owned rows.
    \item Returns paginated \texttt{MyProposalsResponseDto} with linked job/client objects.
    \item UI renders proposal list rows with status pills and ``View $\rightarrow$'' links.
\end{enumerate}

\sublabel{Abnormal Flow / Exception Handling}
\begin{flowtable}{Abnormal Case}{Handling \& System Response}
Missing or invalid authentication token & Global \texttt{AuthGuard} returns HTTP 401 Unauthorized; client redirects to \texttt{/login}.\\ \hline
Non-freelancer active role (e.g. Client) & Service throws \texttt{ProposalFreelancerOnlyException} (HTTP 403 Forbidden).\\ \hline
Invalid \texttt{page}, \texttt{limit > 50}, or unknown \texttt{status} & Zod DTO (\texttt{MyProposalsQueryDto}) rejects with HTTP 400 Bad Request.\\ \hline
Freelancer has no proposals yet & Endpoint returns empty \texttt{data: []} and \texttt{totalItems: 0}; UI displays empty state with ``Find Work'' CTA.\\ \hline
Database query or expiration sync failure & Service catches non-HTTP exception and throws \texttt{FailedToLoadProposalException} (HTTP 500).\\ \hline
\end{flowtable}

\sublabel{Output Data}
\begin{specdatatable}{Data Name}{Description}
Paginated Proposals & Object containing \texttt{data} array (\texttt{ProposalDetailSchema}), \texttt{totalItems}, \texttt{totalPages}, \texttt{page}, and \texttt{limit}.\\ \hline
\end{specdatatable}

\sublabel{Post-Conditions}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Freelancer views accurate paginated proposal records and statuses.
\end{itemize}

\subsubsection{View My Proposal Detail}

\fnhead{Function Trigger}
A freelancer clicks ``View $\rightarrow$'' on a proposal in My Proposals.\newline
\textbf{Navigation path:} My Proposals (\texttt{/freelancer/proposals}) $\rightarrow$ Click ``View $\rightarrow$'' (\texttt{/freelancer/proposals/:id}).

\fnhead{Function Description}

\sublabel{Actors / Roles}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item \textbf{Primary Actor:} Freelancer.
    \item \textbf{Supporting Systems:} Frevia Web application (Next.js), Frevia REST API (\texttt{ProposalController}), Prisma ORM, and PostgreSQL database.
\end{itemize}

\sublabel{Purpose}
\noindent Shows complete details of the freelancer's proposal including status badge, cover letter, proposed bid amount, delivery days, linked job information, client name, and contextual actions (editing/submitting a draft, updating a submitted bid, or withdrawing).

\sublabel{Interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Two-column detail page at \texttt{/freelancer/proposals/:id}.
    \item Left column: Cover letter textarea, bid input (with client budget hint), and delivery days input (editable when \texttt{DRAFT} or \texttt{SUBMITTED}, read-only once finalized).
    \item Right summary card: Bid, Delivery, Submitted date, and ``Withdraw proposal'' button (for \texttt{SUBMITTED}).
\end{itemize}

\sublabel{Data Processing Overview}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Client dispatches \texttt{GET /api/proposals/:id} with positive integer ID (\texttt{ParseIntPipe}).
    \item Backend verifies \texttt{FREELANCER} role and runs \texttt{expireOverdueProposals()}.
    \item Queries proposal with linked job and client info; checks existence (\texttt{ProposalNotFoundException} 404) and ownership (\texttt{proposal.freelancerId === userId}, else \texttt{ProposalForbiddenException} 403).
\end{itemize}

\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Header with back navigation (\texttt{$\leftarrow$ My proposals}), status pill (\texttt{Submitted}), job title, client name, and ``View job'' shortcut.
    \item Left panel showing Cover letter, Your bid (\texttt{2800}), and Delivery days (\texttt{12}) alongside client constraints.
    \item Right ``Proposal summary'' card displaying Bid, Delivery, Submission date, and red-outlined ``Withdraw proposal'' button.
\end{itemize}

\fnhead{Function Details}

\sublabel{Input Data}
\begin{specdatatable}{Data Name}{Description}
\texttt{id} & Positive integer ID of the proposal.\\ \hline
\end{specdatatable}

\sublabel{Validation Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Path parameter \texttt{id} must parse as a valid integer via \texttt{ParseIntPipe}.
    \item Caller must possess \texttt{FREELANCER} role and be the authoring freelancer.
\end{itemize}

\sublabel{Business Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]\brrulelist
    \item BR-PROP-03: Freelancer Proposal Detail Ownership: Viewing a freelancer proposal detail requires the \texttt{FREELANCER} role, synchronizes overdue expirations, and restricts access strictly to non-deleted proposals owned by the caller.
    \item BR-PROP-04: Embedded Job and Client Context Exposure: Proposal detail responses embed the linked job's budget terms, hiring quota, and client summary alongside the freelancer's submitted terms.
\end{itemize}

\sublabel{Normal Flow}
\begin{enumerate}[leftmargin=1.8em,topsep=3pt,itemsep=2pt,parsep=0pt]
    \item Freelancer clicks a proposal card to open \texttt{/freelancer/proposals/:id}.
    \item Backend authenticates \texttt{FREELANCER} role, expires stale proposals, verifies ownership, and retrieves entity.
    \item Returns HTTP 200 with \texttt{ProposalDetailSchema} payload.
    \item UI renders cover letter, bid, delivery timeline, and permissible action buttons.
\end{enumerate}

\sublabel{Abnormal Flow / Exception Handling}
\begin{flowtable}{Abnormal Case}{Handling \& System Response}
Missing or invalid authentication token & Global \texttt{AuthGuard} rejects with HTTP 401 Unauthorized; client redirects to \texttt{/login}.\\ \hline
Non-freelancer active role & Service throws \texttt{ProposalFreelancerOnlyException} (HTTP 403 Forbidden).\\ \hline
Non-numeric \texttt{id} path parameter & \texttt{ParseIntPipe} returns HTTP 400 Bad Request.\\ \hline
Proposal does not exist or soft-deleted & Service throws \texttt{ProposalNotFoundException} (HTTP 404 Not Found).\\ \hline
Caller is not the proposal's freelancer & Service throws \texttt{ProposalForbiddenException} (HTTP 403 Forbidden).\\ \hline
Database query failure & Service throws \texttt{FailedToLoadProposalException} (HTTP 500 Internal Server Error).\\ \hline
\end{flowtable}

\sublabel{Output Data}
\begin{specdatatable}{Data Name}{Description}
Proposal Detail & Full proposal object (\texttt{ProposalDetailSchema}) including terms, timestamps, \texttt{job}, and \texttt{client}.\\ \hline
\end{specdatatable}

\sublabel{Post-Conditions}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Proposal details and allowable lifecycle controls are rendered for the freelancer.
\end{itemize}

\Needspace{12\baselineskip}
\subsubsection{Submit Proposal}

\fnhead{Function Trigger}
A freelancer clicks ``Apply now'' on an open job detail screen and submits (or saves as draft).\newline
\textbf{Navigation path:} Job Detail (\texttt{/freelancer/jobs/:slug}) $\rightarrow$ Click ``Apply now'' $\rightarrow$ Fill Proposal Dialog.

\fnhead{Function Description}

\sublabel{Actors / Roles}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item \textbf{Primary Actor:} Freelancer.
    \item \textbf{Supporting Systems:} Frevia Web application (Next.js), Frevia REST API (\texttt{ProposalController}), Prisma ORM, and PostgreSQL database.
\end{itemize}

\sublabel{Purpose}
\noindent Enables an eligible freelancer to apply to an open job posting with a cover letter, bid amount, and estimated delivery days—either saving an incomplete/in-progress draft (\texttt{DRAFT}) or submitting directly (\texttt{SUBMITTED}) subject to anti-abuse safeguards and hiring quota limits.

\sublabel{Interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item ``Submit a proposal'' modal dialog triggered from \texttt{/freelancer/jobs/:slug}.
    \item Inputs for Cover letter, Your bid (showing client budget reference and warning alerts if outside budget), Delivery days (showing job deadline and remaining days limit), ``Save draft'', and ``Submit proposal'' buttons.
\end{itemize}

\sublabel{Data Processing Overview}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Direct submission posts \texttt{POST /api/proposals/jobs/:jobId}; saving draft posts \texttt{POST /api/proposals/jobs/:jobId/drafts} (or \texttt{PATCH /api/proposals/:id/submit} to promote an existing draft).
    \item \texttt{assertCanPropose()} validates: (1) \texttt{FREELANCER} role, (2) account not banned/deleted, (3) job exists and not deleted, (4) client not banned/deleted, (5) \texttt{job.status === 'OPEN'}, (6) both \texttt{expiryDate} and \texttt{deadline} strictly in future, (7) caller is not \texttt{job.clientId}, (8) \texttt{positionsFilled < positionsRequired}, (9) no existing active proposal (\texttt{DRAFT}/\texttt{SUBMITTED}), and (10) total submissions on this job $< 3$.
    \item \texttt{assertBidSafeguard()} validates \texttt{bidAmount} bounds and \texttt{assertDeliveryDaysSafeguard()} ensures delivery completes on or before \texttt{job.deadline}.
    \item Inserts proposal row in PostgreSQL (with \texttt{expiresAt = now + 30 days} upon submit).
\end{itemize}

\fnhead{Screen Layout (Mock-up Description)}
\screenfigure[0.95]{image/screenshot/sprint3/ptai/SubmitProposal.png}{Submit proposal modal dialog interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Centered dialog displaying header context for the target job title.
    \item Multi-line Cover letter textarea, ``Your bid'' input with client budget hint, ``Delivery days'' input with deadline countdown hint, ``Save draft'', and primary ``Submit proposal'' button.
\end{itemize}

\fnhead{Function Details}

\sublabel{Input Data}
\begin{specdatatable}{Data Name}{Description}
\texttt{jobId} & Positive integer ID of target job.\\ \hline
\texttt{coverLetter} & Trimmed string (1--5000 chars; required on submit, optional on draft).\\ \hline
\texttt{bidAmount} & Positive number between 5 and 1,000,000,000,000 USD (within \texttt{[budgetMin/10, budgetMax*10]}).\\ \hline
\texttt{deliveryDays} & Positive integer between 1 and 365 such that \texttt{now + deliveryDays <= deadline}.\\ \hline
\end{specdatatable}

\sublabel{Validation Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Direct submission requires all three fields (\texttt{coverLetter}, \texttt{bidAmount}, \texttt{deliveryDays}); draft requires at least one field supplied.
    \item Strictly enforces \texttt{MIN\_PROPOSAL\_BID\_AMOUNT = 5}, extreme bid range (\texttt{budgetMin / 10} to \texttt{budgetMax * 10}), and deadline fit.
\end{itemize}

\sublabel{Business Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]\brrulelist
    \item BR-PROP-05: Account Eligibility and Self-Proposal Guard: Submitting or drafting a proposal requires an active, unbanned, non-deleted \texttt{FREELANCER} account; freelancers are forbidden from submitting proposals on their own jobs.
    \item BR-PROP-06: Job Eligibility and Hiring Slots Gate: Proposals may be created or submitted only for non-deleted \texttt{OPEN} jobs owned by active clients, prior to both \texttt{expiryDate} and \texttt{deadline}, and while open hiring positions remain unfilled.
    \item BR-PROP-07: Active Uniqueness and Resubmission Cap: At most 1 non-terminal active proposal (\texttt{DRAFT}, \texttt{SUBMITTED}, or \texttt{HIRED}) per job is allowed, and at most \texttt{MAX\_PROPOSALS\_PER\_JOB\_PER\_FREELANCER} (3) total non-deleted proposals per job per freelancer.
    \item BR-PROP-08: Bid Safeguard and Delivery Timeline Compliance: Bid amounts must fall within \$5 to \$1T USD and within ten times the job's minimum/maximum budget range; cover letters cannot exceed 5,000 characters; delivery days (1--365) must finish before the job deadline.
    \item BR-PROP-09: Draft Persistence and 30-Day Submission Window: Freelancers may save partial proposals as \texttt{DRAFT} or submit complete proposals as \texttt{SUBMITTED} with a 30-day review window (\texttt{PROPOSAL\_EXPIRY\_DAYS} = 30).
\end{itemize}

\sublabel{Normal Flow}
\begin{enumerate}[leftmargin=1.8em,topsep=3pt,itemsep=2pt,parsep=0pt]
    \item Freelancer clicks ``Apply now'' on \texttt{/freelancer/jobs/:slug} and fills terms inside the dialog.
    \item Clicks ``Submit proposal'' (or ``Save draft'').
    \item Backend executes eligibility, quota, active-duplicate, bid safeguard, and deadline checks.
    \item Persists proposal record and returns HTTP 201/200.
    \item UI displays success toast and updates job action button to ``View proposal'' / ``Continue proposal''.
\end{enumerate}

\sublabel{Abnormal Flow / Exception Handling}
\begin{flowtable}{Abnormal Case}{Handling \& System Response}
Missing or invalid authentication token & Global \texttt{AuthGuard} returns HTTP 401 Unauthorized; client redirects to \texttt{/login}.\\ \hline
Non-freelancer active role & Service throws \texttt{ProposalFreelancerOnlyException} (HTTP 403 Forbidden).\\ \hline
Freelancer account banned or deleted & Service throws \texttt{FreelancerBannedException} (HTTP 403 Forbidden).\\ \hline
Applicant tries to apply to own posted job & Service throws \texttt{CannotProposeOwnJobException} (HTTP 403 Forbidden).\\ \hline
Target job does not exist & Service throws \texttt{ProposalJobNotFoundException} (HTTP 404 Not Found).\\ \hline
Job not OPEN, deleted, or client banned/deleted & Service throws \texttt{ProposalJobUnavailableException} (HTTP 422 Unprocessable Entity).\\ \hline
Job expiryDate or project deadline passed & Service throws \texttt{ProposalJobExpiredException} (HTTP 422 Unprocessable Entity).\\ \hline
Job hiring positions already filled & Service throws \texttt{ProposalJobPositionsFilledException} (HTTP 422 Unprocessable Entity).\\ \hline
Active DRAFT or SUBMITTED proposal already exists & Service throws \texttt{ActiveProposalExistsException} (HTTP 409 Conflict).\\ \hline
Max 3 proposals per job exceeded & Service throws \texttt{MaxProposalSubmissionsExceededException} (HTTP 400 Bad Request).\\ \hline
Bid too low ($< 5$), too large, or outside extreme guard & Throws \texttt{BidAmountTooLowException}, \texttt{BidAmountTooLargeException}, or \texttt{ExtremeBidAmountException} (HTTP 422).\\ \hline
Delivery days invalid or finishes past deadline & Throws \texttt{ProposalDeliveryDaysTooLargeException} or \texttt{ProposalDeliveryExceedsDeadlineException} (HTTP 422).\\ \hline
Database insert error & Service catches Prisma error and throws \texttt{FailedToCreateProposalException} (HTTP 500).\\ \hline
\end{flowtable}

\sublabel{Output Data}
\begin{specdatatable}{Data Name}{Description}
Created Proposal & Persisted \texttt{Proposal} entity with integer \texttt{id}, \texttt{status}, and timestamps.\\ \hline
\end{specdatatable}

\sublabel{Post-Conditions}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Proposal is persisted as \texttt{DRAFT} or \texttt{SUBMITTED} and linked to the target job.
\end{itemize}

\Needspace{12\baselineskip}
\subsubsection{Update Proposal}

\fnhead{Function Trigger}
A freelancer edits cover letter, bid amount, or delivery days on an owned \texttt{DRAFT} or \texttt{SUBMITTED} proposal detail page and saves.\newline
\textbf{Navigation path:} My Proposals $\rightarrow$ Select Proposal (\texttt{/freelancer/proposals/:id}) $\rightarrow$ Edit \& Save.

\fnhead{Function Description}

\sublabel{Actors / Roles}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item \textbf{Primary Actor:} Freelancer.
    \item \textbf{Supporting Systems:} Frevia Web application (Next.js), Frevia REST API (\texttt{ProposalController}), Prisma ORM, and PostgreSQL database.
\end{itemize}

\sublabel{Purpose}
\noindent Permits an owning freelancer to modify terms of an editable proposal (\texttt{DRAFT} or \texttt{SUBMITTED}) or promote a completed draft to \texttt{SUBMITTED}, re-verifying all job eligibility and bid/delivery safeguards.

\sublabel{Interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Proposal edit workspace at \texttt{/freelancer/proposals/:id} showing ``Your proposal'' editable inputs for Cover letter, Your bid, Delivery days, and action buttons (``Save draft'' / ``Submit proposal'' for drafts, or update save button for submitted).
    \item Inline caution banner when bid sits below client's minimum budget bound.
\end{itemize}

\sublabel{Data Processing Overview}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item General update calls \texttt{PATCH /api/proposals/:id}; draft updates call \texttt{PATCH /api/proposals/:id/draft}; draft promotion calls \texttt{PATCH /api/proposals/:id/submit}.
    \item Backend validates \texttt{FREELANCER} role, ownership (\texttt{proposal.freelancerId === userId}), and editable status (\texttt{DRAFT} or \texttt{SUBMITTED}).
    \item Re-runs \texttt{assertCanPropose(..., proposal.id)} and validates any supplied \texttt{bidAmount} / \texttt{deliveryDays} against the current job state.
    \item Updates PostgreSQL row and returns updated \texttt{ProposalResponseDto}.
\end{itemize}

\fnhead{Screen Layout (Mock-up Description)}
\screenfigure[0.95]{image/screenshot/sprint3/ptai/UpdateProposal.png}{Edit draft/submitted proposal interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Proposal header showing \texttt{Draft} badge, target job title, client name, and ``View job'' link.
    \item Editable inputs for Cover letter, Your bid, Delivery days, inline warning feedback, ``Save draft'', and ``Submit proposal'' actions, paired with live right-hand ``Proposal summary'' panel.
\end{itemize}

\fnhead{Function Details}

\sublabel{Input Data}
\begin{specdatatable}{Data Name}{Description}
\texttt{id} & Positive integer ID of proposal.\\ \hline
Payload & At least one field among \texttt{coverLetter} (1--5000), \texttt{bidAmount} (\$5--\$1T), or \texttt{deliveryDays} (1--365).\\ \hline
\end{specdatatable}

\sublabel{Validation Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Request body must provide at least one updatable field (\texttt{DRAFT\_CONTENT\_REQUIRED}).
    \item Target proposal status must strictly be \texttt{DRAFT} or \texttt{SUBMITTED} (\texttt{DRAFT} only for \texttt{:id/draft} and \texttt{:id/submit}).
\end{itemize}

\sublabel{Business Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]\brrulelist
    \item BR-PROP-10: Editable Status Window and Ownership: Only the owning \texttt{FREELANCER} may edit proposals in \texttt{DRAFT} or \texttt{SUBMITTED} status; proposals already \texttt{HIRED}, \texttt{REJECTED}, \texttt{WITHDRAWN}, or \texttt{EXPIRED} cannot be modified.
    \item BR-PROP-11: Submission Completeness and Safeguard Re-Verification: Updating a proposal or submitting a draft requires non-empty cover letter, valid bid amount, and valid delivery days, and re-verifies live job eligibility, bid safeguards, and deadline compliance.
\end{itemize}

\sublabel{Normal Flow}
\begin{enumerate}[leftmargin=1.8em,topsep=3pt,itemsep=2pt,parsep=0pt]
    \item Freelancer opens an editable proposal (\texttt{/freelancer/proposals/:id}) and modifies terms.
    \item Clicks save/update or ``Submit proposal''.
    \item Backend validates ownership, status eligibility, and bid/delivery safeguards.
    \item Persists changes in PostgreSQL and returns updated proposal.
    \item UI displays confirmation toast and refreshes summary card.
\end{enumerate}

\sublabel{Abnormal Flow / Exception Handling}
\begin{flowtable}{Abnormal Case}{Handling \& System Response}
Missing or invalid authentication token & Global \texttt{AuthGuard} returns HTTP 401 Unauthorized; client redirects to \texttt{/login}.\\ \hline
Non-freelancer active role or non-owner & Service throws \texttt{ProposalFreelancerOnlyException} or \texttt{ProposalForbiddenException} (HTTP 403 Forbidden).\\ \hline
Proposal not found or soft-deleted & Service throws \texttt{ProposalNotFoundException} (HTTP 404 Not Found).\\ \hline
Empty update payload & Zod schema rejects with HTTP 400 Bad Request (\texttt{DRAFT\_CONTENT\_REQUIRED}).\\ \hline
Proposal already HIRED, REJECTED, WITHDRAWN, or EXPIRED & Service throws \texttt{ProposalNotSubmittedException} or \texttt{ProposalNotDraftException} (HTTP 422 Unprocessable Entity).\\ \hline

\sublabel{Data Processing Overview}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Frontend dispatches \texttt{GET /api/proposals/:id/client}.
    \item Backend enforces \texttt{CLIENT} role and executes \texttt{expireOverdueProposals()}.
    \item Verifies proposal existence (\texttt{ProposalNotFoundException} 404), non-deleted linked job (\texttt{ProposalJobUnavailableException} 422), and ownership (\texttt{access.job.clientId === userId}, else \texttt{ProposalForbiddenException} 403).
    \item Returns full \texttt{ClientProposalDetailDto} including freelancer profile and job summary.
\end{itemize}

\fnhead{Function Details}

\sublabel{Input Data}
\begin{specdatatable}{Data Name}{Description}
\texttt{id} & Positive integer ID of the candidate proposal.\\ \hline
\end{specdatatable}

\sublabel{Validation Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Path parameter \texttt{id} must parse as an integer; caller must be the \texttt{CLIENT} owning the linked job.
\end{itemize}

\sublabel{Business Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]\brrulelist
    \item BR-PROP-20: Client Applicant Detail Authorization and Privacy: Inspecting an individual applicant's proposal requires \texttt{CLIENT} ownership of the non-deleted linked job, excludes drafts, synchronizes overdue expirations, and restricts candidate cover letters, terms, and profile details exclusively to the owning client.
\end{itemize}

\sublabel{Normal Flow}
\begin{enumerate}[leftmargin=1.8em,topsep=3pt,itemsep=2pt,parsep=0pt]
    \item Client clicks ``View Proposal'' on an applicant card.
    \item Frontend dispatches \texttt{GET /api/proposals/:id/client}.
    \item Backend validates ownership and syncs overdue expiration state.
    \item Returns \texttt{ClientProposalDetailDto} containing bid, delivery timeline, cover letter, and freelancer details.
    \item Drawer displays complete details alongside Message, Profile, Reject, and Hire controls.
\end{enumerate}

\sublabel{Abnormal Flow / Exception Handling}
\begin{flowtable}{Abnormal Case}{Handling \& System Response}
Missing or invalid authentication token & Global \texttt{AuthGuard} returns HTTP 401 Unauthorized; client redirects to \texttt{/login}.\\ \hline
Non-client active role & Service throws \texttt{ProposalClientOnlyException} (HTTP 403 Forbidden).\\ \hline
Non-numeric \texttt{id} parameter & \texttt{ParseIntPipe} returns HTTP 400 Bad Request.\\ \hline
Proposal not found or soft-deleted & Service throws \texttt{ProposalNotFoundException} (HTTP 404 Not Found).\\ \hline
Linked job is deleted & Service throws \texttt{ProposalJobUnavailableException} (HTTP 422 Unprocessable Entity).\\ \hline
Calling client does not own the linked job & Service throws \texttt{ProposalForbiddenException} (HTTP 403 Forbidden).\\ \hline
Database query failure & Service catches error and throws \texttt{FailedToLoadProposalException} (HTTP 500).\\ \hline
\end{flowtable}

\sublabel{Output Data}
\begin{specdatatable}{Data Name}{Description}
Client Proposal Detail & \texttt{ClientProposalDetailDto} object embedding proposal fields, \texttt{freelancer} profile, and linked \texttt{job}.\\ \hline
\end{specdatatable}

\sublabel{Post-Conditions}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Client reviews full candidate application and decides to message, hire, or reject.
\end{itemize}
