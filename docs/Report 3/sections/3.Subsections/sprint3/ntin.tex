\Needspace{12\baselineskip}
\subsubsection{ Open Dispute}

\fnhead{Function Trigger}
A Client or Freelancer clicks the ``Open Dispute'' action button on a funded milestone in the contract workspace.\newline
\textbf{Navigation path:} Contract Workspace $\rightarrow$ Milestones Tab $\rightarrow$ Milestone Card $\rightarrow$ Click ``Open Dispute''.

\fnhead{Function Description}

\sublabel{Actors / Roles}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item \textbf{Primary Actor:} Client, Freelancer.
      \item \textbf{Secondary Actor:} Administrator (\texttt{ADMIN} role).
      \item \textbf{Supporting Systems:} Frevia Web application (Next.js), Frevia REST API (NestJS), Prisma ORM layer, and PostgreSQL relational database.
\end{itemize}

\sublabel{Purpose}
\noindent Initiates a formal dispute arbitration case when project counterparties cannot reach mutual consensus regarding milestone deliverables, non-responsive communication, or scope divergence. Filing a dispute immediately freezes the milestone's escrow balance from unilateral release, cancellation, or refund, ensuring financial security while evidence is gathered.

\sublabel{Interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item ``Open Dispute'' action button rendered on active, funded milestone cards.
      \item File Dispute modal dialog with arbitration policy explanation.
      \item Dispute reason dropdown selector (e.g., Deliverables Unsatisfactory, Scope Discrepancy, Counterpart Unresponsive, Payment Disagreement).
      \item Detailed explanation textarea supporting rich text and Markdown (10--5000 characters).
      \item Resolution proposal fields: suggested split amounts for Client and Freelancer, proposed extension deadline.
      \item Evidence files uploader supporting up to 5 attachments (images, PDFs, documents, communication screenshots).
      \item Action buttons: ``Submit Dispute'' (red destructive) and ``Cancel''.
\end{itemize}

\sublabel{Data Processing Overview}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item The user submits an HTTP request: \texttt{POST /api/disputes} with \texttt{milestoneId}, \texttt{reason}, \texttt{description}, optional \texttt{resolutionType}, split proposals, and \texttt{evidenceFiles}.
      \item \texttt{DisputeService.createDispute()} validates caller authorization: verifies that \texttt{userId === contract.clientId || userId === contract.freelancerId}.
      \item Validates milestone eligibility: checks that milestone payment status is \texttt{FUNDED} and status is not \texttt{COMPLETED}.
      \item Checks that no existing active dispute is currently open on this milestone.
      \item Inside a database transaction:
            \begin{itemize}[leftmargin=1.6em,topsep=2pt,itemsep=2pt,parsep=0pt]
                  \item Creates a new \texttt{Dispute} entity with status \texttt{OPEN} and records the initiator ID and respondent ID.
                  \item Persists associated evidence records in \texttt{DisputeEvidence}.
                  \item Updates milestone status to \texttt{DISPUTED}, freezing funds in escrow.
            \end{itemize}
      \item Sets a 48-hour response countdown deadline for the respondent party.
      \item Dispatches in-app and email notifications alerting the counterpart of the active dispute.
\end{itemize}

\fnhead{Screen Layout (Mock-up Description)}
\screenfigure[0.95]{image/screenshot/sprint3/ntin/OpenDispute.png}{Open Dispute modal dialog}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item The modal header displays an amber arbitration gavel icon and prominent heading ``Open Formal Dispute''.
      \item A notice banner explains that filing a dispute freezes the milestone escrow funds and gives the respondent 48 hours to reply.
      \item Form fields: Reason category dropdown, comprehensive explanation textarea, resolution type, and file attachment dropzone (max 5 files).
      \item Bottom bar features policy consent checkbox, ``Cancel'' button, and red primary button ``Submit Formal Dispute''.
\end{itemize}

\fnhead{Function Details}

\sublabel{Input Data}
\begin{specdatatable}{Data Name}{Description}
      \texttt{milestoneId} & Numeric identifier of the funded milestone being disputed (integer, positive).\\ \hline
      \texttt{reason} & Category summary of the dispute claim (string, 3--255 characters).\\ \hline
      \texttt{description} & Detailed claim narrative explaining the issue and requested outcome (string, 10--5000 characters).\\ \hline
      \texttt{resolutionType} & Optional proposed resolution type (e.g., \texttt{REFUND}, \texttt{SPLIT}, \texttt{REVISE}).\\ \hline
      \texttt{proposedFreelancerAmount} & Optional proposed amount to be paid to the freelancer (number, $\ge 0$).\\ \hline
      \texttt{proposedClientAmount} & Optional proposed refund amount to the client (number, $\ge 0$).\\ \hline
      \texttt{proposedDueDate} & Optional proposed revised milestone deadline (ISO 8601 datetime string).\\ \hline
      \texttt{evidenceFiles} & Array of file IDs and optional descriptions (max 5 items).\\ \hline
\end{specdatatable}

\sublabel{Validation Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Caller must be an authenticated participant (Client or Freelancer) of the contract.
      \item Target milestone must be in \texttt{FUNDED} payment status and not \texttt{COMPLETED}.
      \item No other active dispute may exist for the specified milestone.
      \item Maximum 5 evidence attachments allowed during initial claim.
\end{itemize}

\sublabel{Business Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]\brrulelist
      \item BR-DISPUTE-01: Only direct contract parties (Client or Freelancer) can open a dispute.
      \item BR-DISPUTE-02: Filing a dispute immediately freezes milestone escrow balance from release, cancellation, or refund until resolution is achieved.
      \item BR-DISPUTE-03: A maximum of 5 evidence files can be uploaded during the initial dispute claim.
\end{itemize}

\sublabel{Normal Flow}
\begin{enumerate}[leftmargin=1.8em,topsep=3pt,itemsep=2pt,parsep=0pt]
      \item The user expands the disputed milestone in the contract view and clicks ``Open Dispute''.
      \item The system renders the Open Dispute modal outlining arbitration procedures and rules.
      \item The user selects a dispute reason category and enters a comprehensive narrative description.
      \item The user attaches up to 5 supporting evidence files (screenshots, contracts, or deliverable diffs).
      \item The user specifies an optional proposed resolution and clicks ``Submit Dispute''.
      \item The system validates inputs, creates the dispute record, freezes milestone escrow funds, and schedules the response timer.
      \item The modal closes, a success toast is displayed, and the user is redirected to the Dispute Detail page.
\end{enumerate}

\sublabel{Abnormal Flow / Exception Handling}
\begin{flowtable}{Case}{Description}
      A1 & Milestone not funded or already completed. API returns HTTP 400: ``Disputes can only be opened on funded, incomplete milestones.''\\ \hline
      A2 & Dispute already active for this milestone. API returns HTTP 409 Conflict: ``A dispute is already active for this milestone.''\\ \hline
      A3 & User is not a party to the contract. API returns HTTP 403 Forbidden: ``You do not have permission to dispute this milestone.''\\ \hline
      A4 & Evidence file limit exceeded ($> 5$). API returns HTTP 422: ``Maximum 5 evidence files allowed.''\\ \hline
\end{flowtable}

\sublabel{Output Data}
\begin{specdatatable}{Data Name}{Description}
      \texttt{dispute} & Full created dispute object including id, milestoneId, status (\texttt{OPEN}), responseDeadline, and evidence attachments.\\ \hline
\end{specdatatable}

\sublabel{Post-Conditions}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Dispute record created with status \texttt{OPEN}.
      \item Milestone status transitioned to \texttt{DISPUTED}; escrow balance locked.
      \item Respondent notified with 48-hour deadline countdown.
\end{itemize}

\subsubsection{ View Dispute List}

\fnhead{Function Trigger}
A user navigates to the ``Disputes'' section from the main navigation menu.\newline
\textbf{Navigation path:} Main Navigation $\rightarrow$ Click ``Disputes'' (\texttt{/client/disputes} or \texttt{/freelancer/disputes}).

\fnhead{Function Description}

\sublabel{Actors / Roles}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item \textbf{Primary Actor:} Client, Freelancer, Administrator, Expert.
      \item \textbf{Supporting Systems:} Frevia Web application, Frevia REST API, Prisma ORM, and PostgreSQL database.
\end{itemize}

\sublabel{Purpose}
\noindent Allows contract participants, administrators, or expert arbitrators to retrieve a comprehensive, paginated list of dispute cases associated with their account, complete with status filter tabs and urgency countdown indicators.

\sublabel{Interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Disputes dashboard page with status filter tabs: ``All'', ``Waiting Response'', ``Under Review'', ``Escalated'', ``Resolved'', ``Closed''.
      \item Search input allowing query by contract title or milestone name.
      \item Dispute summary cards rendering: Contract Name, Disputed Milestone, Counterparty Avatar/Name, Disputed Amount (\$USD), Response Deadline countdown, and Status Badge.
      \item Pagination controls footer with page size selector.
\end{itemize}

\sublabel{Data Processing Overview}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Client submits an HTTP request: \texttt{GET /api/disputes} with query parameters \texttt{status}, \texttt{page}, \texttt{limit}, \texttt{sortBy}, \texttt{order}.
      \item \texttt{DisputeService.listMyDisputes()} filters disputes where \texttt{openedById === userId || respondentId === userId} (or all for Admin).
      \item Queries database records ordered chronologically and computes pagination metadata.
      \item Returns HTTP 200 with the array of dispute records and pagination envelope.
\end{itemize}

\fnhead{Screen Layout (Mock-up Description)}
\screenfigure[0.95]{image/screenshot/sprint3/ntin/ViewDisputeList.png}{Disputes management list screen}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Header renders page title ``Dispute Resolution Center'' and informational count badges.
      \item Filter tabs across the top allow switching between active disputes, urgent responses, and resolved archives.
      \item Case list displays card items with yellow/red urgency pills indicating hours remaining before automatic default.
      \item Action button ``View Case Details'' opens the dispute workspace.
\end{itemize}

\fnhead{Function Details}

\sublabel{Input Data}
\begin{specdatatable}{Data Name}{Description}
      \texttt{status} & Optional dispute status filter (\texttt{OPEN}, \texttt{WAITING\_RESPONSE}, \texttt{UNDER\_REVIEW}, \texttt{ESCALATED\_TO\_EXPERT}, \texttt{RESOLVED}, \texttt{CLOSED}).\\ \hline
      \texttt{page} & Page number (integer, default 1).\\ \hline
      \texttt{limit} & Items per page (integer, default 10, max 50).\\ \hline
      \texttt{sortBy} & Sort column (\texttt{createdAt}, \texttt{updatedAt}, \texttt{responseDeadline}, default \texttt{createdAt}).\\ \hline
      \texttt{order} & Sort order (\texttt{asc}, \texttt{desc}, default \texttt{desc}).\\ \hline
\end{specdatatable}

\sublabel{Validation Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item User must be authenticated.
      \item Non-administrative callers receive only dispute records where they are a direct party.
\end{itemize}

\sublabel{Business Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]\brrulelist
      \item BR-DISPUTE-04: Strict tenant-level isolation prevents users from browsing disputes outside their own contracts.
      \item BR-DISPUTE-05: Disputes with active response deadlines must display dynamic real-time countdown badges.
\end{itemize}

\sublabel{Normal Flow}
\begin{enumerate}[leftmargin=1.8em,topsep=3pt,itemsep=2pt,parsep=0pt]
      \item User navigates to the Disputes section.
      \item The system queries the API for disputes matching the user and active filters.
      \item The system renders dispute summary cards sorted by urgency and creation date.
      \item The user interacts with filter tabs or clicks a card to open case details.
\end{enumerate}

\sublabel{Abnormal Flow / Exception Handling}
\begin{flowtable}{Case}{Description}
      A1 & Unauthenticated session. API returns HTTP 401; redirects to login.\\ \hline
      A2 & No disputes found. The system renders an empty state illustration: ``No dispute cases found.''\\ \hline
\end{flowtable}

\sublabel{Output Data}
\begin{specdatatable}{Data Name}{Description}
      \texttt{data[]} & Array of dispute records including id, milestone details, parties, status, and responseDeadline.\\ \hline
      \texttt{pagination} & Object containing \texttt{page}, \texttt{limit}, \texttt{totalItems}, and \texttt{totalPages}.\\ \hline
\end{specdatatable}

\sublabel{Post-Conditions}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Paginated dispute list rendered.
\end{itemize}

\Needspace{12\baselineskip}
\subsubsection{ View Dispute Detail}

\fnhead{Function Trigger}
A user clicks on a dispute card from the list or follows a dispute notification link.\newline
\textbf{Navigation path:} Disputes List $\rightarrow$ Click Dispute Card (\texttt{.../disputes/:id}).

\fnhead{Function Description}

\sublabel{Actors / Roles}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item \textbf{Primary Actor:} Client, Freelancer, Administrator, Expert.
      \item \textbf{Supporting Systems:} Frevia Web application, Frevia REST API, Prisma ORM, and PostgreSQL database.
\end{itemize}

\sublabel{Purpose}
\noindent Displays the comprehensive dispute case file, including the initiator's claims, respondent's counter-rebuttal, symmetric evidence files, milestone details, financial escrow balance, resolution findings, and chronological audit history.

\sublabel{Interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Header summary: Dispute ID, Contract Name, Disputed Milestone, Status badge, and response countdown timer.
      \item Opener Claim card: Reason, detailed narrative, submission timestamp, and evidence file attachments.
      \item Respondent Rebuttal card: Counter-explanation, action taken, submission timestamp, and rebuttal evidence.
      \item Arbitration Section: Expert findings, proposed split amounts, or final binding decision.
      \item Action bar: Contextual action buttons (``Submit Response'', ``Withdraw Dispute'', ``Escalate to Expert'').
\end{itemize}

\sublabel{Data Processing Overview}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Client requests \texttt{GET /api/disputes/:id}.
      \item \texttt{DisputeService.getDisputeDetail()} verifies caller authorization: caller must be \texttt{openedById}, \texttt{respondentId}, an \texttt{ADMIN}, or an assigned \texttt{EXPERT}.
      \item Retrieves dispute entity with milestone, contract, evidence files, and decision review records.
      \item Returns HTTP 200 with the complete dispute case file.
\end{itemize}

\fnhead{Screen Layout (Mock-up Description)}
\screenfigure[0.95]{image/screenshot/sprint3/ntin/ViewDisputeDetail.png}{Dispute case details and arbitration console}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Left column presents dispute metadata, financial escrow card (\$amount held), and response deadline clock.
      \item Center workspace renders side-by-side evidence timeline: Initial claim on the left, Counter-response on the right.
      \item Attached evidence cards provide click-to-preview modals for screenshots and PDF deliverables.
      \item Top action bar contains conditional control buttons based on caller role and case status.
\end{itemize}

\fnhead{Function Details}

\sublabel{Input Data}
\begin{specdatatable}{Data Name}{Description}
      \texttt{id} & Numeric dispute identifier (integer path parameter).\\ \hline
\end{specdatatable}

\sublabel{Validation Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Dispute ID must exist in the database.
      \item Caller must be an authorized case participant, Administrator, or assigned Expert.
\end{itemize}

\sublabel{Business Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]\brrulelist
      \item BR-DISPUTE-06: Case evidence and claims are strictly accessible only to direct case participants, designated administrators, and assigned experts.
\end{itemize}

\sublabel{Normal Flow}
\begin{enumerate}[leftmargin=1.8em,topsep=3pt,itemsep=2pt,parsep=0pt]
      \item User clicks on a dispute entry.
      \item System validates caller authorization.
      \item System fetches and renders the complete case file including claims, evidence attachments, and timeline.
      \item User inspects evidence or triggers an available action (Respond, Withdraw, Escalate).
\end{enumerate}

\sublabel{Abnormal Flow / Exception Handling}
\begin{flowtable}{Case}{Description}
      A1 & Caller is not a case participant or admin/expert. API returns HTTP 403 Forbidden: ``Access Denied: You do not have permission to view this dispute.''\\ \hline
      A2 & Dispute ID does not exist. API returns HTTP 404 Not Found: ``Dispute not found.''\\ \hline
\end{flowtable}

\sublabel{Output Data}
\begin{specdatatable}{Data Name}{Description}
      \texttt{dispute} & Comprehensive dispute detail entity including claims, evidence files, resolution status, and financial escrow allocations.\\ \hline
\end{specdatatable}

\sublabel{Post-Conditions}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Dispute case file rendered.
\end{itemize}

\Needspace{12\baselineskip}
\subsubsection{ View Dispute by Milestone}

\fnhead{Function Trigger}
A contract workspace view renders a milestone that has an active or historical dispute.\newline
\textbf{Navigation path:} Contract Workspace $\rightarrow$ Milestones Tab $\rightarrow$ Milestone Card.

\fnhead{Function Description}

\sublabel{Actors / Roles}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item \textbf{Primary Actor:} Client, Freelancer, Administrator.
      \item \textbf{Supporting Systems:} Frevia Web application, Frevia REST API, Prisma ORM, and PostgreSQL database.
\end{itemize}

\sublabel{Purpose}
\noindent Inspects dispute status directly linked to a specific milestone to display contextual warning banners, disable regular deliverable workflows (such as submission, modification, and manual escrow release), and provide seamless navigation to arbitration.

\sublabel{Interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Amber/red alert banner rendered atop the disputed milestone card.
      \item Dispute summary indicator: Current Dispute Status (\texttt{OPEN}, \texttt{WAITING\_RESPONSE}, \texttt{UNDER\_REVIEW}, \texttt{ESCALATED\_TO\_EXPERT}), Initiator Name, and Escrow Locked notice.
      \item Direct action button: ``View Dispute Case \#ID''.
      \item Disabled standard milestone buttons with explanatory tooltip: ``Actions disabled while under formal dispute.''
\end{itemize}

\sublabel{Data Processing Overview}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Client requests \texttt{GET /api/disputes/milestone/:milestoneId}.
      \item \texttt{DisputeService.getDisputeByMilestoneId()} validates caller is a contract participant or Admin.
      \item Queries the active or most recent dispute linked to \texttt{milestoneId}.
      \item Returns HTTP 200 with dispute summary data or HTTP 404 if no dispute is associated.
\end{itemize}

\fnhead{Screen Layout (Mock-up Description)}
\noindent This function operates as an automated query and contextual state-checking service; it does not possess a standalone user interface screen. When a milestone has an active dispute, its status is reflected directly through warning indicators and an alert badge on the milestone workspace card, providing a direct action link to open the Dispute Detail drawer (\textit{UC-33.03}).

\fnhead{Function Details}

\sublabel{Input Data}
\begin{specdatatable}{Data Name}{Description}
      \texttt{milestoneId} & Numeric milestone identifier (integer path parameter).\\ \hline
\end{specdatatable}

\sublabel{Validation Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Milestone ID must exist.
      \item Caller must be a participant of the parent contract or an Administrator.
\end{itemize}

\sublabel{Business Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]\brrulelist
      \item BR-DISPUTE-07: Milestones linked to active disputes must prominently display dispute banners and disable standard milestone submission, modification, and approval buttons.
\end{itemize}

\sublabel{Normal Flow}
\begin{enumerate}[leftmargin=1.8em,topsep=3pt,itemsep=2pt,parsep=0pt]
      \item User opens the contract workspace milestones tab.
      \item The frontend queries dispute status for each milestone.
      \item For milestones with active disputes, the system renders the dispute alert banner and disables conflicting action buttons.
      \item The user clicks ``View Dispute Case'' to navigate directly to the dispute console.
\end{enumerate}

\sublabel{Abnormal Flow / Exception Handling}
\begin{flowtable}{Case}{Description}
      A1 & User not authorized on the contract. API returns HTTP 403 Forbidden.\\ \hline
      A2 & No dispute exists for this milestone. API returns HTTP 404 Not Found; frontend renders standard milestone UI.\\ \hline
\end{flowtable}

\sublabel{Output Data}
\begin{specdatatable}{Data Name}{Description}
      \texttt{dispute} & Summary dispute object with id, status, openedById, and responseDeadline.\\ \hline
\end{specdatatable}

\sublabel{Post-Conditions}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Dispute status displayed on milestone card; restricted actions locked.
\end{itemize}

\Needspace{12\baselineskip}
\subsubsection{ Respond to Dispute}

\fnhead{Function Trigger}
The designated respondent clicks ``Submit Response'' on an open dispute case.\newline
\textbf{Navigation path:} Dispute Detail $\rightarrow$ Click ``Submit Response''.

\fnhead{Function Description}

\sublabel{Actors / Roles}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item \textbf{Primary Actor:} Client, Freelancer (Designated Respondent).
      \item \textbf{Supporting Systems:} Frevia Web application, Frevia REST API, Prisma ORM, and PostgreSQL database.
\end{itemize}

\sublabel{Purpose}
\noindent Allows the respondent party to formally submit their rebuttal statement and counter-evidence within the 48-hour response window, ensuring equal hearing and fair arbitration before escalation or mediator rulings.

\sublabel{Interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Response action panel or modal dialog on the Dispute Detail page.
      \item Action choice selector: Accept Claim or Submit Counter-Rebuttal.
      \item Detailed counter-explanation textarea supporting rich Markdown (max 5000 characters).
      \item Counter-evidence attachment dropzone supporting up to 5 files.
      \item Remaining response countdown timer indicator.
      \item Action buttons: ``Confirm \& Submit Response'' and ``Cancel''.
\end{itemize}

\sublabel{Data Processing Overview}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Respondent submits an HTTP request: \texttt{PATCH /api/disputes/:id/respond} with \texttt{action}, optional \texttt{reason}, and \texttt{evidenceFiles}.
      \item \texttt{DisputeService.respondToDispute()} verifies that \texttt{userId === dispute.respondentId}.
      \item Verifies that dispute status is \texttt{OPEN} or \texttt{WAITING\_RESPONSE}.
      \item Verifies that current time has not exceeded \texttt{responseDeadline}.
      \item Persists rebuttal narrative and counter-evidence records.
      \item Transitions dispute status to \texttt{UNDER\_REVIEW}.
      \item Dispatches notifications alerting the opener and administrator that the response has been submitted.
\end{itemize}

\fnhead{Screen Layout (Mock-up Description)}
\screenfigure[0.95]{image/screenshot/sprint3/ntin/RespondToDispute.png}{Respond to dispute rebuttal form}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Top banner displays the opener's claim summary and remaining response time badge.
      \item Action selector allows choosing between accepting the claim or disputing it with counter-evidence.
      \item Textarea provides prompts for addressing specific allegations (specifications, timeline, deliverable quality).
      \item Dropzone allows uploading up to 5 rebuttal files with instant file size and extension validation.
      \item Primary blue button ``Submit Rebuttal''.
\end{itemize}

\fnhead{Function Details}

\sublabel{Input Data}
\begin{specdatatable}{Data Name}{Description}
      \texttt{id} & Numeric dispute identifier (integer path parameter).\\ \hline
      \texttt{action} & Action enum (\texttt{ACCEPT}, \texttt{REJECT}, \texttt{COUNTER}).\\ \hline
      \texttt{reason} & Detailed counter-explanation narrative (string, max 5000 characters).\\ \hline
      \texttt{evidenceFiles} & Array of counter-evidence file IDs and descriptions (max 5 items).\\ \hline
\end{specdatatable}

\sublabel{Validation Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Caller must be the designated respondent of the dispute.
      \item Dispute status must be \texttt{OPEN} or \texttt{WAITING\_RESPONSE}.
      \item Response must be submitted prior to the 48-hour response deadline.
      \item Maximum 5 counter-evidence files allowed.
\end{itemize}

\sublabel{Business Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]\brrulelist
      \item BR-DISPUTE-08: Only the designated respondent can submit a formal response.
