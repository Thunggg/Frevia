\Needspace{12\baselineskip}
\subsubsection{ Open Dispute}

\fnhead{Function Trigger}
A Client or Freelancer clicks the ``Open a dispute claim'' action button (or Dispute icon) on a funded milestone in the Milestone Detail Overview or Milestones list.\newline
\textbf{Navigation path:} Contract Workspace $\rightarrow$ Milestones / Milestone Detail $\rightarrow$ Click ``Open a dispute claim''.

\fnhead{Function Description}

\sublabel{Actors / Roles}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item \textbf{Primary Actor:} Client, Freelancer.
      \item \textbf{Supporting Systems:} Frevia Web application (Next.js), Frevia REST API (NestJS), Prisma ORM layer, and PostgreSQL relational database.
\end{itemize}

\sublabel{Purpose}
\noindent Initiates a formal dispute resolution request when project counterparties cannot agree on milestone deliverables, deadlines, or scope. Filing a resolution request freezes the milestone's escrow balance from release, cancellation, or refund while parties exchange responses and escalate to a Frevia Expert if needed.

\sublabel{Interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item ``Resolution Center'' right-hand slide-over sheet displaying the milestone title and escrowed amount.
      \item ``Escrow Split Proposal'' section with ``Total'' readout, ``Refund to Client (\$)'', ``Pay to Freelancer (\$)'', and ``Quick presets:'' buttons (``100\% Refund Client'', ``50\% - 50\%'', ``100\% Pay Freelancer'').
      \item Required ``Reason *'' input field (with quick-fill common reason chips) and ``Explain the details (min 10 characters) *'' textarea with live character count.
      \item ``Attach Contract Evidence (count/5)'' section with ``Upload'' button, contract shared-file selector, and locked-evidence integrity notice.
      \item ``2-Tier Dispute Resolution Process'' summary card explaining Step 1 (48-hour counterparty response) and Step 2 (Frevia Expert binding ruling).
      \item Footer buttons: ``Cancel'' and primary blue ``Submit Request''.
\end{itemize}

\sublabel{Data Processing Overview}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item The user submits an HTTP request: \texttt{POST /api/disputes} with \texttt{milestoneId}, \texttt{reason}, \texttt{description}, optional \texttt{resolutionType}, split proposals, and \texttt{evidenceFiles}.
      \item \texttt{DisputeService.createDispute()} validates caller authorization: verifies that \texttt{userId === contract.clientId || userId === contract.freelancerId}.
      \item Validates milestone eligibility: checks that milestone payment status is \texttt{FUNDED} and status is not \texttt{COMPLETED}.
      \item Checks that no existing active dispute is currently open on this milestone.
      \item Inside a database transaction:
            \begin{itemize}[leftmargin=1.6em,topsep=2pt,itemsep=2pt,parsep=0pt]
                  \item Creates a new \texttt{Dispute} entity with status \texttt{WAITING\_RESPONSE} and records the initiator ID and respondent ID.
                  \item Persists associated evidence records in \texttt{DisputeEvidence}.
                  \item Updates milestone status to \texttt{DISPUTED}, freezing funds in escrow.
            \end{itemize}
      \item Sets a 48-hour response countdown deadline for the respondent party.
      \item Dispatches in-app and email notifications alerting the counterpart of the active dispute.
\end{itemize}

\fnhead{Screen Layout (Mock-up Description)}
\screenfigure[0.95]{image/screenshot/sprint3/ntin/OpenDispute.png}{Open Dispute (Resolution Center) slide-over drawer}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item The right-hand ``Resolution Center'' drawer displays a balance scale icon, milestone title, and escrowed budget summary.
      \item ``Escrow Split Proposal'' renders side-by-side ``Refund to Client (\$)'' and ``Pay to Freelancer (\$)'' inputs with ``Quick presets'' pill shortcuts.
      \item Below are ``Reason *'', ``Explain the details (min 10 characters) *'' with counter, ``Attach Contract Evidence'' with ``Upload'' button and immutable-evidence warning, and the ``2-Tier Dispute Resolution Process'' info card.
      \item Footer bar features ``Cancel'' and primary blue ``Submit Request'' button.
\end{itemize}

\fnhead{Function Details}

\sublabel{Input Data}
\begin{specdatatable}{Data Name}{Description}
      \texttt{milestoneId} & Numeric identifier of the funded milestone being disputed (integer, positive).\\ \hline
      \texttt{reason} & Category summary of the dispute claim (string, 3--255 characters).\\ \hline
      \texttt{description} & Detailed claim narrative explaining the issue and requested outcome (string, 10--5000 characters).\\ \hline
      \texttt{resolutionType} & Optional proposed resolution type (\texttt{FULL\_REFUND}, \texttt{FULL\_RELEASE}, \texttt{PARTIAL\_REFUND}, \texttt{EXTEND\_DEADLINE}).\\ \hline
      \texttt{proposed\allowbreak{}Freelancer\allowbreak{}Amount} & Optional proposed amount to be paid to the freelancer (number, $\ge 0$).\\ \hline
      \texttt{proposedClientAmount} & Optional proposed refund amount to the client (number, $\ge 0$).\\ \hline
      \texttt{proposedDueDate} & Optional proposed revised milestone deadline (ISO 8601 datetime string).\\ \hline
      \texttt{evidenceFiles} & Array of file IDs and optional descriptions (max 5 items).\\ \hline
\end{specdatatable}

\sublabel{Validation Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Caller must be an authenticated participant (Client or Freelancer) of the contract.
      \item Target milestone must be in \texttt{FUNDED} payment status and not \texttt{COMPLETED}.
      \item No other active dispute may exist for the specified milestone.
      \item Maximum 5 evidence attachments allowed during initial claim.
\end{itemize}

\sublabel{Business Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]\brrulelist
      \item BR-DISPUTE-01: Only direct contract parties (Client or Freelancer) can open a dispute.
      \item BR-DISPUTE-02: Filing a dispute immediately freezes milestone escrow balance from release, cancellation, or refund until resolution is achieved.
      \item BR-DISPUTE-03: A maximum of 5 evidence files can be uploaded during the initial dispute claim.
\end{itemize}

\sublabel{Normal Flow}
\begin{enumerate}[leftmargin=1.8em,topsep=3pt,itemsep=2pt,parsep=0pt]
      \item The user opens a funded milestone and clicks ``Open a dispute claim'' (or the dispute icon).
      \item The system opens the ``Resolution Center'' slide-over drawer.
      \item The user configures the proposed escrow split, enters ``Reason *'' and detailed explanation ($\ge 10$ characters).
      \item The user attaches up to 5 supporting evidence files (via ``Upload'' or contract shared files).
      \item The user clicks ``Submit Request''.
      \item The system validates inputs, creates the dispute record, freezes milestone escrow funds, and schedules the 48-hour response timer.
      \item The drawer switches/opens the Resolution Detail sheet, and a success toast confirms the submission.
\end{enumerate}

\sublabel{Abnormal Flow / Exception Handling}
\begin{flowtable}{Case}{Description}
      A1 & Milestone not funded or already completed. API returns HTTP 422 (\texttt{MilestoneCannotBeDisputedException}).\\ \hline
      A2 & Dispute already active for this milestone. API returns HTTP 422 (\texttt{MilestoneAlreadyDisputedException}).\\ \hline
      A3 & User is not a party to the contract. API returns HTTP 403 Forbidden (\texttt{DisputeForbiddenException}).\\ \hline
      A4 & Evidence file limit exceeded ($> 5$). API returns HTTP 400 (\texttt{DisputeEvidenceLimitExceededException}).\\ \hline
\end{flowtable}

\sublabel{Output Data}
\begin{specdatatable}{Data Name}{Description}
      \texttt{dispute} & Full created dispute object including id, milestoneId, status (\texttt{WAITING\_RESPONSE}), responseDeadline, and evidence attachments.\\ \hline
\end{specdatatable}

\sublabel{Post-Conditions}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Dispute record created with status \texttt{WAITING\_RESPONSE}.
      \item Milestone status transitioned to \texttt{DISPUTED}; escrow balance locked.
      \item Respondent notified with 48-hour deadline countdown.
\end{itemize}

\Needspace{12\baselineskip}
\subsubsection{ View Disputes List}

\fnhead{Function Trigger}
A user navigates to the ``Disputes'' section from the main navigation menu.\newline
\textbf{Navigation path:} Main Navigation $\rightarrow$ Click ``Disputes'' (\texttt{/client/disputes} or \texttt{/freelancer/disputes}).

\fnhead{Function Description}

\sublabel{Actors / Roles}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item \textbf{Primary Actor:} Client, Freelancer.
      \item \textbf{Supporting Systems:} Frevia Web application, Frevia REST API, Prisma ORM, and PostgreSQL database.
\end{itemize}

\sublabel{Purpose}
\noindent Allows contract participants to retrieve a paginated list of dispute cases where they are a direct party, complete with status filters and urgency countdown indicators.

\sublabel{Interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Disputes dashboard page with status filter tabs: ``All'', ``Waiting Response'', ``Under Review'', ``Finalized'', ``Cancelled''.
      \item Dispute summary cards rendering: Contract Name, Disputed Milestone, Counterparty Avatar/Name, Disputed Amount (\$USD), Response Deadline countdown, and Status Badge.
      \item Pagination controls footer with page size selector.
\end{itemize}

\sublabel{Data Processing Overview}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Client submits an HTTP request: \texttt{GET /\allowbreak{}api/\allowbreak{}disputes} with query parameters \texttt{status}, \texttt{page}, \texttt{limit}, \texttt{sortBy}, \texttt{order}.
      \item \texttt{DisputeService.listMyDisputes()} restricts query strictly to records where \texttt{openedById === userId || respondentId === userId}.
      \item Queries database records ordered by \texttt{sortBy} and \texttt{order}, and computes pagination metadata.
      \item Returns HTTP 200 with the array of dispute records and pagination envelope.
\end{itemize}

\fnhead{Screen Layout (Mock-up Description)}
\screenfigure[0.95]{image/screenshot/sprint3/ntin/ViewDisputeList.png}{Disputes management list screen}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Header renders page title ``Dispute Resolution Center'' and informational count badges.
      \item Filter tabs across the top allow switching between waiting response, under review, finalized, and cancelled disputes.
      \item Case list displays card items with yellow/red urgency pills indicating hours remaining before automatic default.
      \item Action button ``View Case Details'' opens the dispute workspace.
\end{itemize}

\fnhead{Function Details}

\sublabel{Input Data}
\begin{specdatatable}{Data Name}{Description}
      \texttt{status} & Optional \texttt{DisputeStatus} filter (\texttt{OPEN}, \texttt{WAITING\_RESPONSE}, \texttt{UNDER\_REVIEW}, \texttt{DECISION\_MADE}, \texttt{REVIEW\_REQUESTED}, \texttt{FINALIZED}, \texttt{CANCELLED}).\\ \hline
      \texttt{page} & Page number (integer, default 1).\\ \hline
      \texttt{limit} & Items per page (integer, default 10, max 50).\\ \hline
      \texttt{sortBy} & Sort column (\texttt{createdAt}, \texttt{updatedAt}, \texttt{responseDeadline}, default \texttt{createdAt}).\\ \hline
      \texttt{order} & Sort order (\texttt{asc}, \texttt{desc}, default \texttt{desc}).\\ \hline
\end{specdatatable}

\sublabel{Validation Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item User must be authenticated.
      \item Callers receive exclusively dispute records where they are a direct party (\texttt{openedById} or \texttt{respondentId}).
\end{itemize}

\sublabel{Business Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]\brrulelist
      \item BR-DISPUTE-04: Strict tenant-level isolation prevents users from browsing disputes outside their own contracts.
      \item BR-DISPUTE-05: Disputes with active response deadlines must display dynamic real-time countdown badges.
\end{itemize}

\sublabel{Normal Flow}
\begin{enumerate}[leftmargin=1.8em,topsep=3pt,itemsep=2pt,parsep=0pt]
      \item User navigates to the Disputes section.
      \item The system queries the API for disputes matching the user and active filters.
      \item The system renders dispute summary cards sorted by urgency and creation date.
      \item The user interacts with filter tabs or clicks a card to open case details.
\end{enumerate}

\sublabel{Abnormal Flow / Exception Handling}
\begin{flowtable}{Case}{Description}
      A1 & Unauthenticated session. API returns HTTP 401; redirects to login.\\ \hline
      A2 & No disputes found. The system renders an empty state illustration: ``No dispute cases found.''\\ \hline
\end{flowtable}

\sublabel{Output Data}
\begin{specdatatable}{Data Name}{Description}
      \texttt{data[]} & Array of dispute records including id, milestone details, parties, status, and responseDeadline.\\ \hline
      \texttt{pagination} & Object containing \texttt{page}, \texttt{limit}, \texttt{totalItems}, and \texttt{totalPages}.\\ \hline
\end{specdatatable}

\sublabel{Post-Conditions}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Paginated dispute list rendered.
\end{itemize}

\subsubsection{ View Dispute Details}

\fnhead{Function Trigger}
A user clicks on a dispute card from the list or follows a dispute notification link.\newline
\textbf{Navigation path:} Disputes List $\rightarrow$ Click Dispute Card (\texttt{.../disputes/:id}).

\fnhead{Function Description}

\sublabel{Actors / Roles}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item \textbf{Primary Actor:} Client, Freelancer.
      \item \textbf{Supporting Systems:} Frevia Web application, Frevia REST API, Prisma ORM, and PostgreSQL database.
\end{itemize}

\sublabel{Purpose}
\noindent Displays the comprehensive dispute case file, including the initiator's claims, respondent's counter-rebuttal, symmetric evidence files, milestone details, financial escrow balance, resolution findings, and chronological audit history.

\sublabel{Interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Header summary: Dispute ID, Contract Name, Disputed Milestone, Status badge, and response countdown timer.
      \item Opener Claim card: Reason, detailed narrative, submission timestamp, and evidence file attachments.
      \item Respondent Rebuttal card: Counter-explanation, action taken, submission timestamp, and rebuttal evidence.
      \item Arbitration Section: Expert findings, proposed split amounts, or final binding decision.
      \item Action bar: Contextual action buttons (``Submit Response'', ``Withdraw Dispute'', ``Escalate to Expert'').
\end{itemize}

\sublabel{Data Processing Overview}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Client requests \texttt{GET /\allowbreak{}api/\allowbreak{}disputes/\allowbreak{}:\allowbreak{}id}.
      \item \texttt{DisputeService.getDisputeDetail()} verifies caller authorization strictly: caller must be \texttt{openedById} or \texttt{respondentId}.
      \item Retrieves dispute entity with milestone, contract, and evidence files.
      \item Returns HTTP 200 with the complete dispute case file.
\end{itemize}

\fnhead{Screen Layout (Mock-up Description)}
\screenfigure[0.95]{image/screenshot/sprint3/ntin/ViewDisputeDetail.png}{Dispute case details and arbitration console}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Left column presents dispute metadata, financial escrow card (\$amount held), and response deadline clock.
      \item Center workspace renders side-by-side evidence timeline: Initial claim on the left, Counter-response on the right.
      \item Attached evidence cards provide click-to-preview modals for screenshots and PDF deliverables.
      \item Top action bar contains conditional control buttons based on caller role and case status.
\end{itemize}

\fnhead{Function Details}

\sublabel{Input Data}
\begin{specdatatable}{Data Name}{Description}
      \texttt{id} & Numeric dispute identifier (integer path parameter).\\ \hline
\end{specdatatable}

\sublabel{Validation Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Dispute ID must exist in the database.
      \item Caller must be a direct dispute participant (\texttt{openedById} or \texttt{respondentId}).
\end{itemize}

\sublabel{Business Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]\brrulelist
      \item BR-DISPUTE-06: On the user dispute endpoints, case evidence and claims are accessible exclusively to direct dispute participants (\texttt{openedById} or \texttt{respondentId}).
\end{itemize}

\sublabel{Normal Flow}
\begin{enumerate}[leftmargin=1.8em,topsep=3pt,itemsep=2pt,parsep=0pt]
      \item User clicks on a dispute entry.
      \item System validates caller authorization.
      \item System fetches and renders the complete case file including claims, evidence attachments, and timeline.
      \item User inspects evidence or triggers an available action (Respond, Withdraw, Escalate).
\end{enumerate}

\sublabel{Abnormal Flow / Exception Handling}
\begin{flowtable}{Case}{Description}
      A1 & Caller is not a dispute participant. API returns HTTP 403 Forbidden (\texttt{DisputeForbiddenException}).\\ \hline
      A2 & Dispute ID does not exist. API returns HTTP 404 Not Found: ``Dispute not found.''\\ \hline
\end{flowtable}

\sublabel{Output Data}
\begin{specdatatable}{Data Name}{Description}
      \texttt{dispute} & Comprehensive dispute detail entity including claims, evidence files, resolution status, and financial escrow allocations.\\ \hline
\end{specdatatable}

\sublabel{Post-Conditions}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Dispute case file rendered.
\end{itemize}

\subsubsection{ View Dispute by Milestone}

\fnhead{Function Trigger}
A contract workspace view renders a milestone that has an active or historical dispute.\newline
\textbf{Navigation path:} Contract Workspace $\rightarrow$ Milestones Tab $\rightarrow$ Milestone Card.

\fnhead{Function Description}

\sublabel{Actors / Roles}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item \textbf{Primary Actor:} Client, Freelancer.
      \item \textbf{Supporting Systems:} Frevia Web application, Frevia REST API, Prisma ORM, and PostgreSQL database.
\end{itemize}

\sublabel{Purpose}
\noindent Inspects dispute status directly linked to a specific milestone to display contextual warning banners, disable regular deliverable workflows (such as submission, modification, and manual escrow release), and provide seamless navigation to arbitration.

\sublabel{Interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Amber/red alert banner rendered atop the disputed milestone card.
      \item Dispute summary indicator: Current Dispute Status (\texttt{OPEN}, \texttt{WAITING\_RESPONSE}, \texttt{UNDER\_REVIEW}, \texttt{DECISION\_MADE}, \texttt{REVIEW\_REQUESTED}, \texttt{FINALIZED}, \texttt{CANCELLED}), Initiator Name, and Escrow Locked notice.
      \item Direct action button: ``View Dispute Case \#ID''.
      \item Disabled standard milestone buttons with explanatory tooltip: ``Actions disabled while under formal dispute.''
\end{itemize}

\sublabel{Data Processing Overview}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Client requests \texttt{GET /\allowbreak{}api/\allowbreak{}disputes/\allowbreak{}milestone/\allowbreak{}:\allowbreak{}milestoneId}.
      \item \texttt{DisputeService.\allowbreak{}getDisputeByMilestoneId()} validates caller is \texttt{openedById} or \texttt{respondentId}.
      \item Queries the active or most recent dispute linked to \texttt{milestoneId}.
      \item Returns HTTP 200 with dispute summary data or HTTP 404 if no dispute is associated.
\end{itemize}

\fnhead{Screen Layout (Mock-up Description)}
\noindent This function operates as an automated query and contextual state-checking service; it does not possess a standalone user interface screen. When a milestone has an active dispute, its status is reflected directly through warning indicators and an alert badge on the milestone workspace card, providing a direct action link to open the Dispute Detail drawer (\textit{UC-35.03}).

\fnhead{Function Details}

\sublabel{Input Data}
\begin{specdatatable}{Data Name}{Description}
      \texttt{milestoneId} & Numeric milestone identifier (integer path parameter).\\ \hline
\end{specdatatable}

\sublabel{Validation Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Milestone ID must exist.
      \item Caller must be \texttt{openedById} or \texttt{respondentId} of the associated dispute.
\end{itemize}

\sublabel{Business Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]\brrulelist
      \item BR-DISPUTE-07: Milestones linked to active disputes must prominently display dispute banners and disable standard milestone submission, modification, and approval buttons.
\end{itemize}

\sublabel{Normal Flow}
\begin{enumerate}[leftmargin=1.8em,topsep=3pt,itemsep=2pt,parsep=0pt]
      \item User opens the contract workspace milestones tab.
      \item The frontend queries dispute status for each milestone.
      \item For milestones with active disputes, the system renders the dispute alert banner and disables conflicting action buttons.
      \item The user clicks ``View Dispute Case'' to navigate directly to the dispute console.
\end{enumerate}

\sublabel{Abnormal Flow / Exception Handling}
\begin{flowtable}{Case}{Description}
      A1 & User not authorized on the contract. API returns HTTP 403 Forbidden.\\ \hline
      A2 & No dispute exists for this milestone. API returns HTTP 404 Not Found; frontend renders standard milestone UI.\\ \hline
\end{flowtable}

\sublabel{Output Data}
\begin{specdatatable}{Data Name}{Description}
      \texttt{dispute} & Summary dispute object with id, status, openedById, and responseDeadline.\\ \hline
\end{specdatatable}

\sublabel{Post-Conditions}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Dispute status displayed on milestone card; restricted actions locked.
\end{itemize}

\Needspace{12\baselineskip}
\subsubsection{ Respond to Dispute}

\fnhead{Function Trigger}
The designated respondent opens an open resolution case and clicks ``Accept Proposal'' or ``Decline'' inside the ``Respond to this proposal'' panel.\newline
\textbf{Navigation path:} Dispute / Milestone Detail $\rightarrow$ Resolution Drawer $\rightarrow$ ``Respond to this proposal''.

\fnhead{Function Description}

\sublabel{Actors / Roles}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item \textbf{Primary Actor:} Client, Freelancer (Designated Respondent).
      \item \textbf{Supporting Systems:} Frevia Web application, Frevia REST API, Prisma ORM, and PostgreSQL database.
\end{itemize}

\sublabel{Purpose}
\noindent Allows the respondent party to either accept the counterparty's proposed settlement or formally submit a written rebuttal and counter-evidence within the 48-hour response window before escalation to a Frevia Expert.

\sublabel{Interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item ``Resolution \#ID'' slide-over drawer displaying status badge (``Waiting Response''), 48-hour countdown timer, and 4-step progress stepper (``Request'', ``Response'', ``Expert'', ``Resolved'').
      \item Resolution summary rows: ``Proposed Resolution'', ``Proposed Split'', ``Milestone Escrow'', and opener's claim statement.
      \item ``Respond to this proposal'' panel offering ``Accept Proposal'' and ``Decline'' actions; selecting ``Decline'' expands a required explanation textarea (min 5 characters), ``Attach evidence (up to 5 files)'' upload control (``Choose file''), ``Cancel'', and crimson ``Confirm Decline'' button.
\end{itemize}

\sublabel{Data Processing Overview}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Respondent submits an HTTP request: \texttt{PATCH /\allowbreak{}api/\allowbreak{}disputes/\allowbreak{}:\allowbreak{}id/\allowbreak{}respond} with \texttt{action} (\texttt{ACCEPT} or \texttt{DECLINE}), optional \texttt{reason}, and optional \texttt{evidenceFiles}.
      \item {\sloppy \texttt{DisputeService.\allowbreak{}respondToDispute()} verifies that \texttt{userId === dispute.respondentId}, dispute status is \texttt{OPEN} or \texttt{WAITING\_RESPONSE}, and no prior decline exists (\texttt{declinedAt} null).\par}
      \item On \texttt{ACCEPT}: immediately finalizes resolution (extending milestone deadline for \texttt{EXTEND\_DEADLINE} or executing escrow payout/refund settlement and setting \texttt{FINALIZED}).
      \item On \texttt{DECLINE}: requires \texttt{reason} ($\ge 5$ chars), optionally persists up to 5 evidence files, sets \texttt{declinedAt} and status \texttt{UNDER\_REVIEW}, and notifies the opener that the dispute can be escalated to a Frevia Expert.
\end{itemize}

\fnhead{Screen Layout (Mock-up Description)}
\screenfigure[0.95]{image/screenshot/sprint3/ntin/RespondToDispute.png}{Respond to dispute rebuttal form inside Resolution Detail sheet}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Header shows ``Resolution \#0012'', milestone title, ``Waiting Response'' status, 48-hour countdown timer, and 4-stage timeline stepper (``Request'', ``Response'', ``Expert'', ``Resolved'').
      \item Summary block lists ``Proposed Resolution'', ``Proposed Split'', ``Milestone Escrow'', and the opener's reason/description.

      \item Calls \texttt{PaymentService.executeDisputeSettlementPayout()} to execute Stripe refund/transfer settlement.
      \item Inside a database transaction (\texttt{finalizeDisputeTransaction}): marks dispute \texttt{FINALIZED} (recording \texttt{decisionById}, \texttt{decisionReason}, \texttt{finalDecisionAt}), sets milestone status \texttt{COMPLETED} and \texttt{paymentStatus} (\texttt{REFUNDED} or \texttt{RELEASED}).
      \item Dispatches formal ruling notifications to Client and Freelancer.
\end{itemize}

      \item Sticky right panel titled ``Issue Final Binding Decision'' explains that the decision immediately settles milestone escrow via Stripe and closes the dispute.
      \item Linked numeric fields ``Refund to Client (\$)'' and ``Pay to Freelancer (\$)'' enforce a 100\% sum matching the top-right Escrow total.
      \item Multi-line ``Decision Reasoning *'' textarea sits above the full-width black ``Issue Final Binding Decision'' button, which triggers a final confirmation dialog before execution.

      \texttt{clientAmount} & Escrow refund amount awarded to the client (number, $\ge 0$).\\ \hline
      \texttt{freelancerAmount} & Escrow payout amount awarded to the freelancer (number, $\ge 0$).\\ \hline
      \texttt{decisionReason} & Written technical rationale and arbitration findings (string, $\ge 5$ characters).\\ \hline
\end{specdatatable}

\sublabel{Validation Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Caller must hold active \texttt{EXPERT} role (Admin is denied).
      \item Dispute status must be \texttt{UNDER\_REVIEW}, \texttt{DECISION\_MADE}, or \texttt{REVIEW\_REQUESTED}.
      \item The sum of \texttt{clientAmount} and \texttt{freelancerAmount} must equal the milestone escrow amount.
      \item Justification narrative must contain at least 5 characters.

      \item BR-DISPUTE-18: The sum of awarded amounts must mathematically equal 100\% of the disputed milestone escrow funds.
      \item BR-DISPUTE-19: Expert arbitration rulings are irrevocable, set status to \texttt{FINALIZED}, and trigger immediate financial settlement.
      \item BR-DISPUTE-20: Written justification must be provided and preserved for platform auditability and compliance.

\sublabel{Normal Flow}
\begin{enumerate}[leftmargin=1.8em,topsep=3pt,itemsep=2pt,parsep=0pt]
      \item Expert reviews case file in the evaluation console.
      \item Expert enters split allocation in ``Refund to Client (\$)'' / ``Pay to Freelancer (\$)'' and writes ``Decision Reasoning *''.
      \item The system verifies that split amounts match milestone balance.
      \item Expert clicks ``Issue Final Binding Decision'' and confirms in the confirmation dialog.
      \item The system executes Stripe settlement, updates dispute to \texttt{FINALIZED}, updates milestone payment status, and notifies parties.

\subsubsection{ Upload Dispute Evidence File}

\fnhead{Function Trigger}
A Client or Freelancer clicks ``Upload'' inside the Resolution Center drawer (when opening a dispute) or ``Choose file'' inside the ``Respond to this proposal'' section (when declining a dispute).\newline
\textbf{Navigation path:} Resolution Center / Detail Drawer $\rightarrow$ Attach Evidence Control $\rightarrow$ Select File.

\fnhead{Function Description}

\sublabel{Actors / Roles}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item \textbf{Primary Actor:} Client, Freelancer.
      \item \textbf{Secondary Actor:} Cloudinary Storage Service.
      \item \textbf{Supporting Systems:} Frevia Web application, Frevia REST API (\texttt{DisputeController}, \texttt{DisputeService}), Prisma ORM, and PostgreSQL database.
\end{itemize}

\sublabel{Purpose}
\noindent Validates, uploads, and registers supporting evidence files (documents, images, archives) so participants can attach verifiable proof to dispute claims and responses.

\sublabel{Interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Embedded evidence upload controls inside \texttt{OpenResolutionDialog} (``Upload'' button) and \texttt{ResolutionDetailSheet} (``Choose file'' button next to ``Attach evidence (up to 5 files)'').
      \item Native OS file picker for selecting proof documents or screenshots.
      \item Uploaded attachment rows showing filename and remove control before form submission.
\end{itemize}

\sublabel{Data Processing Overview}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Web client sends \texttt{POST /api/disputes/files/upload} as \texttt{multipart/form-data} with field \texttt{file}.
      \item \texttt{DisputeController.uploadEvidenceFile()} runs \texttt{FileInterceptor('file')} and \texttt{buildUploadFilePipe()}.
      \item \texttt{DisputeService.uploadEvidenceFile()} uploads the buffer to Cloudinary (\texttt{frevia/disputes}) when configured, or writes to local \texttt{uploads/disputes/} fallback storage.
      \item Persists file metadata via \texttt{DisputeRepository.createDisputeFile()} and returns \texttt{UploadDisputeFileResponseDTO}.
\end{itemize}

\fnhead{Screen Layout (Mock-up Description)}
\screenfigure[0.95]{image/screenshot/sprint3/ntin/UploadDisputeEvidenceFile.png}{Uploading dispute evidence file via the Resolution Detail sheet}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Inside the ``Resolution \#0012'' right-hand drawer, clicking ``Choose file'' next to ``Attach evidence (up to 5 files)'' opens the operating system file selector.
      \item Selected files upload asynchronously to \texttt{POST /api/disputes/files/upload} and render as removable attachment rows below the upload control prior to final submission.
\end{itemize}

\fnhead{Function Details}

\sublabel{Input Data}
\begin{specdatatable}{Data Name}{Description}
      \texttt{file} & Uploaded binary evidence file (\texttt{multipart/form-data}, field name \texttt{file}).\\ \hline
\end{specdatatable}

\sublabel{Validation Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Caller must be authenticated with an active non-banned account and valid route permission.
      \item File must be non-empty and pass \texttt{buildUploadFilePipe()} file constraints.
\end{itemize}

\sublabel{Business Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]\brrulelist
      \item BR-DISPUTE-03: At most 5 evidence files can be attached during a dispute claim.
      \item BR-DISPUTE-06: Uploaded evidence files are bound to the uploader and exposed strictly to dispute participants and assigned Experts.
\end{itemize}

\sublabel{Normal Flow}
\begin{enumerate}[leftmargin=1.8em,topsep=3pt,itemsep=2pt,parsep=0pt]
      \item User selects a supporting evidence file in the dispute modal.
      \item Web client posts the file to \texttt{POST /api/disputes/files/upload}.
      \item Server validates the file and stores it on Cloudinary or local fallback storage.
      \item Server saves metadata in PostgreSQL and returns the file record.
      \item Modal displays the uploaded evidence file chip ready to be linked on submit.
\end{enumerate}

\sublabel{Abnormal Flow / Exception Handling}
\begin{flowtable}{Case}{Description}
      A1 & No file provided or invalid file type/size. API returns HTTP 400 Bad Request (\texttt{File is required}).\\ \hline
      A2 & Caller unauthenticated or unauthorized. API returns HTTP 401/403.\\ \hline
\end{flowtable}

\sublabel{Output Data}
\begin{specdatatable}{Data Name}{Description}
      \texttt{fileUrl} & Public or protected storage URL of the uploaded evidence file.\\ \hline
      \texttt{publicId} & Storage identifier in Cloudinary or local fallback prefix.\\ \hline
      \texttt{fileName} & Original uploaded file name, byte size, and MIME type.\\ \hline
\end{specdatatable}

\sublabel{Post-Conditions}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Evidence file persisted in storage and metadata stored in PostgreSQL for attachment to a dispute.
\end{itemize}
