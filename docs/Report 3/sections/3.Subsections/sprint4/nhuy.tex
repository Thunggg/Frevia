% -----------------------------------------------------------------
\Needspace{12\baselineskip}
\subsubsection{ View Review List}

\fnhead{Function Trigger}
The actor selects View Review List.\newline
\textbf{Navigation path:} Home $\rightarrow$ \texttt{/\allowbreak{}client/\allowbreak{}contracts/\allowbreak{}[slug]} or \texttt{/\allowbreak{}freelancer/\allowbreak{}contracts/\allowbreak{}[slug].}

\fnhead{Function Description}

\sublabel{Actors / Roles}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item \textbf{Primary Actor:} Client, Freelancer.
    \item \textbf{Supporting Systems:} Frevia Web (Next.js), Frevia BFF proxy, Frevia REST API (NestJS), Prisma ORM, and PostgreSQL database.
\end{itemize}

\sublabel{Purpose}
\noindent Retrieve reviews and active responses for a contract after verifying participant access.

\sublabel{Interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Contract Reviews section with author, rating, breakdown and comment.
    \item Write Review dialog.
    \item Author-only Edit and Delete actions.
\end{itemize}

\sublabel{Data Processing Overview}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item The client issues an HTTP request to \texttt{GET /\allowbreak{}reviews/\allowbreak{}contracts/\allowbreak{}:\allowbreak{}contractId} via Next.js BFF proxy.
    \item NestJS guards verify JWT authentication and validate role-based access permissions.
    \item The backend service queries or updates entities in PostgreSQL using Prisma ORM.
    \item The API returns the structured JSON payload and handles domain invariants consistently.
\end{itemize}

\fnhead{Screen Layout (Mock-up Description)}
\screenfigure[0.95]{image/screenshot/sprint4/nhuy/View Review List.png}{View Review List interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Contract Reviews section with author, rating, breakdown and comment.
    \item Write Review dialog.
    \item Author-only Edit and Delete actions.
\end{itemize}

\fnhead{Function Details}

\sublabel{Input Data}
\begin{specdatatable}{Data Name}{Description}
Selection & Current account or selected profile\texttt{/\allowbreak{}contract}; no pagination is assumed unless exposed by the endpoint.\\ \hline
\end{specdatatable}

\sublabel{Validation Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item The pair (contractId, reviewerId) is unique.
\end{itemize}

\sublabel{Business Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]\brrulelist
    \item BR-3901-01: Only contract participants can list or view contract reviews.
    \item BR-3901-02: A review can be created only after the contract is COMPLETED, for the other party.
    \item BR-3901-03: Only the review author edits or soft-deletes their review.
    \item BR-3901-04: An update must contain at least one accepted field. The current schema does not require a non-empty comment or an integer overall rating.
\end{itemize}

\sublabel{Normal Flow}
\begin{enumerate}[leftmargin=1.8em,topsep=3pt,itemsep=2pt,parsep=0pt]
    \item The actor opens \texttt{/\allowbreak{}client/\allowbreak{}contracts/\allowbreak{}[slug]}; \texttt{/\allowbreak{}freelancer/\allowbreak{}contracts/\allowbreak{}[slug]} and selects View Review List.
    \item The application submits the selected identifier or list filters.
    \item The API checks authentication where required, authorization, resource existence and input validity.
    \item Retrieve reviews and active responses for a contract after verifying participant access.
    \item The screen renders the response, or an explicit empty state for a valid empty collection.
\end{enumerate}

\sublabel{Abnormal Flow / Exception Handling}
\begin{flowtable}{Case}{Description}
A1 & Missing or expired authentication token. Returns HTTP 401 Unauthorized.\\ \hline
A2 & Insufficient permissions or role unauthorized. Returns HTTP 403 Forbidden.\\ \hline
A3 & Target resource not found or soft-deleted. Returns HTTP 404 Not Found.\\ \hline
A4 & Validation error or malformed payload. Returns HTTP 400 Bad Request.\\ \hline
A5 & Database connectivity failure or server error. Returns HTTP 500.\\ \hline
\end{flowtable}

\sublabel{Output Data}
\begin{specdatatable}{Data Name}{Description}
Result & Retrieve reviews and active responses for a contract after verifying participant access.\\ \hline
Failure & Structured error without sensitive data; no successful UI state when the operation failed.\\ \hline
\end{specdatatable}

\sublabel{Post-Conditions}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Successful view review list produces the documented response and survives a reload where data is persisted.
    \item An unauthorized actor cannot access or modify the resource.
    \item Invalid input and missing resources do not create a false success state or expose sensitive data.
\end{itemize}
