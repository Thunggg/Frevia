% UC-43.01: Connect Stripe Account

\Needspace{12\baselineskip}
\subsubsection{ Connect Stripe Account}

\fnhead{Function Trigger}
A Freelancer clicks ``Set up Payouts'' or ``Connect Stripe'' in their Payments dashboard.\newline
\textbf{Navigation path:} Freelancer Dashboard $\rightarrow$ Payments (\texttt{/\allowbreak{}freelancer/\allowbreak{}payments}) $\rightarrow$ Click ``Connect with Stripe''.

\fnhead{Function Description}

\sublabel{Actors / Roles}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item \textbf{Primary Actor:} Freelancer (\texttt{FREELANCER} role).
      \item \textbf{Secondary Actor:} Stripe Connect Service.
      \item \textbf{Supporting Systems:} Frevia Web application (Next.js), Frevia REST API (NestJS), Stripe API SDK, Prisma ORM, and PostgreSQL database.
\end{itemize}

\sublabel{Purpose}
\noindent Enables a Freelancer to link their bank account or debit card via Stripe Connect Express onboarding. Establishing this connection is required to receive automated escrow disbursements, milestone earnings payouts, and dispute resolution settlement funds.

\sublabel{Interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Payments \& Payouts screen at \texttt{/freelancer/payments}.
      \item Status banner indicating connection state: ``Unconnected'', ``Action Required'', or ``Connected \& Verified''.
      \item Primary action button: ``Connect with Stripe'' or ``Update Payout Details''.
      \item Stripe-hosted Express onboarding flow for identity verification, tax details, and bank account entry.
      \item Return confirmation banner upon redirect back to Frevia.
\end{itemize}

\sublabel{Data Processing Overview}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Freelancer submits an HTTP request: \texttt{POST /api/payments/connect/onboarding-link} with optional \texttt{returnUrl}.
      \item \texttt{PaymentService.getOnboardingLink()} validates caller identity and role.
      \item If no Stripe Account ID exists for the user, creates a new Stripe Express account via \texttt{stripe.accounts.create()} and stores \texttt{stripeAccountId} in the database.
      \item Generates a single-use onboarding URL via \texttt{stripe.accountLinks.create()} with refresh and return routes.
      \item Freelancer completes onboarding on Stripe-hosted UI and is redirected back to Frevia.
      \item Returns HTTP 200 with the onboarding URL.
\end{itemize}

\fnhead{Screen Layout (Mock-up Description)}
\screenfigure[0.95]{image/screenshot/sprint4/ntin/ConnectStripeAccount.png}{Stripe Connect payout settings screen}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Header renders title ``Payout Settings'' and a description of secure bank transfers powered by Stripe.
      \item A prominent card displays the Stripe logo, current status badge (e.g., green ``Verified''), and bank account snippet.
      \item If unlinked, a prominent purple ``Set Up Stripe Express Payouts'' button is centered with security badges.
      \item A secondary link allows managing tax documents or changing payout schedules directly in Stripe.
\end{itemize}

\fnhead{Function Details}

\sublabel{Input Data}
\begin{specdatatable}{Data Name}{Description}
      \texttt{returnUrl} & Optional callback URL to return after onboarding (string, URL format).\\ \hline
\end{specdatatable}

\sublabel{Validation Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Caller must be an authenticated Freelancer.
      \item User account must have status \texttt{ACTIVE}.
\end{itemize}

\sublabel{Business Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]\brrulelist
      \item BR-PAYMENT-01: Freelancers must have a fully verified Stripe Connect account with payouts enabled to receive contract payments.
      \item BR-PAYMENT-02: Sensitive bank account numbers are stored exclusively within Stripe; the platform only persists the Stripe Account ID.
\end{itemize}

\sublabel{Normal Flow}
\begin{enumerate}[leftmargin=1.8em,topsep=3pt,itemsep=2pt,parsep=0pt]
      \item Freelancer navigates to Payout Settings.
      \item Freelancer clicks ``Connect with Stripe''.
      \item The system requests an onboarding link from the Stripe Connect API.
      \item Freelancer is redirected to the Stripe-hosted onboarding flow.
      \item Freelancer provides banking, identity, and tax details to Stripe.
      \item Stripe redirects the Freelancer back to Frevia.
      \item Payout status updates to ``Connected'' with a verified badge.
\end{enumerate}

\sublabel{Abnormal Flow / Exception Handling}
\begin{flowtable}{Case}{Description}
      A1 & Stripe gateway unreachable. API returns HTTP 500: ``Unable to initiate Stripe setup. Please try again later.''\\ \hline
      A2 & Unauthenticated user. API returns HTTP 401 Unauthorized.\\ \hline
\end{flowtable}

\sublabel{Output Data}
\begin{specdatatable}{Data Name}{Description}
      \texttt{onboardingUrl} & Single-use Stripe Express onboarding URL (string, URL format).\\ \hline
\end{specdatatable}

\sublabel{Post-Conditions}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Stripe account created and onboarding link generated.
\end{itemize}

% UC-43.02: Check Stripe Connect Status
% -----------------------------------------------------------------
\Needspace{12\baselineskip}
\subsubsection{ Check Stripe Connect Status}

\fnhead{Function Trigger}
An authenticated Freelancer opens the Payments dashboard to check connected Stripe payout readiness.\newline
\textbf{Navigation path:} Freelancer Dashboard $\rightarrow$ Payments (\texttt{/freelancer/payments}).

\fnhead{Function Description}

\sublabel{Actors / Roles}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item \textbf{Primary Actor:} Freelancer (authenticated user).
      \item \textbf{Secondary Actor:} Stripe Connect.
      \item \textbf{Supporting Systems:} Frevia Web application, Frevia REST API, Prisma ORM, and PostgreSQL database.
\end{itemize}

\sublabel{Purpose}
\noindent Synchronizes and verifies freelancer onboarding status, capability activation (\texttt{charges\_enabled}, \texttt{payouts\_enabled}, \texttt{details\_submitted}), and payout readiness directly with the Stripe API.

\sublabel{Interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Status indicators on Payout Settings: ``Connected'', ``Action Required'', ``Restricted''.
      \item Capabilities checklist: Charges Enabled (\checkmark), Payouts Enabled (\checkmark).
      \item Action button: ``Complete Verification'' if additional KYC requirements are due.
\end{itemize}

\sublabel{Data Processing Overview}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Client requests \texttt{GET /api/payments/connect/status}.
      \item \texttt{PaymentService.getConnectStatus()} retrieves the user's \texttt{stripeAccountId}.
      \item Calls \texttt{stripe.accounts.retrieve()} to check live account capabilities.
      \item Synchronizes \texttt{stripeOnboardingCompleted} flag in the database if state changed.
      \item Returns HTTP 200 with connection status and capability booleans.
\end{itemize}

\fnhead{Screen Layout (Mock-up Description)}
\noindent This function operates as an automated background capability inspection and account synchronization query; it does not possess a separate standalone user interface screen. Live account status and verified payout readiness indicators are rendered directly within the Stripe Connect payout management card (\textit{UC-43.01}).

\fnhead{Function Details}

\sublabel{Input Data}
\begin{specdatatable}{Data Name}{Description}
      N/A & Authenticated caller session used (userId from JWT).\\ \hline
\end{specdatatable}

\sublabel{Validation Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item User must be authenticated.
\end{itemize}

\sublabel{Business Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]\brrulelist
      \item BR-PAYMENT-03: Real-time capability synchronization ensures escrow releases are not attempted on restricted bank accounts.
\end{itemize}

\sublabel{Normal Flow}
\begin{enumerate}[leftmargin=1.8em,topsep=3pt,itemsep=2pt,parsep=0pt]
      \item User navigates to Payout Settings.
      \item System retrieves live account capabilities from Stripe.
      \item Database records are updated to match live state.
      \item System displays verified status indicator and payout readiness.
\end{enumerate}

\sublabel{Abnormal Flow / Exception Handling}
\begin{flowtable}{Case}{Description}
      A1 & No Stripe account linked yet. API returns \texttt{isConnected: false}, \texttt{payoutsEnabled: false}.\\ \hline
      A2 & Stripe API unreachable. System returns cached status from database with fallback warning.\\ \hline
\end{flowtable}

\sublabel{Output Data}
\begin{specdatatable}{Data Name}{Description}
      \texttt{isConnected} & Boolean indicating whether a Stripe account exists.\\ \hline
      \texttt{payoutsEnabled} & Boolean indicating whether bank payouts are authorized.\\ \hline
      \texttt{chargesEnabled} & Boolean indicating whether charges are enabled.\\ \hline
      \texttt{stripeAccountId} & Masked Stripe account identifier (string).\\ \hline
\end{specdatatable}

\sublabel{Post-Conditions}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Live Stripe capabilities synchronized and displayed.
\end{itemize}

\Needspace{12\baselineskip}
\subsubsection{ View Saved Payment Methods}

\fnhead{Function Trigger}
A Client navigates to the ``Payment Methods'' section in Billing \& Payments.\newline
\textbf{Navigation path:} Client Workspace $\rightarrow$ Payments (\texttt{/client/payments}).

\fnhead{Function Description}

\sublabel{Actors / Roles}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item \textbf{Primary Actor:} Client (\texttt{CLIENT} role).
      \item \textbf{Secondary Actor:} Stripe API.
      \item \textbf{Supporting Systems:} Frevia Web application, Frevia REST API, Stripe Customer API, Prisma ORM, and PostgreSQL database.
\end{itemize}

\sublabel{Purpose}
\noindent Retrieves and displays saved credit or debit cards attached to the Client's Stripe Customer profile, showing card brand, masked last 4 digits, expiration date, and default status for 1-click milestone funding.

\sublabel{Interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Payment Methods wallet management tab.
      \item Saved card cards displaying brand logo (Visa, Mastercard, Amex), masked card number (\texttt{•••• 4242}), expiration date (MM/YY), and green ``Default'' badge.
      \item Context menu on each card: ``Set as Default'', ``Remove Card''.
      \item Action button: ``+ Add Payment Method''.
\end{itemize}

\sublabel{Data Processing Overview}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Client requests \texttt{GET /api/payments/payment-methods}.
      \item \texttt{PaymentService.getSavedPaymentMethods()} retrieves the client's \texttt{stripeCustomerId}.
      \item Calls \texttt{stripe.paymentMethods.list(\{customer, type: 'card'\})}.
      \item Identifies the default payment method from the customer's \texttt{invoice\_settings}.
      \item Returns HTTP 200 with the array of formatted card objects.
\end{itemize}

      \item Top section features secure payment guarantee badge and total cards counter.
      \item Grid of credit card components showing card brand emblem, masked number, and expiry date.
      \item Default card is highlighted with a blue border and ``Default'' badge.
      \item Saved payment cards attached to the Client's Stripe Customer profile are displayed (new cards can be saved during Stripe Checkout or off-session payments).
\end{itemize}

\fnhead{Function Details}

\sublabel{Input Data}
\begin{specdatatable}{Data Name}{Description}
      N/A & Authenticated caller session used.\\ \hline
\end{specdatatable}

\sublabel{Validation Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item User must be an authenticated Client.
\end{itemize}

\sublabel{Business Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]\brrulelist
      \item BR-PAYMENT-04: Full credit card numbers and CVV codes are never stored on platform servers; only PCI-compliant Stripe tokens and masked metadata are processed.

\sublabel{Normal Flow}
\begin{enumerate}[leftmargin=1.8em,topsep=3pt,itemsep=2pt,parsep=0pt]

\begin{specdatatable}{Data Name}{Description}
      \texttt{paymentMethods[]} & Array of card objects: id, brand, last4, expMonth, expYear, isDefault, cardholderName.\\ \hline

\sublabel{Post-Conditions}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Saved payment cards rendered.
\end{itemize}

\subsubsection{ Set Default Payment Method}

\fnhead{Function Trigger}
A Client clicks ``Set as Default'' on a saved card in Payment Methods.\newline
\textbf{Navigation path:} Payment Methods $\rightarrow$ Card Options $\rightarrow$ Click ``Set as Default''.

\fnhead{Function Description}

\sublabel{Actors / Roles}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item \textbf{Primary Actor:} Client.
      \item \textbf{Secondary Actor:} Stripe Customer Service.
      \item \textbf{Supporting Systems:} Frevia Web application, Frevia REST API, Stripe API, and PostgreSQL database.
\end{itemize}

\sublabel{Purpose}
\noindent Designates a preferred saved credit card as the customer's default payment method in Stripe, ensuring it is automatically pre-selected during milestone escrow funding and contract hold authorizations.

\sublabel{Interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item ``Set as Default'' button on non-default card cards.
      \item Confirmation toast upon successful update.
      \item Immediate visual badge transition to ``Default''.
\end{itemize}

\sublabel{Data Processing Overview}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]

      \item Calls \texttt{stripe.customers.update(customerId, \{invoice\_settings: \{default\_payment\_method\}\})}.

\Needspace{12\baselineskip}
\subsubsection{ Create Contract Hold Deposit Intent}

\fnhead{Function Trigger}
A Client initiates contract creation or signs a contract requiring an upfront security hold deposit.\newline
\textbf{Navigation path:} Contract Workspace $\rightarrow$ Proposal Review $\rightarrow$ Click ``Fund Contract / Authorize Hold''.

\fnhead{Function Description}

\sublabel{Actors / Roles}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item \textbf{Primary Actor:} Client (\texttt{CLIENT} role).
      \item \textbf{Secondary Actor:} Stripe PaymentIntent Service.
      \item \textbf{Supporting Systems:} Frevia Web application, Frevia REST API, Stripe SDK, Prisma ORM, and PostgreSQL database.
\end{itemize}

\sublabel{Purpose}
\noindent Creates a Stripe PaymentIntent with manual capture authorization (\texttt{capture\_method: 'manual'}) when a Client signs or initiates a contract. This pre-authorizes funds for initial milestones plus platform service fees (3\%), freezing the credit line without charging immediately until the Freelancer countersigns.

\sublabel{Interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Contract hold authorization dialog.
      \item Milestone selection checklist (Client can select initial or all milestones).
      \item Financial breakdown: Selected Milestones Amount + Platform Service Fee (3\%) = Total Hold Amount.
      \item Stripe Elements card form or saved card selector.
      \item Action button: ``Authorize Hold Deposit (\$X,XXX)''.
\end{itemize}

\sublabel{Data Processing Overview}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Client requests \texttt{POST /api/payments/contracts/:contractId/fund-hold/intent} with \texttt{milestoneIds}, optional \texttt{paymentMethodId}, and \texttt{returnUrl}.
      \item \texttt{PaymentService.createContractHoldIntent()} validates caller is contract Client.
      \item Calculates total hold amount (sum of selected milestones + 3\% platform fee).
      \item Creates Stripe PaymentIntent with \texttt{capture\_method: 'manual'} and contract metadata.
      \item Saves \texttt{holdPaymentIntentId} and \texttt{holdMilestoneIds} on the Contract record.
      \item Returns HTTP 200 with \texttt{clientSecret}, \texttt{paymentIntentId}, and amount summary.
\end{itemize}

\fnhead{Screen Layout (Mock-up Description)}
\screenfigure[0.95]{image/screenshot/sprint4/ntin/CreateContractHoldDepositIntent.png}{Contract hold deposit pre-authorization screen}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item Header renders security shield icon and title ``Contract Escrow Hold Deposit''.
      \item Itemized summary shows initial milestone amounts, platform guarantee fee (3\%), and total hold authorization.
      \item Notice banner clarifies that funds are frozen on the card and will only be captured once the Freelancer signs.
      \item Primary blue button ``Authorize Escrow Hold Deposit''.
\end{itemize}

\fnhead{Function Details}

\sublabel{Input Data}
\begin{specdatatable}{Data Name}{Description}
      \texttt{contractId} & Numeric contract identifier (integer path parameter).\\ \hline
      \texttt{milestoneIds} & Array of numeric milestone IDs to fund upfront (integer array, min 1 item).\\ \hline
      \texttt{paymentMethodId} & Optional saved card ID (string).\\ \hline
      \texttt{returnUrl} & Optional redirect URL for 3DS authentication (string, URL format).\\ \hline

      \item \textbf{Supporting Systems:} Frevia Web application, Frevia REST API, Stripe Transfer API, Prisma ORM, and PostgreSQL database.
\end{itemize}

\sublabel{Purpose}

\sublabel{Interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
      \item ``Approve \& Release Funds'' button on funded milestone submissions.

\sublabel{Data Processing Overview}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]

      \item \texttt{PaymentService.\allowbreak{}release\allowbreak{}Milestone()} verifies caller is contract Client and milestone is in \texttt{FUNDED} status.
      \item Verifies Freelancer has an active, verified Stripe account with \texttt{payouts\_\allowbreak{}enabled = true}.
      \item Calculates platform commission (5\%) and net freelancer amount (95\%).

      \item Updates milestone paymentStatus to \texttt{RELEASED} and status to \texttt{COMPLETED}.
      \item Creates ledger transaction records for \texttt{MILESTONE\_PAYOUT} and \texttt{PLATFORM\_FEE}.
      \item Dispatches payout notifications to Freelancer.
