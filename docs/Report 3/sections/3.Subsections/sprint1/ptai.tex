\Needspace{12\baselineskip}
\subsubsection{View Jobs List}

\fnhead{Function Trigger}
A guest, freelancer, or client opens the public Find Work catalog or freelancer Find Work tab to browse available job postings.\newline
\textbf{Navigation path:} Public Find Work (\texttt{/find-work}) or Freelancer Workspace $\rightarrow$ Find Work (\texttt{/\allowbreak{}freelancer/\allowbreak{}find-\allowbreak{}work}).

\fnhead{Function Description}

\sublabel{Actors / Roles}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item \textbf{Primary Actor:} Guest, Freelancer, and Client (\texttt{@IsPublic()}).
    \item \textbf{Supporting Systems:} Frevia Web application (Next.js), Frevia REST API (\texttt{BrowseJobController}), Prisma ORM, and PostgreSQL database.
\end{itemize}

\sublabel{Purpose}
\noindent Allows visitors and authenticated users to browse, search, filter, and sort active \texttt{OPEN} job postings with pagination, overlapping budget range checks, skill matching, and active category slug filters.

\sublabel{Interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item \textbf{Freelancer view (\texttt{/\allowbreak{}freelancer/\allowbreak{}find-\allowbreak{}work}):} Search bar, tabs (Best Matches, Most Recent, Saved Searches, Saved Jobs, Invites, Find Clients), job summary cards, bookmark toggles, and profile progress sidebar.
    \item \textbf{Guest/Public view (\texttt{/find-work}):} Public header navigation, hero keyword/category/budget filter bar, job cards with direct ``Apply'' prompt, category quick-filter list, and registration CTA panel.
\end{itemize}

\sublabel{Data Processing Overview}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Client dispatches \texttt{GET /api/jobs} with optional query parameters (\texttt{page}, \texttt{limit}, \texttt{search}, \texttt{status}, \texttt{budgetType}, \texttt{hiringType}, \texttt{budgetMin}, \texttt{budgetMax}, \texttt{createdAfter}, \texttt{skill}, \texttt{category}, \texttt{clientId}, \texttt{sortBy}, \texttt{order}).
    \item Backend validates inputs via \texttt{ViewListJobQueryDto} enforcing \texttt{budgetMin <= budgetMax} and \texttt{limit <= 20}.
    \item \texttt{BrowseJobRepository.getJobLists()} applies mandatory filters: \texttt{deletedAt = null}, \texttt{status = 'OPEN'}, and \texttt{client: \{ isBanned: false, deletedAt: null \}}.
    \item Keyword search checks \texttt{title}, \texttt{description}, and linked \texttt{skills} case-insensitively; budget filters match overlapping bounds; category filters match only \texttt{ACTIVE} non-deleted categories.
    \item Executes \texttt{prisma.job.findMany} and \texttt{prisma.job.count} inside a transaction and returns normalized jobs with pagination metadata.
\end{itemize}

\fnhead{Screen Layout (Mock-up Description)}
\screenfigure[0.95]{image/screenshot/sprint1/ptai/ViewJobList.png}{Job browsing list interface (Freelancer view)}
\screenfigure[0.95]{image/screenshot/sprint1/ptai/ViewJobListGuest.png}{Job browsing list interface (Guest/Public view)}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item \textbf{Freelancer view (\texttt{/freelancer/find-work}):} Search bar, workspace tabs, open job cards displaying status, budget, deadlines, required skills, category badge, and right-hand profile card.
    \item \textbf{Guest/Public view (\texttt{/find-work}):} Hero search bar with category and budget selectors, job feed cards, right sidebar with freelancer registration CTA, and clickable active categories list.
\end{itemize}

\fnhead{Function Details}

\sublabel{Input Data}
\begin{specdatatable}{Data Name}{Description}
\texttt{page} & Integer $\ge 1$ (default 1).\\ \hline
\texttt{limit} & Integer between 1 and 20 (default 10).\\ \hline
\texttt{search} & Optional trimmed string ($\le 255$ chars) matched against title, description, or skill names.\\ \hline
\texttt{budgetMin} / \texttt{budgetMax} & Optional non-negative numbers (\texttt{budgetMin <= budgetMax}).\\ \hline
\texttt{skill} / \texttt{category} & Optional comma-separated strings ($\le 1000$ chars) for skill names or active category slugs.\\ \hline
\texttt{hiringType} / \texttt{budgetType} & Optional enums (\texttt{SINGLE}/\texttt{MULTIPLE}; \texttt{FIXED\_PRICE}).\\ \hline
\texttt{sortBy} / \texttt{order} & \texttt{createdAt}, \texttt{updatedAt}, \texttt{title}, \texttt{budgetMin}, \texttt{budgetMax}, \texttt{deadline}, or \texttt{expiryDate}; \texttt{asc}/\texttt{desc} (default \texttt{createdAt desc}).\\ \hline
\end{specdatatable}

\sublabel{Validation Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item \texttt{page} must be $\ge 1$; \texttt{limit} must be between 1 and 20.
    \item When both \texttt{budgetMin} and \texttt{budgetMax} are specified, \texttt{budgetMin <= budgetMax}.
\end{itemize}

\sublabel{Business Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]\brrulelist
    \item BR-JOB-01: Public Open Visibility: Public job listings display strictly non-deleted jobs in \texttt{OPEN} status belonging to active, non-banned clients; \texttt{DRAFT}, \texttt{CLOSED}, \texttt{IN\_PROGRESS}, \texttt{COMPLETED}, \texttt{CANCELLED}, and soft-deleted jobs are excluded.
    \item BR-JOB-02: Multi-Criteria Overlapping Filter and Search: Text search matches case-insensitively across title, description, and skill names; budget bounds match overlapping offered budget ranges; skill filters support single or multi-skill matching; all active filters combine using logical AND.
    \item BR-JOB-03: Active Category Filter Visibility: Category filtering matches strictly non-deleted categories in \texttt{ACTIVE} status; inactive or soft-deleted categories return no public matches for that filter.
    \item BR-JOB-04: Pagination and Default Ordering: Public job browsing enforces a positive page number and 1--20 items per page (default 10), sorted newest-first by creation timestamp unless overridden by supported sort options.
\end{itemize}

\sublabel{Normal Flow}
\begin{enumerate}[leftmargin=1.8em,topsep=3pt,itemsep=2pt,parsep=0pt]
    \item User opens \texttt{/freelancer/find-work} or \texttt{/find-work}.
    \item Frontend dispatches \texttt{GET /api/jobs} with active query parameters.
    \item Backend validates filter DTO and queries matching \texttt{OPEN} jobs alongside total count.
    \item Returns paginated jobs array and pagination metadata.
    \item UI renders job summary cards and page controls.
\end{enumerate}

\sublabel{Abnormal Flow / Exception Handling}
\begin{flowtable}{Abnormal Case}{Handling \& System Response}
Invalid \texttt{page < 1} or \texttt{limit > 20} & Zod DTO validator rejects with HTTP 400 Bad Request (\texttt{INVALID\_PAGE} / \texttt{INVALID\_LIMIT}); UI resets to default pagination.\\ \hline
\texttt{budgetMin > budgetMax} or negative budget & Custom schema refinement throws HTTP 400 Bad Request (\texttt{BUDGET\_MAX\_MUST\_BE\_GREATER\_THAN\_BUDGET\_MIN}).\\ \hline
Unsupported \texttt{sortBy}, \texttt{order}, or enum value & DTO rejects unknown enum strings with HTTP 400 Bad Request; UI restores default sorting.\\ \hline
Jobs belong to banned clients or non-OPEN states & Automatically excluded by repository \texttt{where} clause so restricted or draft jobs never leak into listings.\\ \hline
No jobs match entered filters & Returns \texttt{data: []} and \texttt{total: 0}; UI displays empty state with filter reset action.\\ \hline
Database query error or timeout & Service catches \texttt{PrismaClientKnownRequestError} and throws \texttt{FailedToLoadJobListException} (HTTP 500).\\ \hline
\end{flowtable}

\sublabel{Output Data}
\begin{specdatatable}{Data Name}{Description}
\texttt{data} & Array of \texttt{JobSchema} records with normalized budgets, skills, categories, and proposal count.\\ \hline
\texttt{pagination} & Object containing \texttt{page}, \texttt{limit}, \texttt{total}, and \texttt{totalPages}.\\ \hline
\end{specdatatable}

\sublabel{Post-Conditions}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Paginated open job feed matching active filters is displayed.
\end{itemize}

\Needspace{12\baselineskip}
\subsubsection{View Job Details}

\fnhead{Function Trigger}
A freelancer/visitor opens a job posting detail page by URL slug, or a client opens their owned job detail page by numeric ID.\newline
\textbf{Navigation path:} Freelancer Job Detail (\texttt{/freelancer/jobs/:slug}) or Client Job Detail (\texttt{/client/jobs/:jobId}).

\fnhead{Function Description}

\sublabel{Actors / Roles}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item \textbf{Primary Actor:} Freelancer, Guest, and Client.
    \item \textbf{Supporting Systems:} Frevia Web application (Next.js), Frevia REST API (\texttt{BrowseJobController} and \texttt{ManageJobController}), Prisma ORM, and PostgreSQL database.
\end{itemize}

\sublabel{Purpose}
\noindent Displays complete job specifications tailored to the viewer's role: public/freelancer view exposes client info, bookmark status, and proposal application controls via slug (\texttt{GET /api/jobs/:slug}), whereas client owner view exposes job management metrics across all statuses via integer ID (\texttt{GET /api/manage-jobs/:id}).

\sublabel{Interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item \textbf{Freelancer view (\texttt{/freelancer/jobs/:slug}):} Left column showing full job description, required skills, job categories, and related open jobs; right sidebar displaying budget, hiring type, dates, client profile card, Bookmark button, and ``Apply now'' CTA.
    \item \textbf{Client view (\texttt{/client/jobs/:jobId}):} Management breadcrumb (\texttt{My Jobs / Job \#id}), title, posting date, full description, 5-column metrics bar (Budget, Deadline, Status, Hiring Type, ``Check proposals $\rightarrow$''), and required skills pills.
\end{itemize}

\sublabel{Data Processing Overview}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item \textbf{Public / Freelancer lookup:} Calls \texttt{GET /api/jobs/:slug} (\texttt{@IsPublic()}), filtering strictly \texttt{deletedAt = null}, \texttt{status = 'OPEN'}, and non-banned active client. Throws \texttt{JobNotFoundException} (404) otherwise.
    \item \textbf{Client owner lookup:} Calls \texttt{GET /api/manage-jobs/:id} verifying \texttt{roleName === 'CLIENT'} and \texttt{clientId === userId} on non-deleted records across any status (\texttt{DRAFT}, \texttt{OPEN}, \texttt{IN\_PROGRESS}, \texttt{CLOSED}, \texttt{COMPLETED}).
\end{itemize}

\fnhead{Screen Layout (Mock-up Description)}
\screenfigure[0.95]{image/screenshot/sprint1/ptai/ViewJobDetail.png}{Job detail interface (Freelancer view)}
\screenfigure[0.95]{image/screenshot/sprint1/ptai/ViewJobDetailClient.png}{Job detail interface (Client view)}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item \textbf{Freelancer view (\texttt{/freelancer/jobs/:slug}):} Title, budget banner, ``Bookmark'' and ``Apply now'' action bar, detailed scope, skill/category badges, and right summary card with client link.
    \item \textbf{Client view (\texttt{/client/jobs/:jobId}):} Client owner layout with breadcrumb, full description, inline metrics bar including ``Check proposals $\rightarrow$'' link to candidate proposals, and required skills badges.
\end{itemize}

\fnhead{Function Details}

\sublabel{Input Data}
\begin{specdatatable}{Data Name}{Description}
\texttt{slug} & URL slug string (for public/freelancer \texttt{GET /api/jobs/:slug}).\\ \hline
\texttt{id} & Positive integer job ID (for client \texttt{GET /api/manage-jobs/:id}).\\ \hline
\end{specdatatable}

\sublabel{Validation Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Public lookup requires non-empty valid \texttt{slug}; client lookup parses \texttt{id} via \texttt{ParseIntPipe} and enforces \texttt{CLIENT} role.
\end{itemize}

\sublabel{Business Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]\brrulelist
    \item BR-JOB-05: Public vs. Owner Visibility Boundary: Public job detail access exposes strictly non-deleted \texttt{OPEN} jobs of active, non-banned clients; non-owners cannot inspect \texttt{DRAFT}, \texttt{CLOSED}, \texttt{IN\_PROGRESS}, \texttt{COMPLETED}, \texttt{CANCELLED}, or soft-deleted jobs.
    \item BR-JOB-06: Client Owner Job Detail Isolation: Client management detail views restrict access to the \texttt{CLIENT} role and strictly isolate non-deleted job inspection to the owning client across all lifecycle statuses.
    \item BR-JOB-07: Complete Metadata, Quota, and Apply Eligibility: Job details expose fixed-price budget bounds, distinct proposal \texttt{expiryDate} vs. project \texttt{deadline}, hiring quota (\texttt{SINGLE} or \texttt{MULTIPLE} with filled vs. required slots), skills, and category tags; applying is permitted only while \texttt{OPEN}, unexpired, and under quota.
\end{itemize}

\sublabel{Normal Flow}
\begin{enumerate}[leftmargin=1.8em,topsep=3pt,itemsep=2pt,parsep=0pt]
    \item Viewer opens \texttt{/freelancer/jobs/:slug} (or client owner opens \texttt{/client/jobs/:jobId}).
    \item Frontend dispatches \texttt{GET /api/jobs/:slug} (or \texttt{GET /api/manage-jobs/:id}).
    \item Backend verifies visibility/ownership constraints and retrieves job relations.
    \item UI renders either the Freelancer application view or the Client management overview.
\end{enumerate}

\sublabel{Abnormal Flow / Exception Handling}
\begin{flowtable}{Abnormal Case}{Handling \& System Response}
Missing or invalid token on client route & Global \texttt{AuthGuard} returns HTTP 401 Unauthorized on \texttt{/api/manage-jobs/:id}; client redirects to \texttt{/login}.\\ \hline
Non-client accesses \texttt{/api/manage-jobs/:id} & Service throws \texttt{JobOnlyForClientException} (HTTP 403 Forbidden).\\ \hline
Non-numeric \texttt{id} on client route & \texttt{ParseIntPipe} rejects invalid ID format with HTTP 400 Bad Request.\\ \hline
Job missing, deleted, or banned client (public route) & \texttt{BrowseJobRepository.viewJobDetail()} throws \texttt{JobNotFoundException} (HTTP 404); Next.js renders 404 page.\\ \hline
Non-owner client requests \texttt{/api/manage-jobs/:id} & \texttt{getClientJobDetail()} returns null; service throws \texttt{JobNotFoundException} (HTTP 404 Not Found).\\ \hline
Database query failure & Catches \texttt{PrismaClientKnownRequestError} and throws \texttt{FailedToLoadJobDetailException} (HTTP 500).\\ \hline
\end{flowtable}

\sublabel{Output Data}
\begin{specdatatable}{Data Name}{Description}
\texttt{ViewJobDetailResSchema} & Detailed job object containing \texttt{id}, \texttt{slug}, \texttt{clientId}, \texttt{title}, \texttt{description}, budget bounds, dates, \texttt{status}, quota, \texttt{skills}, and \texttt{jobCategories}.\\ \hline
\end{specdatatable}

\sublabel{Post-Conditions}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Role-appropriate job specifications and next actions are presented cleanly.
\end{itemize}

\subsubsection{View My Jobs}

\fnhead{Function Trigger}
A client selects ``My Jobs'' from the Client workspace sidebar.\newline
\textbf{Navigation path:} Client Workspace $\rightarrow$ My Jobs (\texttt{/client/jobs}).

\fnhead{Function Description}

\sublabel{Actors / Roles}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item \textbf{Primary Actor:} Client.
    \item \textbf{Supporting Systems:} Frevia Web application (Next.js), Frevia REST API (\texttt{ManageJobController}), Prisma ORM, and PostgreSQL database.
\end{itemize}

\sublabel{Purpose}
\noindent Provides clients a centralized dashboard to inspect, search, filter by lifecycle status, and manage all their posted jobs along with proposal count summaries and row-level actions.

\sublabel{Interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Client job management screen at \texttt{/client/jobs}.
    \item Top KPI summary cards (Active Recruiting, Total Proposals, In Progress, Completed), ``Draft Box'', and ``+ Post New Job'' button.
    \item Status tabs (All, Recruiting, In Progress, Completed, Closed, Drafts), title search input, and job cards with ``View detail'', ``View Proposals'', and 3-dot dropdown actions menu.
\end{itemize}

\sublabel{Data Processing Overview}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Client requests \texttt{GET /api/manage-jobs} with optional filter query params (\texttt{page}, \texttt{limit}, \texttt{search}, \texttt{status}, \texttt{sortBy}, \texttt{order}).
    \item Backend enforces \texttt{roleName === 'CLIENT'} (\texttt{JobOnlyForClientException} HTTP 403).
    \item Queries non-deleted jobs scoped strictly to \texttt{clientId === userId} with proposal counts and pagination.
\end{itemize}

\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Caller must be authenticated with active \texttt{CLIENT} role.
    \item Status/filter query params must conform to \texttt{ViewClientJobQueryDto}.

\sublabel{Business Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]\brrulelist

    \item BR-JOB-09: Lifecycle Status and Proposal Aggregate Visibility: Clients can inspect and filter their own jobs across all lifecycle statuses (\texttt{DRAFT}, \texttt{OPEN}, \texttt{IN\_PROGRESS}, \texttt{CLOSED}, \texttt{COMPLETED}, \texttt{CANCELLED}) alongside live total proposal counts.
    \item BR-JOB-10: Lazy Auto-Close of Expired Open Jobs: Whenever a Client's job list is retrieved, any non-deleted \texttt{OPEN} job whose \texttt{expiryDate} has passed automatically closes to \texttt{CLOSED}, while its pending \texttt{SUBMITTED} proposals transition atomically to \texttt{EXPIRED}.
\end{itemize}

\sublabel{Normal Flow}
\begin{enumerate}[leftmargin=1.8em,topsep=3pt,itemsep=2pt,parsep=0pt]
    \item Client navigates to \texttt{/client/jobs}.
    \item Frontend requests \texttt{GET /api/manage-jobs}.
    \item Backend validates \texttt{CLIENT} role and queries owner's non-deleted jobs.
    \item Returns paginated jobs array with proposal aggregates.
    \item UI displays KPI cards, status tabs, and actionable job cards.

\sublabel{Abnormal Flow / Exception Handling}

Non-client active role (e.g. Freelancer) & Service throws \texttt{JobOnlyForClientException} (HTTP 403 Forbidden).\\ \hline
Invalid \texttt{status}, \texttt{page}, or \texttt{limit > 20} & Zod DTO validator rejects with HTTP 400 Bad Request; UI resets invalid filters.\\ \hline
Client has not created any jobs yet & Repository returns \texttt{data: []} and \texttt{total: 0}; UI displays empty state with ``+ Post New Job'' CTA.\\ \hline
Database query failure & Prisma error propagates HTTP 500 Internal Server Error; UI offers retry.\\ \hline
\end{flowtable}

\sublabel{Output Data}
\begin{specdatatable}{Data Name}{Description}
Client Job List & Paginated \texttt{ViewListJobResponseDto} containing owned jobs and proposal counters.\\ \hline
\end{specdatatable}

\sublabel{Post-Conditions}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Client views portfolio of owned jobs and management shortcuts.

\Needspace{12\baselineskip}
\subsubsection{Create Job}

\fnhead{Function Trigger}
A client clicks ``+ Post New Job'' from the My Jobs dashboard.\newline
\textbf{Navigation path:} My Jobs $\rightarrow$ Click ``+ Post New Job'' (\texttt{/client/jobs/new}).

\fnhead{Function Description}

\sublabel{Actors / Roles}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item \textbf{Primary Actor:} Client.
    \item \textbf{Supporting Systems:} Frevia Web application (Next.js), Frevia REST API (\texttt{ManageJobController}), \texttt{FastTextService}, \texttt{EmbeddingOutboxService}, \texttt{CacheService}, Prisma ORM, and PostgreSQL database.
\end{itemize}

\sublabel{Purpose}
\noindent Allows authenticated clients to compose and store a new job posting initialized in \texttt{DRAFT} status with validated English title/description, fixed budget range, future proposal expiry and work deadline, 1--10 required skills, 1 active category, and hiring quota.

\sublabel{Interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Job creation page at \texttt{/client/jobs/new}.
    \item Inputs for Job title (``English only'' tag), Description (``English only'' tag), Minimum/Maximum budget (\$), Hiring type (\texttt{Hire One} / \texttt{Hire Multiple}) + Positions needed, Work delivery deadline, Proposal submission deadline, Required skills picker, and Job category picker.
\end{itemize}

\sublabel{Data Processing Overview}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item {\sloppy Frontend submits \texttt{POST /\allowbreak{}api/\allowbreak{}manage-\allowbreak{}jobs} validated by \texttt{CreateJobBodySchema.\allowbreak{}strict()}.\par}
    \item Backend enforces \texttt{CLIENT} role and verifies both \texttt{title} and \texttt{description} are English via \texttt{FastTextService.isEnglish()}.
    \item Inside a Prisma transaction: verifies submitted \texttt{jobCategories} exist, are active, and non-deleted; generates unique slug; creates \texttt{Job} record defaulting to \texttt{status = 'DRAFT'}; attaches \texttt{JobSkill} and \texttt{JobJobCategory} mappings.
    \item Invalidates \texttt{job\_categor*} cache entries and enqueues an \texttt{UPSERT} job embedding event via \texttt{EmbeddingOutboxService}.
\end{itemize}

\fnhead{Screen Layout (Mock-up Description)}
\screenfigure[0.95]{image/screenshot/sprint1/ptai/CreateJob.png}{Post New Job creation form interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Single-page clean rounded input layout presenting English-enforced Job title and Description fields, side-by-side Budget (\$1500 / \$3200), Hiring type selector, Deadline and Proposal submission dates, and Required skills picker.
\end{itemize}

\fnhead{Function Details}

\sublabel{Input Data}
\begin{specdatatable}{Data Name}{Description}
\texttt{title} & Required trimmed English string (10--255 characters).\\ \hline
\texttt{description} & Required trimmed English string (20--5000 characters).\\ \hline
\texttt{budgetMin} / \texttt{budgetMax} & Numbers $\ge 10$ USD with \texttt{budgetMin <= budgetMax}; \texttt{budgetType = 'FIXED\_PRICE'}.\\ \hline
\texttt{deadline} / \texttt{expiryDate} & Future dates satisfying \texttt{now < expiryDate < deadline}.\\ \hline
\texttt{skills} & Array of 1 to 10 positive integer skill IDs.\\ \hline
\texttt{jobCategories} & Array of exactly 1 positive integer active category ID.\\ \hline
\texttt{hiringType} / \texttt{positionsRequired} & \texttt{SINGLE} (\texttt{positionsRequired = 1}) or \texttt{MULTIPLE} (\texttt{2..100}).\\ \hline
\end{specdatatable}

\sublabel{Validation Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item \texttt{CreateJobBodySchema.strict()} validates all string bounds, numeric minimums, date ordering, single category, 1--10 skills, and hiring quota rules.
    \item \texttt{FastTextService} checks English language compliance on both title and description.
\end{itemize}

\sublabel{Business Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]\brrulelist
    \item BR-JOB-11: Client Creation and Draft Initialization: Only authenticated \texttt{CLIENT} accounts may create jobs; every newly created job initializes in \texttt{DRAFT} status with a generated unique URL slug, and callers cannot assign arbitrary initial statuses.
    \item BR-JOB-12: English Language and Duplicate Title Enforcement: Job title (10--255 characters) and description (20--5000 characters) must pass English language detection; pre-validation checks also warn against case-insensitive duplicate job titles owned by the same client.
    \item BR-JOB-13: Budget, Timeline, and Skill Bounds: Fixed-price jobs require minimum and maximum budgets of at least \$10 with \texttt{budgetMin} $\le$ \texttt{budgetMax}, valid future timelines where current time $<$ \texttt{expiryDate} $<$ \texttt{deadline}, and 1--10 valid skills.
    \item BR-JOB-14: Single Active Category Assignment: Every job must link to exactly one non-deleted \texttt{ACTIVE} job category; selecting multiple categories or an inactive category is prohibited.
    \item BR-JOB-15: Hiring Quota and Sync Side Effects: \texttt{SINGLE} hiring fixes required positions at 1 and \texttt{MULTIPLE} allows 2--100 positions with filled positions initialized at 0; job creation invalidates category caches and enqueues an embedding index update.
\end{itemize}

\begin{enumerate}[leftmargin=1.8em,topsep=3pt,itemsep=2pt,parsep=0pt]
    \item Client fills title, description, budgets, dates, skills, and category on \texttt{/client/jobs/new} and submits.
    \item {\sloppy Frontend checks language/duplicate warnings via \texttt{/\allowbreak{}api/\allowbreak{}manage-\allowbreak{}jobs/\allowbreak{}check-\allowbreak{}language} and submits \texttt{POST /\allowbreak{}api/\allowbreak{}manage-\allowbreak{}jobs}.\par}
    \item Backend verifies \texttt{CLIENT} role, FastText English compliance, and active category existence.
    \item Creates job in \texttt{DRAFT} status with relational skills and category inside a transaction.
    \item Returns created job; UI navigates/notifies client.
\end{enumerate}

\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item ``Reopen'' option inside a Closed job's dropdown menu returning the job to Recruiting with a confirmation toast.
\end{itemize}

\sublabel{Data Processing Overview}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Client dispatches \texttt{PATCH /api/manage-jobs/:id/reopen}.
    \item Backend enforces \texttt{CLIENT} role and checks ownership (\texttt{checkJobOwner}).
    \item Confirms \texttt{job.status === 'CLOSED'} (\texttt{JobCannotBeReopenedException} HTTP 400) and checks elapsed days since \texttt{job.updatedAt} does not exceed 60 days (\texttt{JobReopenPeriodExpiredException} HTTP 400).
    \item Sets \texttt{status = 'OPEN'}, invalidates category cache, and enqueues \texttt{UPSERT} vector event.
\end{itemize}

\fnhead{Screen Layout (Mock-up Description)}
\screenfigure[0.95]{image/screenshot/sprint1/ptai/ReopenJob.png}{Reopen job action and confirmation toast interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Closed tab list reflecting card removal back into Recruiting alongside confirmation toast ``Job reopened successfully''.
\end{itemize}

\fnhead{Function Details}

\sublabel{Input Data}
\begin{specdatatable}{Data Name}{Description}
\texttt{id} & Positive integer ID of owned \texttt{CLOSED} job.\\ \hline
\end{specdatatable}

\sublabel{Validation Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Job must belong to caller, hold \texttt{CLOSED} status, and have been updated within the last 60 days.
\end{itemize}

\sublabel{Business Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]\brrulelist
    \item BR-JOB-26: Closed-State Ownership Requirement on Reopen: Only the owning \texttt{CLIENT} may reopen a job currently in \texttt{CLOSED} status; jobs in \texttt{DRAFT}, \texttt{OPEN}, \texttt{IN\_PROGRESS}, \texttt{COMPLETED}, or \texttt{CANCELLED} status cannot be reopened.
    \item BR-JOB-27: 60-Day Reopen Window Enforcement: Closed jobs may reopen only within 60 days of their last update timestamp; requests past the 60-day window are rejected.
    \item BR-JOB-28: No Revival of Expired Proposals and Index Restoration: Reopening restores job status to \texttt{OPEN} and re-indexes it for discovery, without reviving previously \texttt{EXPIRED} proposals; freelancers must submit a new proposal if eligible.
\end{itemize}

\sublabel{Normal Flow}
\begin{enumerate}[leftmargin=1.8em,topsep=3pt,itemsep=2pt,parsep=0pt]
    \item Client switches to the Closed tab on \texttt{/client/jobs} and clicks ``Reopen'' on a closed job.
    \item Frontend dispatches \texttt{PATCH /api/manage-jobs/:id/reopen}.
    \item Backend validates ownership, \texttt{CLOSED} status, and 60-day window.
    \item Sets \texttt{status = 'OPEN'}, clears category cache, and triggers vector upsert.
    \item UI displays ``Job reopened successfully'' toast and moves card back to Recruiting.
\end{enumerate}

\sublabel{Abnormal Flow / Exception Handling}
\begin{flowtable}{Abnormal Case}{Handling \& System Response}
Missing or invalid authentication token & Global \texttt{AuthGuard} returns HTTP 401 Unauthorized; client redirects to \texttt{/login}.\\ \hline
Non-client active role & Service throws \texttt{JobOnlyForClientException} (HTTP 403 Forbidden).\\ \hline
Job missing, deleted, or unowned & \texttt{checkJobOwner()} throws \texttt{JobNotFoundException} (HTTP 404 Not Found).\\ \hline
Job is not currently \texttt{CLOSED} & Repository throws \texttt{JobCannotBeReopenedException} (HTTP 400 Bad Request).\\ \hline
Elapsed time since closure exceeds 60 days & Repository throws \texttt{JobReopenPeriodExpiredException} (HTTP 400 Bad Request).\\ \hline
Database update failure & Service catches Prisma error and throws \texttt{FailedToReopenJobException} (HTTP 500).\\ \hline
\end{flowtable}

\sublabel{Output Data}
\begin{specdatatable}{Data Name}{Description}
Reopened Job & Updated \texttt{JobSchema} entity with \texttt{status = OPEN}.\\ \hline
\end{specdatatable}

\sublabel{Post-Conditions}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Job returns to \texttt{OPEN} status and reappears in search and category listings.
\end{itemize}

\subsubsection{Bookmark Job}

\fnhead{Function Trigger}
A freelancer clicks the Bookmark icon on an open job card or job detail header.\newline
\textbf{Navigation path:} Job Detail (\texttt{/freelancer/jobs/:slug}) $\rightarrow$ Click Bookmark icon.

\fnhead{Function Description}

\sublabel{Actors / Roles}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item \textbf{Primary Actor:} Freelancer.
    \item \textbf{Supporting Systems:} Frevia Web application (Next.js), Frevia REST API (\texttt{ManageJobController}), Prisma ORM, and PostgreSQL database.
\end{itemize}

\sublabel{Purpose}
\noindent Saves an active, open, non-expired job posting to the freelancer's bookmark list while enforcing ownership, availability, and duplicate-entry checks.

\sublabel{Interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Bookmark toggle icon on job cards and job detail header updating to active state paired with green confirmation toast ``Job saved to bookmarks''.
\end{itemize}

\sublabel{Data Processing Overview}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Client dispatches \texttt{POST /\allowbreak{}api/\allowbreak{}manage-\allowbreak{}jobs/\allowbreak{}jobs/\allowbreak{}:\allowbreak{}slug/\allowbreak{}bookmark}.
    \item Backend verifies \texttt{FREELANCER} role, locates target job by \texttt{slug} (404 if absent), rejects self-bookmarking (\texttt{job.clientId === userId}), checks \texttt{job.status === 'OPEN'} and non-expired \texttt{expiryDate}, and checks existing \texttt{(userId, jobId)} bookmark.
    \item Inserts \texttt{JobBookmark} record in PostgreSQL and guards against race-condition unique violations (\texttt{P2002}).
\end{itemize}

\fnhead{Screen Layout (Mock-up Description)}
\screenfigure[0.95]{image/screenshot/sprint1/ptai/BookmarkJob.png}{Bookmark job confirmation toast on detail view}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Job detail page displaying top-right green confirmation toast ``Job saved to bookmarks''.
\end{itemize}

\fnhead{Function Details}

\sublabel{Input Data}
\begin{specdatatable}{Data Name}{Description}
\texttt{slug} & URL slug identifying the target open job.\\ \hline
\end{specdatatable}

\sublabel{Validation Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Caller must be authenticated with \texttt{FREELANCER} role; target job must exist, be \texttt{OPEN}, and not be expired.
\end{itemize}

\sublabel{Business Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]\brrulelist
    \item BR-BMK-04: Freelancer-Only and Self-Bookmark Guard: Only active \texttt{FREELANCER} accounts may bookmark jobs, and bookmarking one's own posted job is strictly prohibited.
    \item BR-BMK-05: Open and Non-Expired Target Check: Bookmarking requires the target job to exist as non-deleted under an active client, hold \texttt{OPEN} status, and have either no \texttt{expiryDate} or a future \texttt{expiryDate}.
    \item BR-BMK-06: Composite Uniqueness and Concurrency Protection: Composite uniqueness on \texttt{(userId, jobId)} permits at most one bookmark per freelancer per job; duplicate or concurrent duplicate requests are safely rejected without creating duplicate records.
\end{itemize}

\sublabel{Normal Flow}
\begin{enumerate}[leftmargin=1.8em,topsep=3pt,itemsep=2pt,parsep=0pt]
    \item Freelancer clicks the Bookmark icon on an open job.
    \item Frontend dispatches \texttt{POST /api/manage-jobs/jobs/:slug/bookmark}.
    \item Backend validates role, job existence/openness, non-self ownership, and uniqueness.
    \item Inserts \texttt{JobBookmark} record in PostgreSQL.
    \item UI fills bookmark indicator and renders ``Job saved to bookmarks'' toast.
\end{enumerate}

\sublabel{Abnormal Flow / Exception Handling}
\begin{flowtable}{Abnormal Case}{Handling \& System Response}
Missing or invalid authentication token & Global \texttt{AuthGuard} returns HTTP 401 Unauthorized; client redirects to \texttt{/login}.\\ \hline
Non-freelancer active role & Service throws \texttt{BookmarkJobOnlyForFreelancerException} (HTTP 403 Forbidden).\\ \hline
Job slug does not exist or deleted & Service throws \texttt{JobNotFoundException} (HTTP 404 Not Found).\\ \hline
Freelancer attempts to bookmark own job & Service throws \texttt{CannotBookmarkOwnJobException} (HTTP 400 Bad Request).\\ \hline
Target job is not \texttt{OPEN} or \texttt{expiryDate} passed & Service throws \texttt{CannotBookmarkUnavailableJobException} (HTTP 400 Bad Request).\\ \hline
Job already bookmarked or concurrent \texttt{P2002} & Throws \texttt{JobAlreadyBookmarkedException} (HTTP 400 Bad Request).\\ \hline
Database insert error & Service catches Prisma error and throws \texttt{FailedToBookmarkJobException} (HTTP 500).\\ \hline
\end{flowtable}

\sublabel{Output Data}
\begin{specdatatable}{Data Name}{Description}
HTTP Status & 201/200 OK confirming creation of \texttt{JobBookmark} record.\\ \hline
\end{specdatatable}

\sublabel{Post-Conditions}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Job bookmark is stored and immediately visible in Saved Jobs.
\end{itemize}
