% -----------------------------------------------------------------
% UC-07.01: View Freelancer Profile
% -----------------------------------------------------------------
\subsubsection{ View Freelancer Profile}

\fnhead{Function Trigger}
The actor selects View Freelancer Profile.\newline
\textbf{Navigation path:} Home $\rightarrow$ \texttt{/freelancer/profile} or \texttt{/profiles/[id].}

\fnhead{Function Description}

\sublabel{Actors / Roles}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item \textbf{Primary Actor:} Guest, Client, Freelancer.
    \item \textbf{Supporting Systems:} Frevia Web application (Next.js), Frevia BFF proxy (\texttt{/api/backend}), Frevia REST API (NestJS), Prisma ORM layer, and PostgreSQL relational database.
\end{itemize}

\sublabel{Purpose}
\noindent Retrieve approved Profile and FreelancerProfile fields in one response.

\sublabel{Interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Profile header with name and availability.
    \item Professional sections.
    \item Skills and Portfolio panels.
    \item Owner-only Edit and CV controls.
\end{itemize}

\sublabel{Data Processing Overview}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item The client web application issues an HTTP request to \texttt{GET /profiles/:id} via the Next.js BFF proxy.
    \item NestJS guards verify JWT authentication and validate role-based access permissions.
    \item The backend service queries or updates entities in PostgreSQL using Prisma ORM.
    \item The API returns the structured JSON payload and handles domain invariants consistently.
\end{itemize}

\fnhead{Screen Layout (Mock-up Description)}
\screenfigure[0.95]{image/screenshot/sprint1/nhuy/View Freelancer Profile.png}{View Freelancer Profile interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Profile header with name and availability.
    \item Professional sections.
    \item Skills and Portfolio panels.
    \item Owner-only Edit and CV controls.
\end{itemize}

\fnhead{Function Details}

\sublabel{Input Data}
\begin{specdatatable}{Data Name}{Description}
Identifier & Positive resource ID from the route; ownership is checked by the API.\\ \hline
\end{specdatatable}

\sublabel{Validation Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Omitted optional fields are preserved. An explicit null or empty list clears the corresponding value.
\end{itemize}

\sublabel{Business Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]\brrulelist
    \item BR-0701-01: Use Profile.id for profile routes; User.id identifies the owner and must not be substituted for the profile ID.
    \item BR-0701-02: Only the owner changes professional data. Public output excludes credentials and private CV locations.
    \item BR-0701-03: Freelancer and Client updates below 20 percent profile strength require a pending revision; other updates apply directly. Pending changes are not shown as approved public data.
\end{itemize}

\sublabel{Normal Flow}
\begin{enumerate}[leftmargin=1.8em,topsep=3pt,itemsep=2pt,parsep=0pt]
    \item The actor opens \texttt{/freelancer/profile}; \texttt{/profiles/[id]} and selects View Freelancer Profile.
    \item The application submits the selected identifier or list filters.
    \item The API checks authentication where required, authorization, resource existence and input validity.
    \item Retrieve approved Profile and FreelancerProfile fields in one response.
    \item The screen renders the response, or an explicit empty state for a valid empty collection.
\end{enumerate}

\sublabel{Abnormal Flow / Exception Handling}
\begin{flowtable}{Case}{Description}
A1 & Missing or expired authentication token. The system returns HTTP 401 Unauthorized and prompts re-authentication.\\ \hline
A2 & Insufficient permissions or role unauthorized. The request is rejected with HTTP 403 Forbidden.\\ \hline
A3 & Target resource not found or soft-deleted. The system returns HTTP 404 Not Found.\\ \hline
A4 & Validation error or malformed payload. The system returns HTTP 400 Bad Request with field validation details.\\ \hline
A5 & Database connectivity failure or unexpected internal exception. The transaction rolls back and returns HTTP 500.\\ \hline
\end{flowtable}

\sublabel{Output Data}
\begin{specdatatable}{Data Name}{Description}
Result & Retrieve approved Profile and FreelancerProfile fields in one response.\\ \hline
Failure & Structured error without sensitive data; no successful UI state when the operation failed.\\ \hline
\end{specdatatable}

\sublabel{Post-Conditions}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Successful view freelancer profile produces the documented response and survives a reload where data is persisted.
    \item An unauthorized actor cannot access or modify the resource.
    \item Invalid input and missing resources do not create a false success state or expose sensitive data.
\end{itemize}
