\Needspace{12\baselineskip}
\subsubsection{View Saved Searches List}

\fnhead{Function Trigger}
A freelancer opens the ``Saved Searches'' tab on Find Work or navigates to the saved searches page.\newline
\textbf{Navigation path:} Find Work $\rightarrow$ Saved Searches (\texttt{/freelancer/find-work/saved-searches}) or \texttt{/freelancer/saved-searches}.

\fnhead{Function Description}

\sublabel{Actors / Roles}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item \textbf{Primary Actor:} Freelancer.
    \item \textbf{Supporting Systems:} Frevia Web application (Next.js), Frevia REST API (\texttt{SavedSearchController}), Prisma ORM, and PostgreSQL database.
\end{itemize}

\sublabel{Purpose}
\noindent Allows authenticated users to inspect, filter by name, and manage their stored job search filter configurations, displaying saved criteria tags and one-click execution links.

\sublabel{Interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Saved Searches tab inside \texttt{/freelancer/find-work/saved-searches}.
    \item Search input on saved search management view for client-side/server-side filtering by \texttt{name}.
    \item Saved search cards showing title, criteria badges (keyword, category, budget), ``Search'' run button, detail link, and delete trash button.
\end{itemize}

\sublabel{Data Processing Overview}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Client dispatches \texttt{GET /api/saved-searches} with optional query params \texttt{search} and \texttt{order} (\texttt{asc} or \texttt{desc}, default \texttt{desc}).
    \item Backend \texttt{AuthGuard} identifies \texttt{userId} from the JWT token; \texttt{SavedSearchRepository.findAllByUserId()} queries strictly \texttt{where: \{ userId \}} and filters by case-insensitive substring on \texttt{name} if supplied.
    \item Returns array of \texttt{SavedSearch} records ordered by \texttt{createdAt}.
\end{itemize}

\fnhead{Screen Layout (Mock-up Description)}
\screenfigure[0.95]{image/screenshot/sprint2/ptai/ViewSavedSearchList.png}{Saved searches list interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Find Work banner with search bar and workspace tab showing total saved search count.
    \item Cards listing saved search label, summary criterion chips (Keyword, Budget, Category), blue ``Search'' execution button, and trash icon delete button.
    \item Right profile summary panel.
\end{itemize}

\fnhead{Function Details}

\sublabel{Input Data}
\begin{specdatatable}{Data Name}{Description}
JWT Token & Authenticated session token identifying \texttt{userId}.\\ \hline
\texttt{search} & Optional trimmed text string to match saved search \texttt{name}.\\ \hline
\texttt{order} & Optional sorting order: \texttt{asc} or \texttt{desc} (default \texttt{desc}).\\ \hline
\end{specdatatable}

\sublabel{Validation Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item \texttt{order}, when provided, must strictly be \texttt{asc} or \texttt{desc}.
    \item Caller must be authenticated with a valid JWT access token.
\end{itemize}

\sublabel{Business Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]\brrulelist
    \item BR-SS-01: Authenticated Owner Isolation: Saved search queries and management operations are strictly scoped to the authenticated user's own records so callers cannot view or alter another user's saved searches.
    \item BR-SS-02: Name Search and Chronological Ordering: Saved search listings support case-insensitive substring matching on the configuration name and ascending or descending chronological sorting (defaulting to newest first).
\end{itemize}

\sublabel{Normal Flow}
\begin{enumerate}[leftmargin=1.8em,topsep=3pt,itemsep=2pt,parsep=0pt]
    \item Freelancer opens the Saved Searches tab (\texttt{/freelancer/find-work/saved-searches}).
    \item Frontend requests \texttt{GET /api/saved-searches}.
    \item Backend validates authentication and queries records owned by \texttt{userId}.
    \item Returns HTTP 200 with array of saved search entities.
    \item UI renders cards with criteria chips, Search action, and delete button.
\end{enumerate}

\sublabel{Abnormal Flow / Exception Handling}
\begin{flowtable}{Abnormal Case}{Handling \& System Response}
Missing or invalid authentication token & Backend rejects with HTTP 401 Unauthorized. Frontend redirects to \texttt{/login}.\\ \hline
User lacks route permission & \texttt{PermissionGuard} rejects with HTTP 403 Forbidden; UI displays access denied message.\\ \hline
Invalid \texttt{order} query parameter & Zod DTO (\texttt{GetSavedSearchesQueryDto}) rejects values other than \texttt{asc}/\texttt{desc} with HTTP 400 Bad Request.\\ \hline
No saved searches exist yet & Query returns \texttt{[]}; UI displays empty state prompting the user to save a search from Find Work.\\ \hline
Database query failure or timeout & Service catches \texttt{PrismaClientKnownRequestError} and throws \texttt{FailedToLoadSavedSearchException} (HTTP 500).\\ \hline
\end{flowtable}

\sublabel{Output Data}
\begin{specdatatable}{Data Name}{Description}
Saved Search List & Array of \texttt{SavedSearch} objects containing integer \texttt{id}, \texttt{userId}, \texttt{name}, JSON \texttt{searchParams}, and \texttt{createdAt}.\\ \hline
\end{specdatatable}

\sublabel{Post-Conditions}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Freelancer inspects their saved search list and can open, run, or delete entries.
\end{itemize}

\Needspace{12\baselineskip}
\subsubsection{View Saved Search Details}

\fnhead{Function Trigger}
A freelancer navigates to the detail page of a specific saved search.\newline
\textbf{Navigation path:} Saved Searches $\rightarrow$ View Details (\texttt{/freelancer/saved-searches/:id}).

\fnhead{Function Description}

\sublabel{Actors / Roles}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item \textbf{Primary Actor:} Freelancer.
    \item \textbf{Supporting Systems:} Frevia Web application (Next.js), Frevia REST API (\texttt{SavedSearchController}), Prisma ORM, and PostgreSQL database.
\end{itemize}

\sublabel{Purpose}
\noindent Displays all structured filter attributes stored inside a saved search configuration and provides a direct ``View matching jobs'' action to replay those filters against live job listings.

\sublabel{Interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Saved search detail page at \texttt{/freelancer/saved-searches/:id}.
    \item Top back breadcrumb (\texttt{$\leftarrow$ Saved searches}) and saved search title header.
    \item Filter breakdown tiles (Search keyword, Sort, Budget, Category, Skills, Hiring Type) and primary ``View matching jobs'' CTA button.
\end{itemize}

\sublabel{Data Processing Overview}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Client dispatches \texttt{GET /api/saved-searches/:id} with positive integer path ID parsed by \texttt{ParseIntPipe}.
    \item Backend invokes \texttt{SavedSearchRepository.findByIdAndUserId(id, userId)}.
    \item Returns \texttt{SavedSearch} entity if found and owned; throws \texttt{SavedSearchNotFoundException} (HTTP 404) otherwise.
\end{itemize}

\fnhead{Screen Layout (Mock-up Description)}
\screenfigure[0.95]{image/screenshot/sprint2/ptai/ViewSavedSearchDetail.png}{Saved search detail interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Header icon, ``Saved search'' label, saved query title (e.g. ``Senior Backend \& AI Roles (\$1500+)''), and back link.
    \item Structured ``Search filters'' summary panel listing active filter values, record ID badge, and blue ``View matching jobs'' action button.
\end{itemize}

\fnhead{Function Details}

\sublabel{Input Data}
\begin{specdatatable}{Data Name}{Description}
\texttt{id} & Positive integer ID of the saved search record.\\ \hline
\end{specdatatable}

\sublabel{Validation Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Path parameter \texttt{id} must parse as a valid integer via \texttt{ParseIntPipe}.
    \item Record must exist and belong to the authenticated \texttt{userId}.
\end{itemize}

\sublabel{Business Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]\brrulelist
    \item BR-SS-03: Detail Ownership and Cross-User Isolation: Retrieving a saved search requires matching both the record identifier and the owning user; non-existent or third-party saved searches are treated as not found.
    \item BR-SS-04: Parameter Replay Compatibility: Stored search filter parameters preserve structured query criteria for direct replay onto job search while still honoring all public open-job visibility rules.
\end{itemize}

\sublabel{Normal Flow}
\begin{enumerate}[leftmargin=1.8em,topsep=3pt,itemsep=2pt,parsep=0pt]
    \item Freelancer opens \texttt{/freelancer/saved-searches/:id}.
    \item Backend validates authentication, parses integer \texttt{id}, and verifies ownership.
    \item Returns HTTP 200 with \texttt{SavedSearch} object.
    \item UI renders breakdown cards for every stored filter criterion and the ``View matching jobs'' button.
\end{enumerate}

\sublabel{Abnormal Flow / Exception Handling}
\begin{flowtable}{Abnormal Case}{Handling \& System Response}
Missing or invalid authentication token & Global \texttt{AuthGuard} returns HTTP 401 Unauthorized; client redirects to \texttt{/login}.\\ \hline
Non-numeric \texttt{id} path parameter & \texttt{ParseIntPipe} rejects invalid format with HTTP 400 Bad Request.\\ \hline
Record does not exist or belongs to another user & \texttt{SavedSearchService.getSavedSearchDetail()} throws \texttt{SavedSearchNotFoundException} (HTTP 404). Next.js renders not-found view.\\ \hline
Database query failure & Service catches \texttt{PrismaClientKnownRequestError} and throws \texttt{FailedToLoadSavedSearchException} (HTTP 500).\\ \hline
\end{flowtable}

\sublabel{Output Data}
\begin{specdatatable}{Data Name}{Description}
Saved Search Object & Entity containing integer \texttt{id}, \texttt{userId}, \texttt{name}, JSON \texttt{searchParams}, and \texttt{createdAt}.\\ \hline
\end{specdatatable}

\sublabel{Post-Conditions}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Complete saved filter criteria are shown ready for execution.
\end{itemize}

\Needspace{12\baselineskip}
\subsubsection{Create Saved Search}

\fnhead{Function Trigger}
A freelancer applies search or filter criteria on Search Jobs and clicks ``Save search''.\newline
\textbf{Navigation path:} Search Jobs (\texttt{/freelancer/search/jobs}) $\rightarrow$ Apply Filters $\rightarrow$ Click ``Save search''.

\fnhead{Function Description}

\sublabel{Actors / Roles}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item \textbf{Primary Actor:} Freelancer.
    \item \textbf{Supporting Systems:} Frevia Web application (Next.js), Frevia REST API (\texttt{SavedSearchController}), Prisma ORM, and PostgreSQL database.
\end{itemize}

\sublabel{Purpose}
\noindent Persists the current search keywords and filter settings under a concise user-provided name for quick reuse, while enforcing a 30-item quota per user.

\sublabel{Interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item ``Save search'' trigger button in the Search Jobs toolbar.
    \item ``Save search'' modal dialog containing a ``Saved search name'' text input, filter preview summary, Cancel button, and Save search CTA.
\end{itemize}

\sublabel{Data Processing Overview}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Frontend validates the trimmed label and posts \texttt{POST /api/saved-searches} with \texttt{\{ name, searchParams \}}.
    \item Zod DTO (\texttt{CreateSavedSearchBodySchema.strict()}) validates \texttt{name} (trimmed, 1--100 chars) and \texttt{searchParams} record.
    \item Service queries \texttt{countByUserId(userId)}; rejects with HTTP 400 if count $\ge 30$ (\texttt{MAX\_SAVED\_SEARCHES\_PER\_USER}).
    \item Repository inserts new \texttt{SavedSearch} row and returns HTTP 201/200 with created entity.
\end{itemize}

\fnhead{Screen Layout (Mock-up Description)}
\screenfigure[0.95]{image/screenshot/sprint2/ptai/CreateSavedSearch.png}{Save search modal dialog interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Modal title ``Save search'' and explanatory subtitle.
    \item Focus-ring input field for naming the saved search.
    \item ``Search filters'' preview panel displaying active query parameters and ``Save search'' confirmation button.
\end{itemize}

\fnhead{Function Details}

\sublabel{Input Data}
\begin{specdatatable}{Data Name}{Description}
\texttt{name} & Required trimmed string (1--100 characters) naming the filter preset.\\ \hline
\texttt{searchParams} & Key-value JSON map representing active search/filter query criteria.\\ \hline
\end{specdatatable}

\sublabel{Validation Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item \texttt{name} must be a non-empty trimmed string between 1 and 100 characters.
    \item Unknown extra keys in body are rejected via \texttt{CreateSavedSearchBodySchema.strict()}.
    \item Current saved search count must be strictly less than 30 (\texttt{MAX\_SAVED\_SEARCHES\_PER\_USER = 30}).
\end{itemize}

\sublabel{Business Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]\brrulelist
    \item BR-SS-05: Strict Label and Schema Validation: Creating a saved search requires a trimmed label between 1 and 100 characters and valid structured search parameters while rejecting unrecognized fields.
    \item BR-SS-06: Maximum Saved Search Quota: Each user may store at most \texttt{MAX\_SAVED\_SEARCHES\_PER\_USER} (30) saved searches; attempts to create additional configurations beyond 30 are rejected.
\end{itemize}

\sublabel{Normal Flow}
\begin{enumerate}[leftmargin=1.8em,topsep=3pt,itemsep=2pt,parsep=0pt]
    \item Freelancer sets search filters on \texttt{/freelancer/search/jobs} and clicks ``Save search''.
    \item Modal opens; freelancer enters a descriptive label and submits.
    \item Backend validates schema and verifies user has fewer than 30 saved searches.
    \item Repository persists \texttt{SavedSearch} in PostgreSQL and returns the created record.
    \item Frontend closes modal and displays success feedback.
\end{enumerate}

\sublabel{Abnormal Flow / Exception Handling}
\begin{flowtable}{Abnormal Case}{Handling \& System Response}
Missing or invalid authentication token & Backend returns HTTP 401 Unauthorized; client redirects to \texttt{/login}.\\ \hline
Empty or whitespace-only name & Frontend blocks submission; backend Zod validator returns HTTP 400 Bad Request.\\ \hline
Name exceeds 100 characters or unknown fields sent & Strict DTO validation fails with HTTP 400 Bad Request showing field errors.\\ \hline
Saved search quota reached ($\ge 30$) & Service throws \texttt{SavedSearchLimitExceededException} (HTTP 400). UI shows limit-exceeded alert.\\ \hline
Database write failure or timeout & Service catches \texttt{PrismaClientKnownRequestError} and throws \texttt{FailedToCreateSavedSearchException} (HTTP 500).\\ \hline
\end{flowtable}

\sublabel{Output Data}
\begin{specdatatable}{Data Name}{Description}
Created Saved Search & Newly inserted \texttt{SavedSearch} record with \texttt{id}, \texttt{userId}, \texttt{name}, \texttt{searchParams}, and \texttt{createdAt}.\\ \hline
\end{specdatatable}

\sublabel{Post-Conditions}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item New saved search is persisted in PostgreSQL and appears in Saved Searches.
\end{itemize}

\Needspace{12\baselineskip}
\subsubsection{Delete Saved Search}

\fnhead{Function Trigger}
A freelancer clicks the delete (trash) icon button on a saved search entry.\newline
\textbf{Navigation path:} Saved Searches (\texttt{/freelancer/find-work/saved-searches}) $\rightarrow$ Click Delete button.

\fnhead{Function Description}

\sublabel{Actors / Roles}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item \textbf{Primary Actor:} Freelancer.
    \item \textbf{Supporting Systems:} Frevia Web application (Next.js), Frevia REST API (\texttt{SavedSearchController}), Prisma ORM, and PostgreSQL database.
\end{itemize}

\sublabel{Purpose}
\noindent Permanently removes an unwanted saved search configuration owned by the freelancer and immediately frees a slot within the 30-item quota.

\sublabel{Interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Delete (trash) icon button on each saved search row/card.
    \item Instant removal update on list and green confirmation toast notification.
\end{itemize}

\sublabel{Data Processing Overview}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Client dispatches \texttt{DELETE /api/saved-searches/:id} with positive integer ID.
    \item Backend invokes \texttt{SavedSearchRepository.delete(id, userId)} enforcing both \texttt{id} and \texttt{userId}.
    \item If zero rows deleted, throws \texttt{SavedSearchNotFoundException} (HTTP 404).
\end{itemize}

\fnhead{Screen Layout (Mock-up Description)}
\screenfigure[0.95]{image/screenshot/sprint2/ptai/DeleteSavedSearch.png}{Delete saved search result and confirmation toast}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Updated Saved Searches list immediately removing the deleted entry and decrementing the tab badge count.
    \item Top-right green confirmation toast displaying ``Saved search deleted''.
\end{itemize}

\fnhead{Function Details}

\sublabel{Input Data}
\begin{specdatatable}{Data Name}{Description}
\texttt{id} & Positive integer ID of the saved search to delete.\\ \hline
\end{specdatatable}

\sublabel{Validation Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Path parameter \texttt{id} must be a valid integer (\texttt{ParseIntPipe}).
    \item Target record must exist and belong to authenticated caller \texttt{userId}.
\end{itemize}

\sublabel{Business Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]\brrulelist
    \item BR-SS-07: Hard Delete with Owner Verification: Deleting a saved search verifies both record ownership and existence before permanently hard-deleting the record; missing or non-owned records cannot be deleted.
    \item BR-SS-08: Independent Search Persistence: Removing a saved search deletes only that saved preset and never alters job postings, bookmarks, or job alert subscriptions.
\end{itemize}

\sublabel{Normal Flow}
\begin{enumerate}[leftmargin=1.8em,topsep=3pt,itemsep=2pt,parsep=0pt]
    \item Freelancer clicks the delete icon on a saved search card.
    \item Frontend dispatches \texttt{DELETE /api/saved-searches/:id}.
    \item Backend deletes the owned row in PostgreSQL.
    \item UI removes the item from state and shows ``Saved search deleted'' toast.
\end{enumerate}

\sublabel{Abnormal Flow / Exception Handling}
\begin{flowtable}{Abnormal Case}{Handling \& System Response}
Missing or invalid authentication token & Global \texttt{AuthGuard} returns HTTP 401 Unauthorized; client redirects to \texttt{/login}.\\ \hline
Non-numeric saved search ID & \texttt{ParseIntPipe} rejects malformed ID with HTTP 400 Bad Request.\\ \hline
Record already deleted or owned by another user & Service throws \texttt{SavedSearchNotFoundException} (HTTP 404). Frontend shows error toast and rolls back optimistic state.\\ \hline
Database connection or transaction error & Prisma error triggers HTTP 500 Internal Server Error; UI restores list item and shows failure toast.\\ \hline
\end{flowtable}

\sublabel{Output Data}
\begin{specdatatable}{Data Name}{Description}
HTTP Status & 200 OK confirming hard deletion from database.\\ \hline
\end{specdatatable}

\sublabel{Post-Conditions}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Saved search record is permanently deleted and user's 30-item quota slot is freed.
\end{itemize}

\fnhead{Function Trigger}
A freelancer opens ``Job Alerts'' from the navigation or profile settings.\newline

\fnhead{Function Description}

\sublabel{Actors / Roles}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]

\sublabel{Purpose}

\fnhead{Function Details}

\sublabel{Validation Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]

\end{itemize}

\sublabel{Business Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]\brrulelist
    \item BR-JA-01: Freelancer Role Restriction: Only authenticated users holding the \texttt{FREELANCER} role may list, inspect, create, update, delete, or preview job alerts; other roles are denied access.
    \item BR-JA-02: User Ownership Isolation and Chronological Ordering: Job alert listings return exclusively alerts owned by the authenticated freelancer, ordered chronologically newest first.
\end{itemize}

\sublabel{Normal Flow}
\begin{enumerate}[leftmargin=1.8em,topsep=3pt,itemsep=2pt,parsep=0pt]
    \item Freelancer navigates to \texttt{/freelancer/job-alerts}.
    \item Frontend dispatches \texttt{GET /api/job-alerts}.
    \item Backend validates authentication and \texttt{FREELANCER} role.
    \item Repository queries alerts belonging to \texttt{userId}.
    \item UI displays alert cards, quota pill, and action buttons.
\end{enumerate}

\sublabel{Abnormal Flow / Exception Handling}
\begin{flowtable}{Abnormal Case}{Handling \& System Response}
Missing or invalid authentication token & Global \texttt{AuthGuard} returns HTTP 401 Unauthorized; client redirects to \texttt{/login}.\\ \hline
Non-freelancer active role (e.g. Client) & Service throws \texttt{JobAlertOnlyForFreelancerException} (HTTP 403 Forbidden).\\ \hline
No alerts configured yet & Query returns empty array \texttt{[]}; UI renders onboarding empty state with ``Create Alert'' CTA.\\ \hline
Search input yields zero matches & Frontend renders ``No results'' state with a ``Reset filter'' button.\\ \hline
Database connection or query error & Service catches \texttt{PrismaClientKnownRequestError} and throws \texttt{FailedToLoadJobAlertException} (HTTP 500).\\ \hline
\end{flowtable}

\begin{specdatatable}{Data Name}{Description}
\texttt{alerts} & Array of \texttt{JobAlert} entities (\texttt{id}, \texttt{userId}, \texttt{name}, \texttt{keywords}, \texttt{skills}, \texttt{frequency}, \texttt{channels}, \texttt{isActive}, \texttt{lastSentAt}, \texttt{createdAt}).\\ \hline

\Needspace{12\baselineskip}
\subsubsection{Create Job Alert}

\fnhead{Function Trigger}
A freelancer clicks ``+ Create Job Alert'' on the Job Alerts page.\newline
\textbf{Navigation path:} Job Alerts (\texttt{/freelancer/job-alerts}) $\rightarrow$ Click ``+ Create Job Alert''.

\fnhead{Function Description}

\sublabel{Actors / Roles}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item \textbf{Primary Actor:} Freelancer.
    \item \textbf{Supporting Systems:} Frevia Web application (Next.js), Frevia REST API (\texttt{JobAlertController}), Prisma ORM, and PostgreSQL database.
\end{itemize}

\sublabel{Purpose}
\noindent Registers a new job alert subscription specifying alert name, keywords and/or skill tags, notification frequency, and delivery channels, subject to a 10-alert quota and anti-duplication validation.

\sublabel{Interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item ``Create Job Alert'' modal dialog.
    \item Inputs for optional ``Alert name'', ``Keywords'', ``Required skills'' pill builder, ``Notification frequency'' buttons (\texttt{Instant}, \texttt{Daily digest}, \texttt{Weekly digest}), and ``Notification channels'' checkboxes (\texttt{In-app notification}, \texttt{Email}).
\end{itemize}

\sublabel{Data Processing Overview}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Client submits \texttt{POST /api/job-alerts} validated by \texttt{CreateJobAlertBodySchema.strict()}.
    \item Backend enforces \texttt{FREELANCER} role, then verifies \texttt{countByUserId(userId) < 10}.
    \item Normalizes \texttt{keywords} (trimmed lowercase) and \texttt{skills} (trimmed, lowercased, sorted) to reject duplicate alert criteria for the same user.
    \item Inserts \texttt{JobAlert} with \texttt{isActive = true} and returns HTTP 201/200.
\end{itemize}

\fnhead{Screen Layout (Mock-up Description)}
\screenfigure[0.95]{image/screenshot/sprint2/ptai/CreateJobAlert.png}{Create job alert modal dialog interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Centered dialog displaying title ``Create Job Alert'' and helper text.
    \item Form fields for alert name, keywords, dynamic skill tag addition, segmented frequency selector, channel checkboxes, Cancel, and blue ``Save alert'' button.
\end{itemize}

\fnhead{Function Details}

\sublabel{Input Data}
\begin{specdatatable}{Data Name}{Description}
\texttt{name} & Optional trimmed string (1--100 chars).\\ \hline
\texttt{keywords} & Optional trimmed string ($\le 200$ chars; nullable).\\ \hline
\texttt{skills} & Array of skill strings (1--100 chars each, max 20 skills, default \texttt{[]}).\\ \hline
\texttt{frequency} & Enum: \texttt{INSTANT}, \texttt{DAILY} (default), or \texttt{WEEKLY}.\\ \hline
\texttt{channels} & Non-empty array of \texttt{IN\_APP} and/or \texttt{EMAIL} (default \texttt{['IN\_APP']}).\\ \hline
\end{specdatatable}

\sublabel{Validation Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Either \texttt{keywords} (non-blank, $\le 200$ chars) or at least 1 skill ($\le 20$ skills) must be provided.
    \item At least 1 valid delivery channel (\texttt{IN\_APP} or \texttt{EMAIL}) is required.
    \item Current alert count per freelancer must be strictly $< 10$; strict schema rejects unrecognized keys.
\end{itemize}

\sublabel{Business Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]\brrulelist
    \item BR-JA-04: Criteria, Frequency, and Channel Validation: Job alerts require at least one non-empty keyword (up to 200 characters) or at least one skill (up to 20 items), a supported frequency (\texttt{INSTANT}, \texttt{DAILY}, or \texttt{WEEKLY}), and at least one delivery channel (\texttt{IN\_APP} or \texttt{EMAIL}).
    \item BR-JA-05: Maximum Alert Quota: Each freelancer may maintain at most \texttt{MAX\_JOB\_ALERTS\_PER\_USER} (10) job alerts; exceeding 10 alerts is rejected.
    \item BR-JA-06: Case-Insensitive Duplicate Criteria Prevention: Alert keywords (trimmed lowercase) and skill sets (sorted lowercase) are normalized per user to block duplicate alert configurations upon creation or update.
\end{itemize}

\sublabel{Normal Flow}
\begin{enumerate}[leftmargin=1.8em,topsep=3pt,itemsep=2pt,parsep=0pt]
    \item Freelancer clicks ``+ Create Job Alert'', fills criteria, selects frequency and channels, and clicks ``Save alert''.
    \item Frontend checks criteria/channel presence and submits \texttt{POST /api/job-alerts}.
    \item Backend validates role, Zod schema, $< 10$ quota, and uniqueness of normalized criteria.
    \item Persists new active \texttt{JobAlert} and returns it.
    \item UI prepends the card, closes the modal, increments quota pill, and shows success toast.
\end{enumerate}

\sublabel{Abnormal Flow / Exception Handling}
\begin{flowtable}{Abnormal Case}{Handling \& System Response}
Missing or invalid authentication token & Global \texttt{AuthGuard} returns HTTP 401 Unauthorized; client redirects to \texttt{/login}.\\ \hline
Non-freelancer active role & Service throws \texttt{JobAlertOnlyForFreelancerException} (HTTP 403 Forbidden).\\ \hline
Both \texttt{keywords} and \texttt{skills} empty, or 0 channels selected & Frontend toast blocks empty criteria; backend Zod refinement throws HTTP 400 Bad Request (\texttt{KEYWORD\_OR\_SKILL\_REQUIRED}).\\ \hline
Maximum quota of 10 alerts reached & Service throws \texttt{JobAlertLimitExceededException} (HTTP 400 \texttt{LIMIT\_EXCEEDED}).\\ \hline
Identical normalized criteria already exists for user & Service detects duplicate match and throws \texttt{JobAlertDuplicateCriteriaException} (HTTP 400 \texttt{DUPLICATE\_CRITERIA}).\\ \hline
Database insert failure & Service catches \texttt{PrismaClientKnownRequestError} and throws \texttt{FailedToCreateJobAlertException} (HTTP 500).\\ \hline
\end{flowtable}

\sublabel{Output Data}
\begin{specdatatable}{Data Name}{Description}
Created Alert & Persisted \texttt{JobAlert} entity with \texttt{isActive = true} and assigned integer \texttt{id}.\\ \hline
\end{specdatatable}

\sublabel{Post-Conditions}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Alert is active, persisted in PostgreSQL, and reflected in quota counter.
\end{itemize}
