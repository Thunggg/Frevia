% -----------------------------------------------------------------
\subsubsection{Function: View Saved Search List}

\fnhead{Function Trigger}
A freelancer clicks ``Saved Searches'' in the workspace navigation.\newline
\textbf{Navigation path:} Freelancer Dashboard $\rightarrow$ Saved Searches (\texttt{/freelancer/saved-searches}).

\fnhead{Function Description}

\sublabel{Actors / Roles}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item \textbf{Primary Actor:} Freelancer.
    \item \textbf{Supporting Systems:} Frevia Web application, Frevia REST API, and PostgreSQL database.
\end{itemize}

\sublabel{Purpose}
\noindent Allows freelancers to inspect, search, and manage their saved job search queries, view summarized filter parameters, and execute stored searches with one click.

\sublabel{Interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Saved searches management view at \texttt{/freelancer/saved-searches}.
    \item Search input to filter saved queries by name.
    \item Cards or table rows displaying search title, summary of criteria (keywords, skills, budget), and direct ``Run Search'' action button.
\end{itemize}

\sublabel{Data Processing Overview}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Client dispatches \texttt{GET /api/freelancer/saved-searches}.
    \item Backend filters \texttt{SavedSearch} records by authenticated \texttt{userId}.
    \item Sorts by \texttt{createdAt DESC} with pagination support.
    \item Returns list of saved search configurations.
\end{itemize}

\fnhead{Screen Layout (Mock-up Description)}
\screenfigure[0.95]{image/screenshot/sprint2/ptai/ViewSavedSearchList.png}{Saved searches list interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Top search bar to locate saved search filters by name.
    \item Cards summarizing saved filters, monitored keywords, and target categories.
    \item Action buttons on each item to run, edit, or delete the search configuration.
\end{itemize}

\fnhead{Function Details}

\sublabel{Input Data}
\begin{specdatatable}{Data Name}{Description}
JWT Token & Authenticated freelancer session token.\\ \hline
\texttt{search} & Optional name search query string.\\ \hline
\texttt{page} / \texttt{limit} & Pagination control numbers.\\ \hline
\end{specdatatable}

\sublabel{Validation Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Caller must have \texttt{FREELANCER} role.
\end{itemize}

\sublabel{Business Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]\brrulelist
    \item BR-SS-01: Only authenticated freelancers may view their own saved searches.
    \item BR-SS-02: Server strictly isolates queries using \texttt{where: { userId = currentUserId }}.
    \item BR-SS-03: Default order is newest first (\texttt{createdAt: 'desc'}).
\end{itemize}

\sublabel{Normal Flow}
\begin{enumerate}[leftmargin=1.8em,topsep=3pt,itemsep=2pt,parsep=0pt]
    \item Freelancer opens Saved Searches view.
    \item Frontend requests \texttt{GET /api/freelancer/saved-searches}.
    \item Backend fetches saved searches owned by user.
    \item UI renders list with criteria badges and ``Run Search'' buttons.
\end{enumerate}

\sublabel{Abnormal Flow / Exception Handling}
\begin{flowtable}{Case}{Description}
A1 & No saved searches exist. UI displays empty state with a ``Save Your First Search'' prompt.\\ \hline
\end{flowtable}

\sublabel{Output Data}
\begin{specdatatable}{Data Name}{Description}
\texttt{searches} & Array of objects (\texttt{id}, \texttt{name}, \texttt{query}, \texttt{categoryId}, \texttt{skills}, \texttt{budgetMin}, \texttt{createdAt}).\\ \hline
\end{specdatatable}

\sublabel{Post-Conditions}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Freelancer views list of saved search criteria.
\end{itemize}

\subsubsection{Function: Create Saved Search}

\fnhead{Function Trigger}
A freelancer applies filters on the jobs page and clicks ``Save Search''.\newline
\textbf{Navigation path:} Browse Jobs (\texttt{/jobs}) $\rightarrow$ Set Filters $\rightarrow$ Click ``Save Search''.

\fnhead{Function Description}

\sublabel{Actors / Roles}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item \textbf{Primary Actor:} Freelancer.
    \item \textbf{Supporting Systems:} Frevia Web application, Frevia REST API, and PostgreSQL database.
\end{itemize}

\sublabel{Purpose}
\noindent Saves the currently applied search parameters and keyword filters under a user-defined name for future access.

\sublabel{Interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item ``Save Search'' button on job catalog filter bar.
    \item Modal dialog prompting for a recognizable search name and optional alert preferences.
\end{itemize}

\sublabel{Data Processing Overview}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Client submits \texttt{POST /api/freelancer/saved-searches}.
    \item Backend validates that name is 1--100 characters and filters are well-formed.
    \item Persists search configuration associated with the user ID.
\end{itemize}

\fnhead{Screen Layout (Mock-up Description)}
\screenfigure[0.95]{image/screenshot/sprint2/ptai/CreateSavedSearch.png}{Save search modal interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Input field for naming the saved search.
    \item Summary chips reflecting current active filters (keywords, category, budget).
    \item Primary ``Save Search'' submission button.
\end{itemize}

\fnhead{Function Details}

\sublabel{Input Data}
\begin{specdatatable}{Data Name}{Description}
\texttt{name} & String (1--100 chars), descriptive label.\\ \hline
\texttt{criteria} & JSON object containing active query filters (\texttt{keywords}, \texttt{categoryId}, \texttt{skills}, \texttt{budgetMin}, \texttt{budgetMax}).\\ \hline
\end{specdatatable}

\sublabel{Validation Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Search name must not be blank and $\le 100$ characters.
    \item At least one search parameter must be present.
\end{itemize}

\sublabel{Business Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]\brrulelist
    \item BR-SS-06: Name is required and trimmed of whitespace.
    \item BR-SS-07: Prevents duplicate identical configurations for the same freelancer.
\end{itemize}

\sublabel{Normal Flow}
\begin{enumerate}[leftmargin=1.8em,topsep=3pt,itemsep=2pt,parsep=0pt]
    \item Freelancer clicks ``Save Search'' on browse jobs view.
    \item Modal opens; freelancer inputs name.
    \item Submits payload to backend.
    \item Backend commits new \texttt{SavedSearch} entity.
    \item Toast notification confirms creation.
\end{enumerate}

\sublabel{Abnormal Flow / Exception Handling}
\begin{flowtable}{Case}{Description}
A1 & Name is empty. Client validation displays error ``Search name is required.''\\ \hline
\end{flowtable}

\sublabel{Output Data}
\begin{specdatatable}{Data Name}{Description}
Created Entity & Saved search record with assigned UUID and timestamp.\\ \hline
\end{specdatatable}

\sublabel{Post-Conditions}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Search configuration is permanently saved.
\end{itemize}

\subsubsection{Function: Create Job Alert}

\fnhead{Function Trigger}
A freelancer clicks ``Create Job Alert'' on the alerts page.\newline
\textbf{Navigation path:} Job Alerts $\rightarrow$ Click ``Create Alert'' (\texttt{/freelancer/job-alerts/new}).

\fnhead{Function Description}

\sublabel{Actors / Roles}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item \textbf{Primary Actor:} Freelancer.
    \item \textbf{Supporting Systems:} Frevia Web application, Frevia REST API, and PostgreSQL database.
\end{itemize}

\sublabel{Purpose}
\noindent Configures automated background delivery of matching job postings via In-App and Email notifications.

\sublabel{Interface}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Alert setup modal or form.
    \item Inputs for name, keywords, skill selection, frequency selector (INSTANT, DAILY, WEEKLY), and channel checkboxes.
\end{itemize}

\sublabel{Data Processing Overview}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Client submits payload to \texttt{POST /api/freelancer/job-alerts}.
    \item Backend checks active alert count; rejects if user already has 10 alerts.
    \item Validates criteria and checks for duplicate active rules.
    \item Persists \texttt{JobAlert} record in PostgreSQL.
\end{itemize}

\fnhead{Screen Layout (Mock-up Description)}
\screenfigure[0.95]{image/screenshot/sprint2/ptai/CreateJobAlert.png}{Create job alert modal}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Quota indicator (e.g. 3 of 10 used).
    \item Inputs for alert name and target keywords.
    \item Radio group for notification frequency and channel checkboxes.
    \item Primary ``Activate Alert'' button.
\end{itemize}

\fnhead{Function Details}

\sublabel{Input Data}
\begin{specdatatable}{Data Name}{Description}
\texttt{name} & String (1--100 chars), alert title.\\ \hline
\texttt{keywords} & Optional keyword string to match.\\ \hline
\texttt{skills} & Array of skill UUIDs (up to 20 skills).\\ \hline
\texttt{frequency} & Enum: \texttt{INSTANT}, \texttt{DAILY}, \texttt{WEEKLY}.\\ \hline
\texttt{channels} & Array of enums: \texttt{IN\_APP}, \texttt{EMAIL}.\\ \hline
\end{specdatatable}

\sublabel{Validation Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Must specify at least one keyword or skill.
    \item At least one notification channel must be selected.
    \item User alert count must be $< 10$.
\end{itemize}

\sublabel{Business Rules}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]\brrulelist
    \item BR-JA01: Rejects creation when user reaches 10 alerts limit.
    \item BR-JA02: Prevents identical criteria from being registered twice.
    \item BR-JA06: Instant notifications are queued immediately upon job publishing.
\end{itemize}

\sublabel{Normal Flow}
\begin{enumerate}[leftmargin=1.8em,topsep=3pt,itemsep=2pt,parsep=0pt]
    \item Freelancer enters alert parameters.
    \item Clicks ``Activate Alert''.
    \item Backend validates quota and fields, then inserts record.
    \item UI returns to list with new alert active.
\end{enumerate}

\sublabel{Abnormal Flow / Exception Handling}
\begin{flowtable}{Case}{Description}
A1 & Quota reached (10 alerts). Backend returns HTTP 400 Bad Request with message ``Maximum 10 alerts limit reached.''\\ \hline
A2 & Duplicate criteria detected. Backend returns HTTP 400 with duplicate notice.\\ \hline
\end{flowtable}

\sublabel{Output Data}
\begin{specdatatable}{Data Name}{Description}
Created Alert & New \texttt{JobAlert} entity with status \texttt{isActive = true}.\\ \hline
\end{specdatatable}

\sublabel{Post-Conditions}
\begin{itemize}[leftmargin=1.6em,topsep=3pt,itemsep=2pt,parsep=0pt,label=$\bullet$]
    \item Alert is registered and background worker begins match dispatch.
\end{itemize}
