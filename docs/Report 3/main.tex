%====================================================
% FUNCTION DESCRIPTION FORM
%====================================================
% The template describes every function of section 3 with the same four
% fields (function trigger, function description, screen layout, function
% details). The fields are set as small tinted labels, the screen layout is
% framed and always kept together with its label, and the sub-headings inside
% the details use the brown of the template table headers.

\newcommand{\functionchip}[1]{%
  {\setlength{\fboxsep}{3.5pt}%
   \colorbox{headerbg}{\textcolor{fptHeaderText}{\small\bfseries #1}}}}

% A field label followed by its text; the label is moved to the next page when
% fewer than five lines are left, so it is never stranded alone at the bottom
% of a page.
\newcommand{\functionfield}[1]{%
  \par\addvspace{10pt plus 2pt}%
  \dimen0=\pagegoal \advance\dimen0 by -\pagetotal
  \ifdim\dimen0<5\baselineskip \newpage\fi
  \noindent\functionchip{#1}%
  \par\addvspace{2pt}\penalty10000\noindent\ignorespaces}

% Screen layout: the label belongs to the float, so it can never be stranded
% at the bottom of a page while the mock-up moves to the next one.
\newlength{\ringwidth}
\setlength{\ringwidth}{\fboxrule}

\newcommand{\screenshotframe}[2][0.88]{%
  \setlength{\fboxrule}{0.4pt}\setlength{\fboxsep}{2pt}%
  \fcolorbox{black!20}{white}{\includegraphics[width=#1\linewidth]{#2}}%
  \setlength{\fboxrule}{\ringwidth}}

\newcommand{\screenfield}[3][0.88]{%
  \begin{figure}[H]
    \centering
    \makebox[\linewidth][l]{\functionchip{Screen layout}}\par\vspace{5pt}%
    \screenshotframe[#1]{#2}%
    \caption{#3}
  \end{figure}}

\newcommand{\screenfieldpair}[3]{%
  \begin{figure}[H]
    \centering
    \makebox[\linewidth][l]{\functionchip{Screen layout}}\par\vspace{5pt}%
    \setlength{\fboxrule}{0.4pt}\setlength{\fboxsep}{2pt}%
    \fcolorbox{black!20}{white}{\includegraphics[width=0.455\linewidth]{#1}}\hfill
    \fcolorbox{black!20}{white}{\includegraphics[width=0.455\linewidth]{#2}}%
    \setlength{\fboxrule}{\ringwidth}%
    \caption{#3}
  \end{figure}}

% Sub-headings inside one function description (Actors / Roles, Purpose,
% Input Data, Validation Rules, ..., Post-Conditions) are plain black bold
% text, like every other heading of the report.
\newcommand{\detailhead}[1]{\textbf{#1}}

% Screen layout used by the detailed form: the numbered field heading already
% carries the label, so these variants only frame the mock-up.
\newcommand{\screenfigure}[3][0.88]{%
  \begin{figure}[H]
    \centering
    \screenshotframe[#1]{#2}%
    \caption{#3}
  \end{figure}}

\newcommand{\screenfigurepair}[3]{%
  \begin{figure}[H]
    \centering
    \setlength{\fboxrule}{0.4pt}\setlength{\fboxsep}{2pt}%
    \fcolorbox{black!20}{white}{\includegraphics[width=0.455\linewidth]{#1}}\hfill
    \fcolorbox{black!20}{white}{\includegraphics[width=0.455\linewidth]{#2}}%
    \setlength{\fboxrule}{\ringwidth}%
    \caption{#3}
  \end{figure}}

%====================================================
% DETAILED FUNCTION DESCRIPTION
%====================================================
% The detailed form describes each function through four numbered fields
% (3.2.1.1 Function trigger, 3.2.1.2 Function description, 3.2.1.3 Screen
% layout, 3.2.1.4 Function details). \fnhead prints the numbered field heading
% and moves it to the next page when fewer than six lines are left, so a
% heading is never stranded at the bottom of a page.

\newcommand{\fnhead}[1]{%
  \par
  \dimen0=\pagegoal \advance\dimen0 by -\pagetotal
  \ifdim\dimen0<6\baselineskip \newpage\fi
  \paragraph{#1}}

% Sub-label inside one field (Actors / Roles, Purpose, Input Data, ...).
\newcommand{\sublabel}[1]{%
  \par\addvspace{5pt}\noindent\detailhead{#1}%
  \par\nopagebreak}

% Business-rule lists. The identifier of every rule (BR-FORUM-01, ...) is set
% in bold. \brrulelist is issued right after the itemize starts: it captures
% the current \item and makes every following \item print its text up to the
% first colon in bold, so the identifiers are not marked up one by one and the
% source of the report keeps "BR-FORUM-01: ..." readable.
\let\brsaveitem\relax

\newcommand{\brrulelist}{%
  \let\brsaveitem\item
  \def\item{\brsaveitem\brgrabref}}

\def\brgrabref#1:{\textbf{#1}:}

% Two-column tables of the Function details field: input data, output data.
% The tables are short, so they are moved to the next page instead of being
% split with a single row left over.
\newenvironment{specdatatable}[2]{%
  \par\addvspace{3pt}%
  \dimen0=\pagegoal \advance\dimen0 by -\pagetotal
  \ifdim\dimen0<11\baselineskip \newpage\fi
  \renewcommand{\arraystretch}{1.22}%
  \begin{longtable}{|>{\raggedright\arraybackslash}p{0.26\linewidth}|p{0.62\linewidth}|}
  \hline\rowcolor{headerbg}\textbf{#1} & \textbf{#2}\\ \hline
  \endfirsthead
  \hline\rowcolor{headerbg}\textbf{#1} & \textbf{#2}\\ \hline
  \endhead
  \hline\endfoot
  }{%
  \hline
  \end{longtable}}

% Two-column table of the abnormal flow: case identifier and description.

  \endfirsthead
  \hline\rowcolor{headerbg}\textbf{#1} & \textbf{#2}\\ \hline
  \endhead
  \hline\endfoot
  }{%
  \hline
  \end{longtable}}
